   153→   153→/** Fetch the user's accounts as a plain array (defensive). Returns [] on
   154→   154→ *  any error or unexpected shape. */
   155→   155→async function fetchAccounts(accountId?: string) {
   156→   156→  try {
   157→   157→    const qs = new URLSearchParams({ ...(accountId ? { accountId } : {}) }).toString();
   158→   158→    const url = `/api/accounts${qs ? `?${qs}` : ""}`;
   159→   159→    const res = await fetch(url, { cache: "no-store" });
   160→   160→    if (!res.ok) return [];
   161→   161→    const data = await res.json();
   162→   162→    const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
   163→   163→    return Array.isArray(arr) ? arr : [];
   164→   164→  } catch {
   165→   165→    return [];
   166→   166→  }
   167→   167→}
   168→   168→
   169→   169→/** Fetch the user's strategies as a plain array (defensive). Returns [] on
   170→   170→ *  any error or unexpected shape. */
   171→   171→async function fetchStrategies() {
   172→   172→  try {
   173→   173→    const res = await fetch("/api/strategies", { cache: "no-store" });
   174→   174→    if (!res.ok) return [];
   175→   175→    const data = await res.json();
   176→   176→    const arr = data?.items ?? data?.strategies ?? (Array.isArray(data) ? data : []);
   177→   177→    return Array.isArray(arr) ? arr : [];
   178→   178→  } catch {
   179→   179→    return [];
   180→   180→  }
   181→   181→}

   183→   183→/** Parse a JSON string into the typed fallback; never throws. */
   184→   184→function safeJsonParse(json: string | null | undefined, fallback: any): any {
   185→   185→  if (!json) return fallback;
   186→   186→  try { return JSON.parse(json); } catch { return fallback; }
   187→   187→}

   189→   189→/** Format a numeric price with the given decimal precision (default 5,
   190→   190→ *  matching the historical forex-major behaviour). Returns "—" when the
   191→   191→ *  value is missing or non-finite. Pass the selected instrument's
   192→   192→ *  `pricePrecision` to render prices with the right number of decimals
   193→   193→ *  (e.g. 2 for gold, 1 for indices, 0 for YM). */
   194→   194→function fmtPrice(
   195→   195→  value: number | string | null | undefined,
   196→   196→  precision: number = 5,
   197→   197→): string {
   198→   198→  if (value == null || value === "") return "—";
   199→   199→  const n = typeof value === "string" ? Number(value) : value;
   200→   200→  if (!Number.isFinite(n)) return "—";
   201→   201→  const p =
   202→   202→    Number.isFinite(precision) && precision >= 0 && precision <= 8
   203→   203→      ? Math.round(precision)
   204→   204→      : 5;
   205→   205→  return n.toFixed(p);
   206→   206→}
   207→   207→
   208→   208→/** Format a money amount given a dollars-number (NOT cents). */
   209→   209→function fmtMoney(dollars: number | null | undefined, currency = "USD"): string {
   210→   210→  if (dollars == null || !Number.isFinite(dollars)) return "—";
   211→   211→  try {
   212→   212→    return new Intl.NumberFormat("en-US", {
   213→   213→      style: "currency",
   214→   214→      currency,
   215→   215→      minimumFractionDigits: 2,
   216→   216→      maximumFractionDigits: 2,
   217→   217→    }).format(dollars);
   218→   218→  } catch {
   219→   219→    return `$${dollars.toFixed(2)}`;
   220→   220→  }
   221→   221→}

   223→   223→/** Generate a short random id for client-side list items. */
   224→   224→function uid(): string {
   225→   225→  return Math.random().toString(36).slice(2, 10);
   226→   226→}

   704→   704→  /* ---------- File upload (non-blocking, concurrent, with progress) ----------
   705→   705→   *
   706→   706→   * Each selected file is added to `form.uploadedMedia` immediately as a
   707→   707→   * placeholder with `status: "uploading"` and a per-item `progress` (0-100).
   708→   708→   * Uploads run concurrently via XMLHttpRequest so progress events fire per
   709→   709→   * file without blocking the form (the old implementation awaited each
   710→   710→   * upload in a sequential for-loop). The File + XHR for each tempId live in
   711→   711→   * the `fileRefs` ref Map (not state), so Cancel can abort and Retry can
   712→   712→   * re-send without storing heavy File objects in React state.
   713→   713→   */
   714→   714→
   715→   715→  function handleFileUpload(files: FileList | File[]) {
   716→   716→    const fileList = Array.from(files);
   717→   717→    for (const file of fileList) {
   718→   718→      const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   719→   719→      // Detect kind from MIME type so the form immediately knows whether to
   720→   720→      // render an <img> (annotation-able) or a <video controls> element.
   721→   721→      const clientKind = file.type.startsWith("video/") ? "video" : "image";
   722→   722→      const mediaItem: any = {
   723→   723→        tempId,
   724→   724→        filename: file.name,
   725→   725→        originalName: file.name,
   726→   726→        kind: clientKind,
   727→   727→        status: "uploading",
   728→   728→        progress: 0,
   729→   729→        fileSize: file.size,
   730→   730→        annotations: [],
   731→   731→        caption: "",
   732→   732→        timeframe: "",
   733→   733→      };
   734→   734→      // Add the placeholder immediately so the user sees progress right
   735→   735→      // away, then kick off the upload in the background (non-blocking).
   736→   736→      setForm((f: any) => ({
   737→   737→        ...f,
   738→   738→        uploadedMedia: [...(f.uploadedMedia ?? []), mediaItem],
   739→   739→      }));
   740→   740→      uploadFile(file, tempId);
   741→   741→    }
   742→   742→  }

   744→   744→  function uploadFile(file: File, tempId: string) {
   745→   745→    const formData = new FormData();
   746→   746→    formData.append("file", file);
   747→   747→    if (editingId) formData.append("tradeId", editingId);
   748→   748→
   749→   749→    const xhr = new XMLHttpRequest();
   750→   750→    xhr.open("POST", "/api/media");
   751→   751→    fileRefs.current.set(tempId, { file, xhr });
   752→   752→
   753→   753→    // Real upload progress — only XMLHttpRequest exposes this; fetch()
   754→   754→    // ReadableStream request bodies don't have progress events yet.
   755→   755→    xhr.upload.onprogress = (e) => {
   756→   756→      if (e.lengthComputable) {
   757→   757→        const progress = Math.round((e.loaded / e.total) * 100);
   758→   758→        setForm((f: any) => ({
   759→   759→          ...f,
   760→   760→          uploadedMedia: f.uploadedMedia.map((m: any) =>
   761→   761→            m.tempId === tempId ? { ...m, progress } : m,
   762→   762→          ),
   763→   763→        }));
   764→   764→      }
   765→   765→    };
   766→   766→
   767→   767→    xhr.onload = () => {
   768→   768→      if (xhr.status >= 200 && xhr.status < 300) {
   769→   769→        try {
   770→   770→          const data = JSON.parse(xhr.responseText);
   771→   771→          // Fetch the signed URL so the preview can render immediately.
   772→   772→          fetch(`/api/media/${data.id}`)
   773→   773→            .then((r) => r.json())
   774→   774→            .then((detail) => {
   775→   775→              setForm((f: any) => ({
   776→   776→                ...f,
   777→   777→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   778→   778→                  m.tempId === tempId
   779→   779→                    ? {
   780→   780→                        ...m,
   781→   781→                        ...data,
   782→   782→                        id: data.id,
   783→   783→                        url: detail.url,
   784→   784→                        status: "uploaded",
   785→   785→                        progress: 100,
   786→   786→                      }
   787→   787→                    : m,
   788→   788→                ),
   789→   789→              }));
   790→   790→            })
   791→   791→            .catch(() => {
   792→   792→              // Signed-URL fetch failed — still mark uploaded with raw data.
   793→   793→              setForm((f: any) => ({
   794→   794→                ...f,
   795→   795→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   796→   796→                  m.tempId === tempId
   797→   797→                    ? {
   798→   798→                        ...m,
   799→   799→                        ...data,
   800→   800→                        id: data.id,
   801→   801→                        status: "uploaded",
   802→   802→                        progress: 100,
   803→   803→                      }
   804→   804→                    : m,
   805→   805→                ),
   806→   806→              }));
   807→   807→            });
   808→   808→          // Upload finished — drop the File/XHR ref (Retry not needed).
   809→   809→          fileRefs.current.delete(tempId);
   810→   810→          toast.success(`${file.name} uploaded.`);
   811→   811→        } catch {
   812→   812→          markUploadFailed(tempId, file.name);
   813→   813→        }
   814→   814→      } else {
   815→   815→        // Surface the server's error message when present (e.g. "File too
   816→   816→        // large. Max 20 MB.").
   817→   817→        let msg = `Failed to upload ${file.name}`;
   818→   818→        try {
   819→   819→          const err = JSON.parse(xhr.responseText);
   820→   820→          if (err?.error) msg = err.error;
   821→   821→        } catch {}
   822→   822→        markUploadFailed(tempId, file.name, msg);
   823→   823→      }
   824→   824→    };
   825→   825→
   826→   826→    xhr.onerror = () => markUploadFailed(tempId, file.name);
   827→   827→    // onabort is fired when cancelUpload calls xhr.abort(); cancelUpload
   828→   828→    // already removes the item from state and clears the ref, so this is
   829→   829→    // a no-op kept for completeness.
   830→   830→    xhr.onabort = () => {};
   831→   831→
   832→   832→    xhr.send(formData);
   833→   833→  }

   835→   835→  function markUploadFailed(tempId: string, filename: string, msg?: string) {
   836→   836→    setForm((f: any) => ({

   937→   937→  /* ---------- Risk & sizing calculations ---------- */
   938→   938→
   939→   939→  const account = accounts.find((a: any) => a.id === form.accountId);
   940→   940→  const accountCurrency = account?.currency ?? "USD";
   941→   941→  const balanceCents = account?.currentBalanceCents ?? account?.startingBalanceCents ?? 0;
   942→   942→
   943→   943→  const entryNum = Number(form.plannedEntryPrice);
   944→   944→  const stopNum = Number(form.plannedStopPrice);
   945→   945→  const targetNum = Number(form.plannedTargetPrice);
   946→   946→  const lotNum = Number(form.lotSize) || 1;
   947→   947→  const riskPctNum = Number(form.riskPct);
   948→   948→
   949→   949→  const hasEntry = Number.isFinite(entryNum) && form.plannedEntryPrice !== "";
   950→   950→  const hasStop = Number.isFinite(stopNum) && form.plannedStopPrice !== "";
   951→   951→  const hasTarget = Number.isFinite(targetNum) && form.plannedTargetPrice !== "";
   952→   952→
   953→   953→  // Stop distance (always positive).
   954→   954→  const stopDistance = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   955→   955→
   956→   956→  // Risk per unit / reward per unit (per 1 lot).
   957→   957→  const riskPerUnit = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   958→   958→  const rewardPerUnit =
   959→   959→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;
   960→   960→
   961→   961→  // Planned R:R = reward / risk (1.0 = breakeven, 2.0 = 2R for 1R).
   962→   962→  const plannedRR =
   963→   963→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
   964→   964→      ? rewardPerUnit / riskPerUnit
   965→   965→      : null;
   966→   966→
   967→   967→  // Risk amount ($) = balance * riskPct.
   968→   968→  const riskAmountDollars =
   969→   969→    Number.isFinite(riskPctNum) && balanceCents > 0
   970→   970→      ? (balanceCents / 100) * riskPctNum
   971→   971→      : null;
   972→   972→
   973→   973→  // Planned profit ($) = riskAmount * RR (if RR is known).
   974→   974→  const plannedProfitDollars =
   975→   975→    riskAmountDollars != null && plannedRR != null
   976→   976→      ? riskAmountDollars * plannedRR
   977→   977→      : null;
   978→   978→
   979→   979→  /* ---------- Instrument-aware risk preview ---------- */
   980→   980→  // Uses pipSize, contractSize, and pointValueCents from the catalog to
   981→   981→  // express the stop distance in the instrument's native unit (pips /
   982→   982→  // ticks / points) and to derive a per-instrument dollar risk that is
   983→   983→  // consistent with the server-side calculateTradePnl formula:
   984→   984→  //   risk$ = stopDistance × qty × contractSize × (pointValueCents/100)
   985→   985→  // (See computePointValueCents in src/lib/calculations.ts for the
   986→   986→  // convention used to derive pointValueCents from cs × tickSize.)
   987→   987→  const stopDistanceUnits = stopDistance != null
   988→   988→    ? distanceInUnits(stopDistance, selectedInstrument.pipSize)
   989→   989→    : 0;
   990→   990→  const contractSizeNum = Number(selectedInstrument.contractSize) || 1;
   991→   991→  const pointValueCents = selectedInstrument.pointValueCents;
   992→   992→  const instrumentRiskDollars =
   993→   993→    stopDistance != null
   994→   994→      ? stopDistance * lotNum * contractSizeNum * (pointValueCents / 100)
   995→   995→      : null;
   996→   996→
   997→   997→  /* ---------- Partial exits calculations ---------- */
   998→   998→
   999→   999→  const validExits = form.partialExits.filter(
  1000→  1000→    (e: any) => e.price !== "" && Number.isFinite(Number(e.price)),
  1001→  1001→  );

  1003→  1003→    validExits.length > 0
  1004→  1004→      ? validExits.reduce((sum: number, e: any) => sum + Number(e.price), 0) /
  1005→  1005→        validExits.length
  1006→  1006→      : null;
  1007→  1007→  const totalRExits =
  1008→  1008→    weightedExitPrice != null && riskPerUnit != null && riskPerUnit > 0
  1009→  1009→      ? form.direction === "long"
  1010→  1010→        ? (weightedExitPrice - entryNum) / riskPerUnit
  1011→  1011→        : (entryNum - weightedExitPrice) / riskPerUnit
  1012→  1012→      : null;
  1013→  1013→
  1014→  1014→  /* ---------- Evidence grouping (by chart timeframe) ---------- */
  1015→  1015→
  1016→  1016→  // Bucket each uploaded media item into a timeframe group so the user can
  1017→  1017→  // see at a glance whether their HTF context, execution chart, and entry
  1018→  1018→  // trigger are all captured. Items with no timeframe selected land in an
  1019→  1019→  // "Unassigned" bucket first to nudge the user toward picking one.
  1020→  1020→  const groupedMedia = useMemo(() => {
  1021→  1021→    const items: Array<{ m: any; i: number }> = (form.uploadedMedia ?? []).map(
  1022→  1022→      (m: any, i: number) => ({ m, i }),

  1025→  1025→    for (const entry of items) {
  1026→  1026→      const g = entry.m.timeframe ? getTimeframeGroup(entry.m.timeframe) : null;
  1027→  1027→      const key = g ?? "__unassigned";
  1028→  1028→      if (!buckets[key]) buckets[key] = [];
  1029→  1029→      buckets[key].push(entry);
  1030→  1030→    }
  1031→  1031→    const ordered: Array<{ group: string; items: Array<{ m: any; i: number }> }> = [];
  1032→  1032→    if (buckets["__unassigned"]) {
  1033→  1033→      ordered.push({ group: "Unassigned", items: buckets["__unassigned"] });
  1034→  1034→    }

  1037→  1037→    }
  1038→  1038→    // Any leftover custom groups (shouldn't normally happen, but defensive).
  1039→  1039→    for (const key of Object.keys(buckets)) {

  1041→  1041→      if (TIMEFRAME_GROUPS.includes(key)) continue;
