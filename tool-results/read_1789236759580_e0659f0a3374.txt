    46→async function fetchTrade(id: string) {
    47→  const res = await fetch(`/api/trades/${id}`, { cache: "no-store" });
    48→  if (!res.ok) throw new Error("Failed to load trade");
    49→  return res.json();
    50→}

    56→  const { data: trade, isLoading } = useQuery({
    57→    queryKey: ["trade", id],
    58→    queryFn: () => fetchTrade(id),
    59→    enabled: !!id,
    60→  });

    64→  const [executeTradeOpen, setExecuteTradeOpen] = useState(false);
    65→

    74→    if (!trade) return null;
    75→    const evals = Array.isArray(trade.checklistEvaluations)
    76→      ? trade.checklistEvaluations
    77→      : [];

    86→      checklistName,
    87→      versionLabel: version.versionLabel,
    88→      weightedScore: latest.weightedScore,
    89→      finalGrade: latest.finalGrade,
    90→      evaluatedAt: latest.evaluatedAt,
    91→      items,
    92→      answers,
    93→    };
    94→  }, [trade]);

    96→  if (isLoading) {
    97→    return (
    98→      <div className="p-4 md:p-6 space-y-3">
    99→        <Skeleton className="h-8 w-32" />
   100→        <Skeleton className="h-24" />
   101→        <Skeleton className="h-64" />
   102→      </div>
   103→    );
   104→  }
   105→
   106→  if (!trade) return null;
   107→
   108→  const pnl = formatSignedCents(trade.netPnlCents);

   110→  const thesis = safeJson(trade.thesisJson, {});
   111→  const psychBefore = safeJson(trade.psychBeforeJson, {});
   112→  const psychAfter = safeJson(trade.psychAfterJson, {});

   114→  const tags: string[] = safeJson(trade.tagsJson, []);

   138→        </Button>

   142→            <Badge variant="outline" className={cn(trade.direction === "long" ? "text-profit" : "text-loss")}>

   154→              variant="secondary"
   155→              className={cn(
   156→                "capitalize",
   157→                trade.status === "planned" && "bg-primary/10 text-primary",
   158→                trade.status === "open" && "bg-warning/15 text-warning",
   159→              )}
   160→            >
   161→              {trade.status}
   162→            </Badge>
   163→          </div>

   165→            {trade.entryTime ? new Date(trade.entryTime).toLocaleString() : "Open"} · {trade.session ?? "-"}
   166→          </div>
   167→        </div>
   168→        <div className="text-right">
   169→          <div className={cn("text-xl font-semibold tnum", pnl.sign > 0 ? "text-profit" : pnl.sign < 0 ? "text-loss" : "")}>
   170→            {pnl.text}
   171→          </div>
   172→          <div className={cn("text-sm tnum", trade.actualR && dToNumber(trade.actualR) > 0 ? "text-profit" : trade.actualR && dToNumber(trade.actualR) < 0 ? "text-loss" : "text-muted-foreground")}>
   173→            {formatR(trade.actualR)}
   174→          </div>
   175→        </div>
   176→      </div>
   177→

   192→            variant="default"

   254→        <MiniStat label="Costs" value={centsToNumberString(trade.feesCents + trade.commissionCents + trade.swapCents)} />
   255→        <MiniStat label="Strategy" value={strategyLabel} />

   257→

   262→          <TabsTrigger value="setup">Setup</TabsTrigger>
   263→          <TabsTrigger value="execution">Execution</TabsTrigger>
   264→          <TabsTrigger value="psychology">Psychology</TabsTrigger>
   265→          <TabsTrigger value="evidence">Evidence</TabsTrigger>
   266→          <TabsTrigger value="review">Review</TabsTrigger>
   267→        </TabsList>
   268→
   269→        <TabsContent value="overview" className="space-y-4 mt-3">
   270→          <Card className="p-4">
   271→            <h3 className="font-medium mb-2">Trading Thesis</h3>
   272→            {thesis?.why ? (
   273→              <p className="text-sm text-muted-foreground leading-relaxed">{thesis.why}</p>
   274→            ) : (
   275→              <p className="text-sm text-muted-foreground">No thesis recorded.</p>
   276→            )}
   277→            {thesis?.narrative && (
   278→              <div className="mt-3 text-sm">
   279→                <span className="text-muted-foreground">Narrative:</span> {thesis.narrative}
   280→              </div>
   281→            )}
   282→            <div className="mt-3 grid sm:grid-cols-2 gap-2 text-sm">
   283→              {thesis?.liquidityTarget && <div><span className="text-muted-foreground">Liquidity target:</span> {thesis.liquidityTarget}</div>}
   284→              {thesis?.target && <div><span className="text-muted-foreground">Target:</span> {thesis.target}</div>}
   285→            </div>
   286→            {thesis?.confirms && Array.isArray(thesis.confirms) && thesis.confirms.length > 0 && (
   287→              <div className="mt-3 text-sm">
   288→                <span className="text-muted-foreground">Confirmations:</span>
   289→                <div className="flex flex-wrap gap-1 mt-1">
   290→                  {thesis.confirms.map((c: string) => <Badge key={c} variant="secondary" className="text-xs">{c}</Badge>)}
   291→                </div>
   292→              </div>
   293→            )}
   294→            {thesis?.invalidates && Array.isArray(thesis.invalidates) && thesis.invalidates.length > 0 && (
   295→              <div className="mt-3 text-sm">
   296→                <span className="text-muted-foreground">Invalidations:</span>
   297→                <div className="flex flex-wrap gap-1 mt-1">
   298→                  {thesis.invalidates.map((c: string) => <Badge key={c} variant="outline" className="text-xs text-loss">{c}</Badge>)}
   299→                </div>

   303→
   304→          {tags.length > 0 && (

   308→                {tags.map((t) => <Badge key={t} variant="secondary">{t}</Badge>)}
   309→              </div>
   310→            </Card>
   311→          )}
   312→
   313→          {behaviorFlags.length > 0 && (
   314→            <Card className="p-4 border-warning/50">
   315→              <h3 className="font-medium mb-2 flex items-center gap-2"><AlertTriangle className="h-4 w-4 text-warning" /> Behavior Flags</h3>
   316→              <div className="flex flex-wrap gap-1">
   317→                {behaviorFlags.map((f) => (
   318→                  <Badge key={f} variant="outline" className="text-warning">
   319→                    {f.split("_").map((w) => w.charAt(0).toUpperCase() + w.slice(1)).join(" ")}
   320→                  </Badge>
   321→                ))}
   322→              </div>
   323→            </Card>
   324→          )}
   325→
   326→          {trade.notes && (
   327→            <Card className="p-4">
   328→              <h3 className="font-medium mb-2">Notes</h3>
   329→              <p className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">{trade.notes}</p>
   330→            </Card>
   331→          )}
   332→        </TabsContent>
   333→
   334→        <TabsContent value="setup" className="space-y-4 mt-3">
   335→          <Card className="p-4">
   336→            <h3 className="font-medium mb-3">ICT Setup</h3>
   337→            <div className="grid sm:grid-cols-2 gap-4 text-sm">
   338→              <SetupSection title="Higher Timeframe Context" data={setup?.htfContext} />
   339→              <SetupSection title="Liquidity" data={setup?.liquidity} isList />
   340→              <SetupSection title="Structure" data={setup?.structure} isList />
   341→              <SetupSection title="Entry Model" data={setup?.entryModel} />
   342→              <SetupSection title="Session" data={setup?.session} />
   343→              <SetupSection title="Setup Score" data={trade.setupScore ? `${(dToNumber(trade.setupScore) * 100).toFixed(0)}%` : null} hint="Calculated from the weighted rules defined by your strategy" />
   344→            </div>
   345→          </Card>
   346→
   347→          {/* Checklist Evaluation (Fix 3). Shows the snapshot of the A+
   348→              checklist answers + the computed score/grade at the time the
   349→              trade was saved. The evaluation row is created server-side on
   350→              POST/PATCH /api/trades/[id] (see checklist-evaluation.ts). */}
   351→          {checklistEvaluation ? (
   352→            <Card className="p-4 space-y-4">
   353→              <div className="flex items-start justify-between gap-3 flex-wrap">
   354→                <div className="min-w-0">
   355→                  <h3 className="font-medium flex items-center gap-2">
   356→                    Checklist Evaluation
   357→                    <Sparkles className="h-4 w-4 text-primary" />
   358→                  </h3>
   359→                  <p className="text-xs text-muted-foreground mt-0.5">

   371→                    {(dToNumber(checklistEvaluation.weightedScore) * 100).toFixed(1)}%

   458→              </div>

   461→        </TabsContent>
   462→
   463→        <TabsContent value="execution" className="space-y-4 mt-3">

   470→                    <th className="px-3 py-2 text-left">Type</th>

   714→function MiniStat({ label, value, tone, hint }: { label: string; value: string; tone?: "profit" | "loss" | "neutral"; hint?: string }) {

   746→function SetupSection({ title, data, isList, hint }: { title: string; data: any; isList?: boolean; hint?: string }) {
   747→  const [showHint, setShowHint] = useState(false);
   748→  if (!data) return null;
   749→  return (
   750→    <div>
   751→      <div className="text-xs text-muted-foreground uppercase tracking-wide flex items-center gap-1.5">
   752→        {title}
   753→        {hint && (
   754→          <span
   755→            className="relative inline-flex normal-case"
   756→            onMouseEnter={() => setShowHint(true)}
   757→            onMouseLeave={() => setShowHint(false)}
   758→            onClick={(e) => { e.preventDefault(); setShowHint((s) => !s); }}
   759→          >
   760→            <HelpCircle className="h-3 w-3 text-muted-foreground cursor-help" />
   761→            {showHint && (
   762→              <span
   763→                className="absolute left-0 top-4 z-50 w-48 rounded-md border border-border bg-popover p-2 text-xs font-normal normal-case text-popover-foreground shadow-md"
   764→                role="tooltip"
   765→              >
   766→                {hint}
   767→              </span>
   768→            )}
   769→          </span>
   770→        )}
   771→      </div>
   772→      {isList && Array.isArray(data) ? (
   773→        <div className="flex flex-wrap gap-1 mt-1">
   774→          {data.map((d: string, i: number) => <Badge key={i} variant="secondary" className="text-xs">{d}</Badge>)}
   775→        </div>
   776→      ) : typeof data === "object" ? (
   777→        <div className="mt-1 text-sm">
   778→          {Object.entries(data).map(([k, v]) => (
   779→            <div key={k}>
   780→              <span className="text-muted-foreground">{k}:</span> {String(v)}
   781→            </div>
   782→          ))}
   783→        </div>
   784→      ) : (
   785→        <div className="font-medium mt-1">{String(data)}</div>
   786→      )}
   787→    </div>
   788→  );
   789→}
