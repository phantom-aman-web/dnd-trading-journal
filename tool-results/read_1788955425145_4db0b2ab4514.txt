   111→   111→
   112→   112→async function fetchDashboard(preset: string, accountId?: string): Promise<DashboardResponse> {
   113→   113→  const params = new URLSearchParams({ preset });
   114→   114→  if (accountId) params.set("accountId", accountId);
   115→   115→  const res = await fetch(`/api/dashboard?${params.toString()}`, { cache: "no-store" });
   116→   116→  if (!res.ok) throw new Error("Failed to load dashboard");
   117→   117→  return res.json();
   118→   118→}
   119→   119→
   120→   120→async function fetchRecentTrades(accountId?: string): Promise<TradesResponse> {
   121→   121→  const params = new URLSearchParams({ limit: "10", sortBy: "entryTime", sortDir: "desc" });
   122→   122→  if (accountId) params.set("accountId", accountId);
   123→   123→  const res = await fetch(`/api/trades?${params.toString()}`, { cache: "no-store" });
   124→   124→  if (!res.ok) throw new Error("Failed to load trades");
   125→   125→  return res.json();
   126→   126→}

   132→   132→function safeParseArr(json: string | null | undefined): string[] {
   133→   133→  if (!json) return [];
   134→   134→  try {
   135→   135→    const v = JSON.parse(json);
   136→   136→    return Array.isArray(v) ? v : [];
   137→   137→  } catch {
   138→   138→    return [];
   139→   139→  }
   140→   140→}
   141→   141→
   142→   142→/** Per-trade adherence: 100% when no behavior flags are recorded, else 0%. */
   143→   143→function tradeAdherencePct(trade: TradeRow): number {
   144→   144→  return safeParseArr(trade.behaviorFlagsJson).length === 0 ? 100 : 0;
   145→   145→}

   147→   147→function formatDate(iso: string | null): string {
   148→   148→  if (!iso) return "—";
   149→   149→  const d = new Date(iso);
   150→   150→  if (Number.isNaN(d.getTime())) return "—";
   151→   151→  return d.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "2-digit" });
   152→   152→}

   164→   164→    queryKey: ["dashboard", preset, accountId],
   165→   165→    queryFn: () => fetchDashboard(preset, accountId),

   167→   167→
   168→   168→  const { data: tradesData, isLoading: tradesLoading } = useQuery({
   169→   169→    queryKey: ["dashboard-recent-trades", accountId],
   170→   170→    queryFn: () => fetchRecentTrades(accountId),
   171→   171→  });
   172→   172→
   173→   173→  const recentTrades: TradeRow[] = Array.isArray(tradesData) ? tradesData : (tradesData?.items ?? []);
   174→   174→
   175→   175→  // Adherence breakdown derived from BOTH the dashboard's period-correct
   176→   176→  // aggregate + byBehavior (rule violations + top violated rules) AND the
   177→   177→  // recent-trades snapshot (recent adherence rate).
   178→   178→  const breakdown = useMemo(() => {
   179→   179→    const totalTrades = dash?.aggregate.totalTrades ?? 0;
   180→   180→    const byBehavior = dash?.byBehavior ?? [];
   181→   181→
   182→   182→    // Total violation instances (a trade with 2 flags counts as 2).
   183→   183→    const totalViolations = byBehavior.reduce((s, b) => s + (b.trades ?? 0), 0);
   184→   184→
   185→   185→    // Top 3 violated rules sorted by trade count, then P&L impact.
   186→   186→    const topViolated = [...byBehavior]
   187→   187→      .sort((a, b) => b.trades - a.trades || Math.abs(b.pnlCents) - Math.abs(a.pnlCents))
   188→   188→      .slice(0, 3);
   189→   189→
   190→   190→    // Recent-trades adherence snapshot — fraction of the recent 10 trades
   191→   191→    // that have zero behavior flags. Reported as a percentage.
   192→   192→    const recentCount = recentTrades.length;
   193→   193→    const recentCompliant = recentTrades.filter(
   194→   194→      (t) => safeParseArr(t.behaviorFlagsJson).length === 0,
   195→   195→    ).length;
   196→   196→    const recentAdherenceRate =
   197→   197→      recentCount > 0 ? Math.round((recentCompliant / recentCount) * 100) : null;
   198→   198→
   199→   199→    return { totalTrades, totalViolations, topViolated, recentAdherenceRate, recentCount };
   200→   200→  }, [dash, recentTrades]);
   201→   201→
   202→   202→  const isLoading = dashLoading || tradesLoading;
   203→   203→  const hasNoData = !isLoading && breakdown.totalTrades === 0 && recentTrades.length === 0;
   204→   204→
   205→   205→  if (isLoading) {
   206→   206→    return <DashboardSkeleton />;
   207→   207→  }
   208→   208→
   209→   209→  if (hasNoData) {
   210→   210→    return (

   212→   212→        <DashboardHeader preset={preset} setPreset={setPreset} />
   213→   213→        <EmptyDashboard onLogTrade={() => navigate("tradeNew")} />
   214→   214→      </div>
   215→   215→    );
   216→   216→  }
   217→   217→
   218→   218→  return (
   219→   219→    <div className="p-4 md:p-6 space-y-6">
   220→   220→      <DashboardHeader preset={preset} setPreset={setPreset} />
   221→   221→
   222→   222→      {/* Section 1 — Process & Adherence Breakdown */}
   223→   223→      <section className="space-y-3">
   224→   224→        <SectionTitle
   225→   225→          icon={ShieldCheck}
   226→   226→          title="PROCESS & ADHERENCE BREAKDOWN"
   227→   227→          subtitle="How well are you following your trading checklist?"
   228→   228→        />
   229→   229→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   230→   230→          <MetricTile
   231→   231→            label="Total Trades"
   232→   232→            value={String(breakdown.totalTrades)}
   233→   233→            sub={dash?.aggregate ? `${dash.aggregate.wins}W · ${dash.aggregate.losses}L` : undefined}
   234→   234→            icon={ClipboardList}
   235→   235→          />
   236→   236→          <MetricTile
   237→   237→            label="Adherence Rate"
   238→   238→            value={
   239→   239→              breakdown.recentAdherenceRate != null
   240→   240→                ? `${breakdown.recentAdherenceRate}%`
   241→   241→                : "—"
   242→   242→            }
   243→   243→            sub={
   244→   244→              breakdown.recentCount > 0
   245→   245→                ? `last ${breakdown.recentCount} trades`
   246→   246→                : undefined
   247→   247→            }
   248→   248→            icon={CheckCircle2}
   249→   249→            tone={
   250→   250→              breakdown.recentAdherenceRate == null
   251→   251→                ? "neutral"
   252→   252→                : breakdown.recentAdherenceRate >= 80
   253→   253→                  ? "profit"
   254→   254→                  : breakdown.recentAdherenceRate < 50
   255→   255→                    ? "loss"
   256→   256→                    : "neutral"
   257→   257→            }
   258→   258→          />
   259→   259→          <MetricTile
   260→   260→            label="Rule Violations"
   261→   261→            value={String(breakdown.totalViolations)}
   262→   262→            sub={
   263→   263→              breakdown.totalViolations > 0
   264→   264→                ? `${dash?.byBehavior?.length ?? 0} distinct rule${(dash?.byBehavior?.length ?? 0) === 1 ? "" : "s"}`
   265→   265→                : "no violations"
   266→   266→            }
   267→   267→            icon={AlertTriangle}
   268→   268→            tone={breakdown.totalViolations > 0 ? "loss" : "profit"}
   269→   269→          />
   270→   270→          <MetricTile
   271→   271→            label="Net P&L"
   272→   272→            value={formatSignedCents(dash?.aggregate.totalPnlCents ?? 0).text}
   273→   273→            sub={dash?.aggregate ? `${dash.aggregate.closedTrades} closed` : undefined}
   274→   274→            icon={Activity}
   275→   275→            tone={
   276→   276→              (dash?.aggregate.totalPnlCents ?? 0) > 0
   277→   277→                ? "profit"
   278→   278→                : (dash?.aggregate.totalPnlCents ?? 0) < 0
   279→   279→                  ? "loss"
   280→   280→                  : "neutral"
   281→   281→            }
   282→   282→          />
   283→   283→        </div>
   284→   284→
   285→   285→        {/* Top violated rules */}

   295→   295→          </div>

   300→   300→            </div>
   301→   301→          ) : (
   302→   302→            <ul className="divide-y divide-border">
   303→   303→              {breakdown.topViolated.map((rule) => {
   304→   304→                const pnl = formatSignedCents(rule.pnlCents);
   305→   305→                return (
   306→   306→                  <li
   307→   307→                    key={rule.key}

   332→   332→            </ul>

   337→   337→      {/* Section 2 — Performance Overview */}

   366→   366→        </div>
   367→   367→
   368→   368→        <Card className="bg-card border border-border rounded-xl shadow-sm overflow-hidden p-0">
   369→   369→          {recentTrades.length === 0 ? (
   370→   370→            <div className="p-8 text-center">
   371→   371→              <p className="text-sm text-muted-foreground mb-3">No trades logged yet.</p>
   372→   372→              <Button

   374→   374→                className="bg-primary hover:bg-primary/90 text-primary-foreground"

   376→   376→              >
   377→   377→                <Plus className="h-3.5 w-3.5" />
   378→   378→                Log Trade
   379→   379→              </Button>
   380→   380→            </div>
   381→   381→          ) : (

   383→   383→              {/* Desktop / tablet table */}
