   446→function AccountCreateDialog({ onCreated }: { onCreated: () => void }) {
   447→  const [open, setOpen] = useState(false);
   448→  const [name, setName] = useState("");
   449→  const [broker, setBroker] = useState("");
   450→  const [currency, setCurrency] = useState("USD");
   451→  const [balance, setBalance] = useState("10000");
   452→  const [isDefault, setIsDefault] = useState(false);
   453→
   454→  async function submit() {
   455→    const res = await fetch("/api/accounts", {
   456→      method: "POST",
   457→      headers: { "Content-Type": "application/json" },
   458→      body: JSON.stringify({ name, broker, currency, startingBalanceCents: Number(balance) * 100, isDefault }),
   459→    });
   460→    if (res.ok) { toast.success("Account created."); setOpen(false); onCreated(); setName(""); }
   461→    else toast.error("Failed to create account.");
   462→  }
   463→
   464→  return (
   465→    <Dialog open={open} onOpenChange={setOpen}>
   466→      <DialogTrigger asChild><Button size="sm" variant="outline"><Plus className="h-4 w-4" /> Add</Button></DialogTrigger>
   467→      <DialogContent>
   468→        <DialogHeader><DialogTitle>New Account</DialogTitle></DialogHeader>
   469→        <div className="space-y-3">
   470→          <div className="space-y-2">
   471→            <FieldLabel required>Account Name</FieldLabel>
   472→            <Input value={name} onChange={(e) => setName(e.target.value)} />
   473→          </div>
   474→          <div className="grid grid-cols-2 gap-3">
   475→            <div className="space-y-2">
   476→              <FieldLabel hint="Broker name">Broker</FieldLabel>
   477→              <Input value={broker} onChange={(e) => setBroker(e.target.value)} />
   478→            </div>
   479→            <div className="space-y-2">
   480→              <FieldLabel required>Currency</FieldLabel>
   481→              <Select value={currency} onValueChange={setCurrency}>
   482→                <SelectTrigger><SelectValue /></SelectTrigger>
   483→                <SelectContent>
   484→                  <SelectItem value="USD">USD</SelectItem>
   485→                  <SelectItem value="EUR">EUR</SelectItem>
   486→                  <SelectItem value="GBP">GBP</SelectItem>

   573→function ImportCsvDialog({ onImported }: { onImported: () => void }) {
   574→  const [open, setOpen] = useState(false);
   575→  const [step, setStep] = useState(1);
   576→  const [filename, setFilename] = useState("");
   577→  const [content, setContent] = useState("");
   578→  const [importId, setImportId] = useState<string | null>(null);
   579→  const [headers, setHeaders] = useState<string[]>([]);
   580→  const [sampleRows, setSampleRows] = useState<Record<string, string>[]>([]);
   581→  const [mapping, setMapping] = useState<Record<string, string>>({});
   582→  const [validation, setValidation] = useState<any>(null);
   583→
   584→  const FIELDS = ["instrumentSymbol", "direction", "entryPrice", "exitPrice", "stopPrice", "targetPrice", "quantity", "entryTime", "exitTime", "session", "setupGrade", "feesCents", "commissionCents", "swapCents"];
   585→
   586→  async function upload() {
   587→    const res = await fetch("/api/imports", {
   588→      method: "POST",
   589→      headers: { "Content-Type": "application/json" },
   590→      body: JSON.stringify({ filename, source: "csv", content }),
   591→    });
   592→    if (!res.ok) { toast.error("Upload failed."); return; }
   593→    const data = await res.json();
   594→    setImportId(data.id);
   595→    const summary = JSON.parse(data.summaryJson ?? "{}");
   596→    setHeaders(summary.headers ?? []);
   597→    setSampleRows(summary.sampleRows ?? []);
   598→    setStep(2);
   599→  }
   600→
   601→  async function validateImport() {
   602→    const res = await fetch(`/api/imports/${importId}`, {
   603→      method: "PATCH",
   604→      headers: { "Content-Type": "application/json" },
   605→      body: JSON.stringify({ action: "map", mapping }),
   606→    });
   607→    if (!res.ok) { toast.error("Validation failed."); return; }
   608→    const data = await res.json();
   609→    setValidation(data);
   610→    setStep(3);
   611→  }
   612→
   613→  async function confirmImport() {
   614→    const res = await fetch(`/api/imports/${importId}`, {
   615→      method: "PATCH",
   616→      headers: { "Content-Type": "application/json" },
   617→      body: JSON.stringify({ action: "confirm" }),
   618→    });
   619→    if (!res.ok) { toast.error("Import failed."); return; }
   620→    const data = await res.json();
   621→    toast.success(`Imported ${data.imported} trades.`);
   622→    setOpen(false);
   623→    setStep(1);
   624→    onImported();
   625→  }
   626→
   627→  function onFile(file: File) {
   628→    setFilename(file.name);
   629→    const reader = new FileReader();
   630→    reader.onload = () => setContent(String(reader.result));
   631→    reader.readAsText(file);
   632→  }
   633→
   634→  return (
   635→    <Dialog open={open} onOpenChange={(o) => { setOpen(o); if (!o) setStep(1); }}>
   636→      <DialogTrigger asChild><Button size="sm" variant="outline">Import CSV</Button></DialogTrigger>
   637→      <DialogContent className="max-w-3xl max-h-[85vh] overflow-y-auto">
   638→        <DialogHeader><DialogTitle>Import Trades from CSV</DialogTitle></DialogHeader>
   639→        {step === 1 && (
   640→          <div className="space-y-3">
   641→            <div className="space-y-2">
   642→              <FieldLabel required>CSV File</FieldLabel>
   643→              <Input type="file" accept=".csv" onChange={(e) => e.target.files?.[0] && onFile(e.target.files[0])} />
   644→            </div>
   645→            <div className="rounded-md border border-border p-3 text-xs text-muted-foreground">
   646→              Expected columns: instrumentSymbol, direction (long/short), entryPrice, exitPrice, stopPrice, targetPrice, quantity, entryTime, exitTime, session, setupGrade, feesCents, commissionCents, swapCents.
   647→              A header row is required.
   648→            </div>
   649→            <Button onClick={upload} disabled={!content}>Upload & Detect</Button>
   650→          </div>
   651→        )}
   652→        {step === 2 && (
   653→          <div className="space-y-3">
   654→            <h3 className="font-medium text-sm">Map columns</h3>
   655→            <div className="space-y-1 max-h-48 overflow-y-auto">
   656→              {headers.map((h) => (
   657→                <div key={h} className="flex items-center gap-2">
   658→                  <span className="text-xs w-32 truncate">{h}</span>
   659→                  <span className="text-muted-foreground">→</span>
   660→                  <Select
   661→                    value={mapping[h] ?? ""}
   662→                    onValueChange={(v) => setMapping((m) => ({ ...m, [h]: v }))}
   663→                  >
   664→                    <SelectTrigger className="h-7 text-xs"><SelectValue /></SelectTrigger>
   665→                    <SelectContent>
   666→                      {FIELDS.map((f) => <SelectItem key={f} value={f}>{f}</SelectItem>)}
   667→                    </SelectContent>
   668→                  </Select>
   669→                </div>
   670→              ))}
   671→            </div>
   672→            <Button onClick={validateImport}>Validate</Button>
   673→          </div>
   674→        )}
   675→        {step === 3 && validation && (
   676→          <div className="space-y-3">
   677→            <div className="grid grid-cols-3 gap-2 text-center">
   678→              <div className="rounded-md border border-border p-3">
   679→                <div className="text-2xl font-semibold text-profit tnum">{validation.valid}</div>
   680→                <div className="text-xs text-muted-foreground">Valid</div>
   681→              </div>
   682→              <div className="rounded-md border border-border p-3">
   683→                <div className="text-2xl font-semibold text-warning tnum">{validation.duplicates}</div>
   684→                <div className="text-xs text-muted-foreground">Duplicates (skipped)</div>
   685→              </div>
   686→              <div className="rounded-md border border-border p-3">
   687→                <div className="text-2xl font-semibold text-loss tnum">{validation.invalid}</div>
   688→                <div className="text-xs text-muted-foreground">Invalid (skipped)</div>
   689→              </div>
   690→            </div>
   691→            {validation.issues.length > 0 && (
   692→              <div className="rounded-md border border-warning/50 bg-warning/5 p-3 max-h-32 overflow-y-auto">
   693→                <div className="text-xs font-medium text-warning mb-1">Issues:</div>
   694→                <ul className="text-xs text-muted-foreground space-y-0.5">
   695→                  {validation.issues.slice(0, 10).map((iss: string, i: number) => <li key={i}>{iss}</li>)}
   696→                </ul>
   697→              </div>
   698→            )}
   699→            <Button onClick={confirmImport}>Confirm Import ({validation.valid} trades)</Button>
   700→          </div>
   701→        )}
   702→      </DialogContent>
   703→    </Dialog>
   704→  );
   705→}

   707→function LegalDoc({ doc, label }: { doc: string; label: string }) {
   708→  const [state, setState] = useState<any>(null);
   709→  const [open, setOpen] = useState(false);
   710→
   711→  useEffect(() => {
   712→    fetch(`/api/legal/${doc}`).then((r) => r.json()).then(setState);
   713→  }, [doc]);
   714→
   715→  async function accept() {
   716→    const res = await fetch(`/api/legal/${doc}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({}) });
   717→    if (res.ok) {
   718→      toast.success(`${label} accepted.`);
   719→      const data = await res.json();
   720→      setState({ ...state, accepted: true, acceptedAt: data.acceptedAt });
   721→    }
   722→  }
   723→
   724→  return (
   725→    <div className="flex items-center justify-between gap-3 py-2">
   726→      <div>
   727→        <div className="font-medium text-sm">{label}</div>
   728→        <div className="text-xs text-muted-foreground">
   729→          Version {state?.version ?? "1.0"} · {state?.accepted ? `Accepted ${new Date(state.acceptedAt).toLocaleDateString()}` : "Not yet accepted"}
   730→        </div>
   731→      </div>
   732→      <div className="flex gap-2">
   733→        <Button size="sm" variant="ghost" onClick={() => setOpen(true)}>View</Button>
   734→        {state && !state.accepted && <Button size="sm" onClick={accept}>Accept</Button>}
   735→      </div>
   736→      <Dialog open={open} onOpenChange={setOpen}>
   737→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
   738→          <DialogHeader><DialogTitle>{label}</DialogTitle></DialogHeader>
   739→          <div className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">
   740→            {LEGAL_DOCS[doc] ?? "Document text."}
   741→          </div>
   742→        </DialogContent>
   743→      </Dialog>
   744→    </div>
   745→  );
   746→}
