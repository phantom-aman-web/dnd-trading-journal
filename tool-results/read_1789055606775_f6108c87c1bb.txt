   136→export function SetupWizard() {
   137→  const { state, setupStep, nextSetupStep, prevSetupStep, setState } = useOnboarding();
   138→  const visible = state === "setup";
   139→  const { navigate } = useNav();
   140→  const [data, setData] = useState<SetupData>(DEFAULT_DATA);
   141→  const [loading, setLoading] = useState(false);
   142→
   143→  useEffect(() => {
   144→    if (!visible) return;
   145→    const handler = (e: KeyboardEvent) => {
   146→      if (e.key === "Escape") {
   147→        e.preventDefault();
   148→        setState("skipped");
   149→      }
   150→    };
   151→    window.addEventListener("keydown", handler);
   152→    return () => window.removeEventListener("keydown", handler);
   153→  }, [visible, setState]);
   154→
   155→  function update(patch: Partial<SetupData>) {
   156→    setData((d) => ({ ...d, ...patch }));
   157→  }
   158→
   159→  function handleSkip() {
   160→    setState("tour");
   161→  }
   162→
   163→  if (!visible) return null;
   164→
   165→  const progress = ((setupStep + 1) / TOTAL_SETUP_STEPS) * 100;
   166→
   167→  return (
   168→    <div
   169→      role="dialog"
   170→      aria-modal="true"
   171→      aria-labelledby="setup-title"
   172→      className="fixed inset-0 z-[100] flex items-center justify-center bg-black/60 backdrop-blur-sm p-4 animate-in fade-in duration-200"

   174→      <div className="w-full max-w-xl max-h-[92vh] flex flex-col rounded-xl border border-border bg-card text-card-foreground shadow-xl animate-in zoom-in-95 duration-200">
   175→        {/* Header — minimal, just close button */}
   176→        <button
   177→          aria-label="Close setup"
   178→          onClick={() => setState("skipped")}
   179→          className="absolute top-4 right-4 z-10 inline-flex h-8 w-8 items-center justify-center rounded-md text-muted-foreground hover:bg-muted hover:text-foreground"
   180→        >
   181→          <X className="h-4 w-4" />

   183→

   202→            <Step1Profile data={data} update={update} loading={loading} />
   203→          </div>
   204→          <div className={setupStep === 1 ? "block" : "hidden"}>
   205→            <Step2AccountMarkets data={data} update={update} loading={loading} />

   218→        {/* Footer — Back / Skip / Next */}

   221→            variant="ghost"

   325→function Step2AccountMarkets({
   326→  data,
   327→  update,
   328→  loading,
   329→}: {
   330→  data: SetupData;
   331→  update: (patch: Partial<SetupData>) => void;
   332→  loading: boolean;
   333→}) {
   334→  const [showAccount, setShowAccount] = useState(true);

   336→    if (!acc.find((m) => m.market === inst.market)) {
   337→      acc.push({ market: inst.market, label: inst.market.charAt(0).toUpperCase() + inst.market.slice(1) });
   338→    }
   339→    return acc;
   340→  }, [] as { market: string; label: string }[]);
   341→
   342→  function toggleInstrument(symbol: string) {
   343→    const current = data.selectedInstruments;
   344→    if (current.includes(symbol)) {
   345→      update({ selectedInstruments: current.filter((s) => s !== symbol) });
   346→    } else {
   347→      update({ selectedInstruments: [...current, symbol] });
   348→    }
   349→  }
   350→
   351→  return (
   352→    <div className="space-y-5">
   353→      <div>
   354→        <h2 className="text-xl font-semibold tracking-tight">What do you trade?</h2>
   355→        <p className="mt-1.5 text-sm text-muted-foreground leading-relaxed">
   356→          Choose the markets you trade most often. DnD will use this information throughout your trading workflow.
   357→        </p>
   358→      </div>
   359→
   360→      {/* Primary Trading Account */}
   361→      <div className="space-y-3 rounded-lg border border-border bg-muted/30 p-4">
   362→        <button
   363→          onClick={() => setShowAccount(!showAccount)}
   364→          className="flex items-center justify-between w-full text-left"
   365→        >
   366→          <div className="flex items-center gap-2">

   371→        </button>
   372→        {showAccount && (
   373→          <div className="space-y-3 pt-2">
   374→            <div className="grid grid-cols-2 gap-3">

   378→                  value={data.accountName}
   379→                  onChange={(e) => update({ accountName: e.target.value })}
   380→                  placeholder="Primary"
   381→                  disabled={loading}
   382→                  className="h-9 text-sm"
   383→                />
   384→              </div>

   388→                  value={data.broker}
   389→                  onChange={(e) => update({ broker: e.target.value })}
   390→                  placeholder="e.g. FTMO, IC Markets"
   391→                  disabled={loading}
   392→                  className="h-9 text-sm"
   393→                />
   394→              </div>
   395→            </div>
   396→            <div className="grid grid-cols-2 gap-3">
   397→              <div className="space-y-1.5">
   398→                <label className="text-[11px] font-medium text-muted-foreground">Account Type</label>

   401→                  <SelectContent>
   402→                    <SelectItem value="live">Live</SelectItem>
   403→                    <SelectItem value="demo">Demo</SelectItem>
   404→                    <SelectItem value="prop">Prop Firm</SelectItem>
   405→                    <SelectItem value="funded">Funded</SelectItem>
   406→                    <SelectItem value="backtest">Backtest</SelectItem>
   407→                  </SelectContent>
   408→                </Select>
   409→              </div>
   410→              <div className="space-y-1.5">
   411→                <label className="text-[11px] font-medium text-muted-foreground">Starting Balance</label>

   413→                  type="number"
   414→                  value={data.startingBalance}

   425→      {/* Markets You Trade */}

   427→        <div className="flex items-center gap-2">

   498→            onChange={(sym) => update({ defaultInstrument: sym })}
   499→            placeholder="Select your most-traded instrument…"
   500→          />
   501→        </div>
   502→      )}
   503→    </div>
   504→  );
   505→}

   510→
   511→function Step3Strategy({
   512→  data,
   513→  update,
   514→  loading,
   515→}: {
   516→  data: SetupData;
   517→  update: (patch: Partial<SetupData>) => void;
   518→  loading: boolean;
   519→}) {
   520→  function applyTemplate(template: string) {
   521→    update({ strategyTemplate: template });
   522→    let rules: { title: string; required: boolean; description: string }[] = [];
   523→    if (template === "ict") rules = ICT_RULES.map((r) => ({ ...r }));
   524→    else if (template === "trend") rules = TREND_RULES.map((r) => ({ ...r }));
   525→    else if (template === "breakout") rules = BREAKOUT_RULES.map((r) => ({ ...r }));
   526→    else rules = [{ title: "", required: true, description: "" }];
   527→    update({ rules });
   528→
   529→    if (template === "ict" && !data.strategyName) update({ strategyName: "ICT Setup" });
   530→    else if (template === "trend" && !data.strategyName) update({ strategyName: "Trend Following" });
   531→    else if (template === "breakout" && !data.strategyName) update({ strategyName: "Breakout" });
   532→  }
   533→
   534→  function addRule() {
   535→    update({ rules: [...data.rules, { title: "", required: true, description: "" }] });
   536→  }
   537→
   538→  function removeRule(idx: number) {
   539→    update({ rules: data.rules.filter((_, i) => i !== idx) });
   540→  }
   541→
   542→  function updateRule(idx: number, patch: Partial<{ title: string; required: boolean; description: string }>) {
   543→    update({ rules: data.rules.map((r, i) => (i === idx ? { ...r, ...patch } : r)) });
   544→  }
   545→
   546→  return (
   547→    <div className="space-y-5">
   548→      <div>
   549→        <h2 className="text-xl font-semibold tracking-tight">How do you trade?</h2>
   550→        <p className="mt-1.5 text-sm text-muted-foreground leading-relaxed">
   551→          Create the strategy you want DnD to use when evaluating your setups. These rules become the checklist DnD uses when you evaluate your trades.
   552→        </p>
   553→      </div>
   554→
   555→      {/* Template selector */}
   556→      <div className="space-y-1.5">
   557→        <label className="text-xs font-semibold text-foreground">Start from a template</label>
   558→        <Select value={data.strategyTemplate} onValueChange={applyTemplate}>
   559→          <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   560→          <SelectContent>
   561→            {STRATEGY_TEMPLATES.map((t) => (
   562→              <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>
   563→            ))}

   846→/* ------------------------------------------------------------------ */
   847→/* Save + Finish — persists all data then launches the tour            */
   848→/* ------------------------------------------------------------------ */
   849→
   850→function SaveAndFinish({
   851→  data,
   852→  loading,
   853→  setLoading,
   854→}: {
   855→  data: SetupData;
   856→  loading: boolean;
   857→  setLoading: (b: boolean) => void;
   858→}) {
   859→  const qc = useQueryClient();
   860→  const { setState } = useOnboarding();
   861→  const { navigate } = useNav();
   862→
   863→  async function save() {
   864→    setLoading(true);
   865→    try {
   866→      // 1. Save profile + risk defaults to settings
   867→      const settingsPatch: Record<string, unknown> = {
   868→        name: data.name || undefined,
   869→        timezone: data.timezone,
   870→        baseCurrencySymbol: data.currency === "USD" ? "$" : data.currency,
   871→        defaultRiskPct: data.defaultRiskPct,
   872→        defaultSession: data.defaultSession,
   873→        defaultTimeframe: data.defaultTimeframe,
   874→        maxTradesPerDay: data.maxTradesPerDay ? parseInt(data.maxTradesPerDay) : null,
   875→        dailyLossLimitPct: data.dailyLossLimitPct ? parseFloat(data.dailyLossLimitPct) : null,
   876→        maxRiskPerTradePct: data.maxRiskPerTradePct ? parseFloat(data.maxRiskPerTradePct) : null,
   877→      };
   878→      if (data.defaultInstrument) settingsPatch.defaultInstrument = data.defaultInstrument;
   879→      if (data.selectedInstruments.length > 0) {
   880→        settingsPatch.preferredInstruments = JSON.stringify(data.selectedInstruments);
   881→      }
   882→
   883→      await fetch("/api/settings", {
   884→        method: "PATCH",
   885→        headers: { "Content-Type": "application/json" },
   886→        body: JSON.stringify(settingsPatch),
   887→      });
   888→
   889→      // 2. Create trading account
   890→      const accountRes = await fetch("/api/accounts", {
   891→        method: "POST",
   892→        headers: { "Content-Type": "application/json" },
   893→        body: JSON.stringify({
   894→          name: data.accountName || "Primary",
   895→          broker: data.broker || undefined,
   896→          accountType: data.accountType,
   897→          currency: data.currency,
   898→          startingBalanceCents: Math.round(parseFloat(data.startingBalance || "10000") * 100),
   899→          isDefault: true,
   900→        }),
   901→      });
   902→      if (accountRes.ok) {
   903→        const account = await accountRes.json();
   904→        // Update defaultAccountId
   905→        await fetch("/api/settings", {
   906→          method: "PATCH",
   907→          headers: { "Content-Type": "application/json" },
   908→          body: JSON.stringify({ defaultAccountId: account.id }),
   909→        });
   910→      }
   911→
   912→      // 3. Create strategy + version
   913→      if (data.strategyName) {
   914→        const cleanRules = data.rules
   915→          .filter((r) => r.title.trim())
   916→          .map((r) => ({ title: r.title.trim(), required: r.required, description: r.description.trim() }));
   917→
   918→        const stratRes = await fetch("/api/strategies", {
   919→          method: "POST",
   920→          headers: { "Content-Type": "application/json" },
   921→          body: JSON.stringify({
   922→            name: data.strategyName,
   923→            description: data.strategyDescription || undefined,
   924→            market: "mixed",
   925→            rules: cleanRules,
   926→          }),
   927→        });
   928→        if (!stratRes.ok) {
   929→          toast.error("Failed to create strategy — you can set it up later.");
   930→        }
   931→      }
   932→
   933→      // 4. Invalidate everything
   934→      qc.invalidateQueries();
   935→
   936→      // 5. Launch tour
   937→      setState("tour");
   938→    } catch {
   939→      toast.error("Something went wrong. Your setup may be incomplete.");
   940→      setLoading(false);
   941→    }
   942→  }
   943→
   944→  return (
   945→    <Button size="sm" onClick={save} disabled={loading}>
   946→      {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Check className="h-4 w-4" />}
   947→      Go to Dashboard
   948→    </Button>
   949→  );
   950→}
