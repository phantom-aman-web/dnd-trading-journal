    46→async function fetchTrade(id: string) {
    47→  const res = await fetch(`/api/trades/${id}`, { cache: "no-store" });
    48→  if (!res.ok) throw new Error("Failed to load trade");
    49→  return res.json();
    50→}

    57→    queryKey: ["trade", id],

    64→  const [executeTradeOpen, setExecuteTradeOpen] = useState(false);
    65→

    74→    if (!trade) return null;

    76→      ? trade.checklistEvaluations
    77→      : [];

    86→      checklistName,
    87→      versionLabel: version.versionLabel,
    88→      weightedScore: latest.weightedScore,
    89→      finalGrade: latest.finalGrade,
    90→      evaluatedAt: latest.evaluatedAt,
    91→      items,
    92→      answers,
    93→    };

    96→  if (isLoading) {

    98→      <div className="p-4 md:p-6 space-y-3">
    99→        <Skeleton className="h-8 w-32" />

   101→        <Skeleton className="h-64" />
   102→      </div>

   104→  }
   105→
   106→  if (!trade) return null;

   108→  const pnl = formatSignedCents(trade.netPnlCents);

   112→  const psychAfter = safeJson(trade.psychAfterJson, {});

   138→        </Button>

   140→          <div className="flex items-center gap-2 flex-wrap">
   141→            <h1 className="text-2xl font-semibold tracking-tight">{trade.instrumentSymbol}</h1>
   142→            <Badge variant="outline" className={cn(trade.direction === "long" ? "text-profit" : "text-loss")}>
   143→              {trade.direction === "long" ? "LONG" : "SHORT"}
   144→            </Badge>

   147→            )}

   150→                v{trade.strategyVersion.versionLabel}
   151→              </Badge>

   154→              variant="secondary"
   155→              className={cn(
   156→                "capitalize",
   157→                trade.status === "planned" && "bg-primary/10 text-primary",
   158→                trade.status === "open" && "bg-warning/15 text-warning",

   161→              {trade.status}
   162→            </Badge>
   163→          </div>
   164→          <div className="mt-1 text-sm text-muted-foreground">
   165→            {trade.entryTime ? new Date(trade.entryTime).toLocaleString() : "Open"} · {trade.session ?? "-"}

   167→        </div>
   168→        <div className="text-right">
   169→          <div className={cn("text-xl font-semibold tnum", pnl.sign > 0 ? "text-profit" : pnl.sign < 0 ? "text-loss" : "")}>
   170→            {pnl.text}
   171→          </div>
   172→          <div className={cn("text-sm tnum", trade.actualR && dToNumber(trade.actualR) > 0 ? "text-profit" : trade.actualR && dToNumber(trade.actualR) < 0 ? "text-loss" : "text-muted-foreground")}>
   173→            {formatR(trade.actualR)}
   174→          </div>
   175→        </div>
   176→      </div>

   242→          <Trash2 className="h-4 w-4" /> Delete

   251→        <MiniStat label="Net P&L" value={pnl.text} tone={pnl.sign > 0 ? "profit" : pnl.sign < 0 ? "loss" : "neutral"} />
   252→        <MiniStat label="Gross P&L" value={formatSignedCents(trade.grossPnlCents).text} tone={trade.grossPnlCents > 0 ? "profit" : trade.grossPnlCents < 0 ? "loss" : "neutral"} />

   255→          value={trade.plannedRR ? formatR(trade.plannedRR) : "—"}
   256→          tone="neutral"
   257→          hint="Target distance ÷ stop distance (planned before entry)"
   258→        />
   259→        <MiniStat
   260→          label="Realized R"
   261→          value={
   262→            trade.status === "open" || trade.status === "planned"
   263→              ? "—"
   264→              : formatR(trade.actualR)

   266→          tone={trade.actualR && dToNumber(trade.actualR) > 0 ? "profit" : trade.actualR && dToNumber(trade.actualR) < 0 ? "loss" : "neutral"}
   267→          hint="Realized result relative to your planned risk"
   268→        />
   269→        <MiniStat label="Costs" value={centsToNumberString(trade.feesCents + trade.commissionCents + trade.swapCents)} />
   270→        <MiniStat label="Strategy" value={strategyLabel} />
   271→      </div>
   272→
   273→      {/* Tabs */}
   274→      <Tabs defaultValue="overview" className="w-full">
   275→        <TabsList className="w-full justify-start overflow-x-auto">
   276→          <TabsTrigger value="overview">Overview</TabsTrigger>
   277→          <TabsTrigger value="setup">Setup</TabsTrigger>
   278→          <TabsTrigger value="execution">Execution</TabsTrigger>
   279→          <TabsTrigger value="psychology">Psychology</TabsTrigger>
   280→          <TabsTrigger value="evidence">Evidence</TabsTrigger>
   281→          <TabsTrigger value="review">Review</TabsTrigger>
   282→        </TabsList>
   283→
   284→        <TabsContent value="overview" className="space-y-4 mt-3">
   285→          <Card className="p-4">
   286→            <h3 className="font-medium mb-2">Trading Thesis</h3>
   287→            {thesis?.why ? (
   288→              <p className="text-sm text-muted-foreground leading-relaxed">{thesis.why}</p>
   289→            ) : (
   290→              <p className="text-sm text-muted-foreground">No thesis recorded.</p>
   291→            )}
   292→            {thesis?.narrative && (
   293→              <div className="mt-3 text-sm">
   294→                <span className="text-muted-foreground">Narrative:</span> {thesis.narrative}
   295→              </div>
   296→            )}
   297→            <div className="mt-3 grid sm:grid-cols-2 gap-2 text-sm">
   298→              {thesis?.liquidityTarget && <div><span className="text-muted-foreground">Liquidity target:</span> {thesis.liquidityTarget}</div>}
   299→              {thesis?.target && <div><span className="text-muted-foreground">Target:</span> {thesis.target}</div>}
   300→            </div>
   301→            {thesis?.confirms && Array.isArray(thesis.confirms) && thesis.confirms.length > 0 && (
   302→              <div className="mt-3 text-sm">
   303→                <span className="text-muted-foreground">Confirmations:</span>
   304→                <div className="flex flex-wrap gap-1 mt-1">
   305→                  {thesis.confirms.map((c: string) => <Badge key={c} variant="secondary" className="text-xs">{c}</Badge>)}
   306→                </div>
   307→              </div>
   308→            )}
   309→            {thesis?.invalidates && Array.isArray(thesis.invalidates) && thesis.invalidates.length > 0 && (
   310→              <div className="mt-3 text-sm">
   311→                <span className="text-muted-foreground">Invalidations:</span>
   312→                <div className="flex flex-wrap gap-1 mt-1">

   314→                </div>
   315→              </div>
   316→            )}
   317→          </Card>
   318→
   319→          {tags.length > 0 && (
   320→            <Card className="p-4">
   321→              <h3 className="font-medium mb-2 flex items-center gap-2"><Tag className="h-4 w-4" /> Tags</h3>
   322→              <div className="flex flex-wrap gap-1">
   323→                {tags.map((t) => <Badge key={t} variant="secondary">{t}</Badge>)}
   324→              </div>
   325→            </Card>
   326→          )}
   327→
   328→          {behaviorFlags.length > 0 && (
   329→            <Card className="p-4 border-warning/50">
   330→              <h3 className="font-medium mb-2 flex items-center gap-2"><AlertTriangle className="h-4 w-4 text-warning" /> Behavior Flags</h3>
   331→              <div className="flex flex-wrap gap-1">
   332→                {behaviorFlags.map((f) => (
   333→                  <Badge key={f} variant="outline" className="text-warning">
   334→                    {f.split("_").map((w) => w.charAt(0).toUpperCase() + w.slice(1)).join(" ")}
   335→                  </Badge>
   336→                ))}
   337→              </div>
   338→            </Card>
   339→          )}
   340→
   341→          {trade.notes && (
   342→            <Card className="p-4">
   343→              <h3 className="font-medium mb-2">Notes</h3>
   344→              <p className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">{trade.notes}</p>
   345→            </Card>
   346→          )}
   347→        </TabsContent>
   348→
   349→        <TabsContent value="setup" className="space-y-4 mt-3">
   350→          <Card className="p-4">
   351→            <h3 className="font-medium mb-3">ICT Setup</h3>
   352→            <div className="grid sm:grid-cols-2 gap-4 text-sm">
   353→              <SetupSection title="Higher Timeframe Context" data={setup?.htfContext} />
   354→              <SetupSection title="Liquidity" data={setup?.liquidity} isList />
   355→              <SetupSection title="Structure" data={setup?.structure} isList />
   356→              <SetupSection title="Entry Model" data={setup?.entryModel} />
   357→              <SetupSection title="Session" data={setup?.session} />
   358→              <SetupSection title="Setup Score" data={trade.setupScore ? `${(dToNumber(trade.setupScore) * 100).toFixed(0)}%` : null} hint="Calculated from the weighted rules defined by your strategy" />
   359→            </div>
   360→          </Card>

   362→          {/* Checklist Evaluation (Fix 3). Shows the snapshot of the A+

   365→              POST/PATCH /api/trades/[id] (see checklist-evaluation.ts). */}

   383→                    {checklistEvaluation.finalGrade}

   761→function SetupSection({ title, data, isList, hint }: { title: string; data: any; isList?: boolean; hint?: string }) {
   762→  const [showHint, setShowHint] = useState(false);
   763→  if (!data) return null;
   764→  return (
   765→    <div>
   766→      <div className="text-xs text-muted-foreground uppercase tracking-wide flex items-center gap-1.5">
   767→        {title}
   768→        {hint && (
   769→          <span
   770→            className="relative inline-flex normal-case"
   771→            onMouseEnter={() => setShowHint(true)}
   772→            onMouseLeave={() => setShowHint(false)}
   773→            onClick={(e) => { e.preventDefault(); setShowHint((s) => !s); }}
   774→          >
   775→            <HelpCircle className="h-3 w-3 text-muted-foreground cursor-help" />
   776→            {showHint && (
   777→              <span
   778→                className="absolute left-0 top-4 z-50 w-48 rounded-md border border-border bg-popover p-2 text-xs font-normal normal-case text-popover-foreground shadow-md"
   779→                role="tooltip"
   780→              >
   781→                {hint}
   782→              </span>
   783→            )}
   784→          </span>
   785→        )}
   786→      </div>
   787→      {isList && Array.isArray(data) ? (
   788→        <div className="flex flex-wrap gap-1 mt-1">
   789→          {data.map((d: string, i: number) => <Badge key={i} variant="secondary" className="text-xs">{d}</Badge>)}
   790→        </div>
   791→      ) : typeof data === "object" ? (
   792→        <div className="mt-1 text-sm">
   793→          {Object.entries(data).map(([k, v]) => (
   794→            <div key={k}>
   795→              <span className="text-muted-foreground">{k}:</span> {String(v)}
   796→            </div>
   797→          ))}
   798→        </div>
   799→      ) : (
   800→        <div className="font-medium mt-1">{String(data)}</div>
   801→      )}
   802→    </div>
   803→  );
   804→}
