   296→   296→  function setField(path: string, value: any) {
   297→   297→    setForm((f: any) => {
   298→   298→      const next = { ...f };
   299→   299→      const parts = path.split(".");
   300→   300→      let cur: any = next;
   301→   301→      for (let i = 0; i < parts.length - 1; i++) {
   302→   302→        cur[parts[i]] = { ...cur[parts[i]] };
   303→   303→        cur = cur[parts[i]];
   304→   304→      }
   305→   305→      cur[parts[parts.length - 1]] = value;
   306→   306→      return next;
   307→   307→    });
   308→   308→  }

   310→   310→  function toggleArrayField(path: string, value: string) {
   311→   311→    setForm((f: any) => {
   312→   312→      const next = { ...f };
   313→   313→      const parts = path.split(".");
   314→   314→      let cur: any = next;
   315→   315→      for (let i = 0; i < parts.length - 1; i++) {
   316→   316→        cur[parts[i]] = { ...cur[parts[i]] };
   317→   317→        cur = cur[parts[i]];
   318→   318→      }
   319→   319→      const key = parts[parts.length - 1];
   320→   320→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   321→   321→      const idx = arr.indexOf(value);
   322→   322→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   323→   323→      cur[key] = arr;
   324→   324→      return next;
   325→   325→    });
   326→   326→  }

   328→   328→  // Update one A+ checklist answer field (`checked` or `note`) for the given

   330→   330→  function updateChecklistAnswer(
   331→   331→    itemId: string,
   332→   332→    field: "checked" | "note",
   333→   333→    value: boolean | string,
   334→   334→  ) {
   335→   335→    setForm((f: any) => {
   336→   336→      const current = f.checklistAnswers?.[itemId] ?? { checked: false };
   337→   337→      const next = {
   338→   338→        ...f.checklistAnswers,
   339→   339→        [itemId]: {
   340→   340→          ...current,
   341→   341→          [field]: value,
   342→   342→        },
   343→   343→      };
   344→   344→      // If a checked item is being unchecked, drop its note so storage stays clean.
   345→   345→      if (field === "checked" && value === false && next[itemId].note) {
   346→   346→        delete next[itemId].note;
   347→   347→      }
   348→   348→      return { ...f, checklistAnswers: next };
   349→   349→    });

   392→   392→        setupGrade: form.checklistVersionId ? evaluation.grade : null,

   399→   399→        executions,
   400→   400→        feesCents: Number(form.feesCents) || 0,
   401→   401→        commissionCents: Number(form.commissionCents) || 0,
   402→   402→        swapCents: Number(form.swapCents) || 0,
   403→   403→        slippageCents: Number(form.slippageCents) || 0,
   404→   404→        entryTime: form.entryTime ? new Date(form.entryTime).toISOString() : null,
   405→   405→        exitTime: form.exitTime ? new Date(form.exitTime).toISOString() : null,
   406→   406→        tradingTimezone: form.tradingTimezone,
   407→   407→        setup: form.setup,
   408→   408→        thesis: form.thesis,
   409→   409→        notes: form.notes || null,
   410→   410→        lessons: form.lessons || null,
   411→   411→        psychBefore: form.psychBefore,
   412→   412→        psychAfter: form.psychAfter,
   413→   413→        behaviorFlags: form.behaviorFlags,
   414→   414→        ruleCompliance: form.ruleCompliance,
   415→   415→        isDraft,
   416→   416→      };
   417→   417→      const url = editingId ? `/api/trades/${editingId}` : "/api/trades";
   418→   418→      const method = editingId ? "PATCH" : "POST";
   419→   419→      const res = await fetch(url, {
   420→   420→        method,
   421→   421→        headers: { "Content-Type": "application/json" },
   422→   422→        body: JSON.stringify(payload),
   423→   423→      });

   426→   426→        toast.error(data?.error ?? "Failed to save trade.");
   427→   427→        setSaving(false);
   428→   428→        return;
   429→   429→      }
   430→   430→      // Attach any orphaned uploaded media to the newly created trade and save captions
   431→   431→      if (form.uploadedMedia?.length > 0) {
   432→   432→        for (const m of form.uploadedMedia) {
   433→   433→          if (m.id) {
