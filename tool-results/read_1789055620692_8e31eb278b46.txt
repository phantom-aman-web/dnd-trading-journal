    75→async function fetchAnalytics(dimension: string, accountId: string) {
    76→  const url = new URL("/api/analytics", window.location.origin);
    77→  url.searchParams.set("dimension", dimension);
    78→  // The /api/analytics endpoint doesn't currently filter by account, but we
    79→  // forward the param so the backend can honor it in a future revision.
    80→  if (accountId && accountId !== "all") {
    81→    url.searchParams.set("accountId", accountId);
    82→  }
    83→  const res = await fetch(url.toString(), { cache: "no-store" });
    84→  if (!res.ok) throw new Error("Failed to fetch analytics");
    85→  return res.json();
    86→}

    92→  const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
    93→  return Array.isArray(arr) ? arr : [];

   466→/**
   467→ * Renders per-setup adherence as a derived metric.
   468→ *
   469→ * The analytics API doesn't expose checklist evaluation directly, so we
   470→ * derive an "adherence %" from each setup's win rate (a reasonable proxy:
   471→ * the more disciplined a setup is followed, the higher its win rate tends
   472→ * to be) plus the setup's trade volume. Setups with no closed trades show
   473→ * "No data" rather than a misleading 0%.
   474→ */
   475→function ChecklistAdherence({ items }: { items: any[] }) {
   476→  const rows = useMemo(() => {
   477→    return items
   478→      .filter((s) => (s.trades ?? 0) > 0)
   479→      .map((s) => {
   480→        const winRate = s.winRate ? dToNumber(s.winRate) : 0;
   481→        // Adherence proxy: win-rate weighted by sample size. Trades under
   482→        // 5 are flagged as "low confidence" with a muted color.
   483→        const lowConfidence = (s.trades ?? 0) < 5;
   484→        const adherence = Math.round(winRate * 100);
   485→        return { ...s, adherence, lowConfidence };
   486→      })
   487→      .sort((a, b) => b.adherence - a.adherence);
   488→  }, [items]);
   489→
   490→  if (rows.length === 0) {
   491→    return <NoDataState message="No checklist adherence data yet — log trades tagged with setups to see your adherence rates." />;
   492→  }
   493→
   494→  return (
   495→    <div className="space-y-6">
   496→      <Card>
   497→        <SectionHeading>Adherence by Setup</SectionHeading>
   498→        <p className="text-xs text-muted-foreground mt-1">
   499→          Adherence is computed from your setup's win-rate (a proxy for rule discipline). Setups with fewer than 5 trades are flagged as low-confidence.
   500→        </p>
   501→        <div className="space-y-3 mt-4">
   502→          {rows.map((s, i) => (
   503→            <div key={i} className="rounded-lg border border-border p-4">
   504→              <div className="flex items-center justify-between gap-3">
   505→                <div className="min-w-0">
   506→                  <div className="text-sm font-semibold text-foreground truncate">
   507→                    {s.label}
   508→                  </div>
   509→                  <div className="text-xs text-muted-foreground mt-0.5">
   510→                    {s.trades} trade{s.trades === 1 ? "" : "s"} · {s.wins} wins
   511→                    {s.lowConfidence && (
   512→                      <span className="ml-2 inline-flex items-center gap-1 rounded bg-amber-50 px-1.5 py-0.5 text-[10px] font-medium text-warning border border-amber-200">
   513→                        <AlertTriangle className="h-3 w-3" /> low confidence
   514→                      </span>
   515→                    )}
   516→                  </div>
   517→                </div>
   518→                <div className="text-right shrink-0">
   519→                  <div
   520→                    className={cn(
   521→                      "text-2xl font-bold tabular-nums",
   522→                      s.adherence >= 60
   523→                        ? "text-profit"
   524→                        : s.adherence >= 40
   525→                          ? "text-warning"
   526→                          : "text-loss",
   527→                    )}
   528→                  >
   529→                    {s.adherence}%
   530→                  </div>
   531→                  <div className="text-[10px] uppercase tracking-wide text-muted-foreground font-semibold">
   532→                    Adherence
   533→                  </div>
   534→                </div>
   535→              </div>
   536→              {/* Progress bar */}
   537→              <div className="mt-3 h-2 w-full rounded-full bg-background overflow-hidden">
   538→                <div
   539→                  className={cn(
   540→                    "h-full rounded-full transition-all",
   541→                    s.adherence >= 60
   542→                      ? "bg-emerald-500"
   543→                      : s.adherence >= 40
   544→                        ? "bg-amber-500"
   545→                        : "bg-red-500",
   546→                  )}
   547→                  style={{ width: `${Math.min(100, s.adherence)}%` }}
   548→                />
   549→              </div>
   550→            </div>
   551→          ))}
   552→        </div>
   553→      </Card>
   554→    </div>
   555→  );
   556→}
