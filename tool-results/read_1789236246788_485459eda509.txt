    52→    52→/**
    53→    53→ * Compute the pointValueCents from instrument metadata.
    54→    54→ *
    55→    55→ * pointValueCents = monetary value (in cents) of a 1.0 price move per 1.0 quantity.
    56→    56→ *
    57→    57→ * For futures: this is the dollar-per-point value (e.g. NQ = $20/point = 2000 cents).
    58→    58→ * For forex: this is contractSize × 100 (e.g. 100,000 × $1 = $100,000 = 10,000,000 cents).
    59→    59→ * For stocks/crypto: 100 ($1 per $1 move).
    60→    60→ *
    61→    61→ * The old implementation computed TICK value (tickSize × contractSize × 100)
    62→    62→ * and mislabeled it as pointValueCents. This is now corrected.
    63→    63→ */
    64→    64→export function computePointValueCents(
    65→    65→  contractSize: string | number | null | undefined,
    66→    66→  tickSize: string | number | null | undefined,
    67→    67→  pipSize: string | number | null | undefined,
    68→    68→): Cents {
    69→    69→  const cs = Number(contractSize ?? "1");
    70→    70→  if (!Number.isFinite(cs) || cs <= 0) return 100;
    71→    71→
    72→    72→  // For forex (cs = 100000): pointValueCents = cs × 100 = 10,000,000
    73→    73→  // For gold (cs = 100): pointValueCents = cs × 100 = 10,000
    74→    74→  // For futures NQ (cs = 20, $20/point): pointValueCents = cs × 100 = 2000
    75→    75→  // For futures ES (cs = 50, $50/point): pointValueCents = cs × 100 = 5000
    76→    76→  // For crypto/stocks (cs = 1): pointValueCents = 100
    77→    77→  const cents = Math.round(cs * 100);
    78→    78→  if (!Number.isFinite(cents) || cents <= 0) return 100;
    79→    79→  return cents;
    80→    80→}

   102→   102→/**
   103→   103→ * Extract mood-tag strings from a `psychBeforeJson` (or `psychAfterJson`)
   104→   104→ * column. The column is stored as a JSON **object** whose `moodTags` field is
   105→   105→ * an array of strings (per spec section 38), but legacy rows may store a
   106→   106→ * bare array — both shapes are supported. Any other shape yields `[]`.
   107→   107→ *
   108→   108→ * Never throws: malformed JSON, null, or unexpected shapes degrade to `[]`.
   109→   109→ */
   110→   110→export function safeParsePsychTags(json: string | null | undefined): string[] {
   111→   111→  if (!json) return [];
   112→   112→  let v: unknown;
   113→   113→  try {
   114→   114→    v = JSON.parse(json);
   115→   115→  } catch {
   116→   116→    return [];
   117→   117→  }
   118→   118→  if (Array.isArray(v)) {
   119→   119→    // Backward-compat: legacy rows stored a bare array of tags.
   120→   120→    return v.filter((x): x is string => typeof x === "string");
   121→   121→  }
   122→   122→  if (v && typeof v === "object") {
   123→   123→    const tags = (v as { moodTags?: unknown }).moodTags;
   124→   124→    if (Array.isArray(tags)) {
   125→   125→      return tags.filter((x): x is string => typeof x === "string");
   126→   126→    }
   127→   127→  }
   128→   128→  return [];
   129→   129→}

   131→   131→// ---------------------------------------------------------------------------
   132→   132→// Core: compute trade P&L from execution ledger (spec section 33)
   133→   133→// ---------------------------------------------------------------------------
   134→   134→
   135→   135→/**
   136→   136→ * Calculate the gross P&L in cents from a list of fills.
   137→   137→ * Method: volume-weighted average entry and exit prices, signed by direction.
   138→   138→ * gross = (exitAvg - entryAvg) * qty * contractSize * pointValue  (long)
   139→   139→ * gross = (entryAvg - exitAvg) * qty * contractSize * pointValue  (short)
   140→   140→ * Only matches qty between entries and exits.
   141→   141→ */
   142→   142→export function calculateTradePnl(input: TradeCalcInput): TradeCalcResult {
   143→   143→  // pointValueCents is the authoritative monetary value of a 1.0 price move
   144→   144→  // per 1.0 quantity, in cents. It already encodes contractSize economics.
   145→   145→  // Formula: grossPnlCents = priceDiff × matchedQty × pointValueCents
   146→   146→  // DO NOT multiply by contractSize again — it's already in pointValueCents.
   147→   147→  const pointValueCents = input.pointValueCents ?? 100;
   148→   148→
   149→   149→  const entries = input.fills.filter((f) => f.kind === "entry");
   150→   150→  const exits = input.fills.filter((f) => f.kind === "exit");
   151→   151→
   152→   152→  // Volume-weighted average entry price
   153→   153→  const entryAvg = vwap(entries);
   154→   154→  const exitAvg = vwap(exits);
   155→   155→
   156→   156→  // Matched quantity (min of total entry qty and total exit qty)
   157→   157→  const entryQty = sumQty(entries);
   158→   158→  const exitQty = sumQty(exits);
   159→   159→  const matchedQty = dCmp(entryQty, exitQty) <= 0 ? entryQty : exitQty;
   160→   160→
   161→   161→  let grossPnlCents: Cents = 0;
   162→   162→  if (!dIsZero(matchedQty) && entryAvg && exitAvg) {
   163→   163→    // CANONICAL FORMULA (spec §9):
   164→   164→    // grossPnlCents = directionalMove × matchedQuantity × pointValueCents
   165→   165→    // pointValueCents is ALREADY in cents (no /100, no toCents conversion).
   166→   166→    const diff = input.direction === "long"
   167→   167→      ? dSub(exitAvg, entryAvg)
   168→   168→      : dSub(entryAvg, exitAvg);
   169→   169→    const grossDecimal = dMul(dMul(diff, matchedQty), dNormalize(String(pointValueCents)));
   170→   170→    // pointValueCents is already in cents, so just round — do NOT call toCents
   171→   171→    grossPnlCents = Math.round(dToNumber(grossDecimal));
   172→   172→  }
   173→   173→
   174→   174→  // Costs
   175→   175→  const fees = input.feesCents ?? 0;
   176→   176→  const commission = input.commissionCents ?? 0;
   177→   177→  const swap = input.swapCents ?? 0;
   178→   178→  const slippage = input.slippageCents ?? 0;
   179→   179→  const totalCostsCents = addCents(addCents(addCents(fees, commission), swap), slippage);
   180→   180→
   181→   181→  // Net = gross - costs (costs always reduce PnL)
   182→   182→  const netPnlCents = subCents(grossPnlCents, totalCostsCents);
   183→   183→
   184→   184→  // Stop & target distances
   185→   185→  const stopDistance = (input.plannedStopPrice && input.plannedEntryPrice && !dIsZero(dSub(input.plannedEntryPrice, input.plannedStopPrice)))
   186→   186→    ? dAbs(dSub(input.plannedEntryPrice, input.plannedStopPrice))
   187→   187→    : null;
   188→   188→  const targetDistance = (input.plannedTargetPrice && input.plannedEntryPrice && !dIsZero(dSub(input.plannedTargetPrice, input.plannedEntryPrice)))
   189→   189→    ? dAbs(dSub(input.plannedTargetPrice, input.plannedEntryPrice))
   190→   190→    : null;
   191→   191→
   192→   192→  // Planned R:R
   193→   193→  const plannedRR = (stopDistance && targetDistance)
   194→   194→    ? dDiv(targetDistance, stopDistance)
   195→   195→    : null;
   196→   196→
   197→   197→  // Risk amount: stopDistance × plannedQty × pointValueCents
   198→   198→  // (no contractSize — already encoded in pointValueCents)
   199→   199→  const plannedQty = entryQty;
   200→   200→  let riskAmountCents: Cents | null = null;
   201→   201→  let rewardAmountCents: Cents | null = null;
   202→   202→  if (stopDistance && !dIsZero(plannedQty)) {
   203→   203→    const riskDecimal = dMul(dMul(stopDistance, plannedQty), dNormalize(String(pointValueCents)));
   204→   204→    riskAmountCents = Math.round(dToNumber(riskDecimal));
   205→   205→  }
   206→   206→  if (targetDistance && !dIsZero(plannedQty)) {
   207→   207→    const rewardDecimal = dMul(dMul(targetDistance, plannedQty), dNormalize(String(pointValueCents)));
   208→   208→    rewardAmountCents = Math.round(dToNumber(rewardDecimal));
   209→   209→  }
   210→   210→
   211→   211→  // Actual R: netPnlCents / riskAmountCents (both in cents, ratio is unitless)
   212→   212→  let actualR: Decimal | null = null;
   213→   213→  if (riskAmountCents && riskAmountCents > 0) {
   214→   214→    actualR = dDiv(String(netPnlCents), String(riskAmountCents));
   215→   215→  }
   216→   216→
   217→   217→  // Status
   218→   218→  const status = deriveStatus(netPnlCents, entryQty, exitQty);
   219→   219→
   220→   220→  return {
   221→   221→    entryPriceAvg: entryAvg,
   222→   222→    exitPriceAvg: exitAvg,
   223→   223→    totalQuantity: entryQty, // sum of ALL entry fill quantities (not first fill)
   224→   224→    grossPnlCents,
   225→   225→    totalCostsCents,
   226→   226→    netPnlCents,
   227→   227→    stopDistance,
   228→   228→    targetDistance,
   229→   229→    plannedRR,
   230→   230→    actualR,
   231→   231→    riskAmountCents,
   232→   232→    rewardAmountCents,
   233→   233→    status,
   234→   234→  };
   235→   235→}
   236→   236→
   237→   237→function vwap(fills: ExecutionFill[]): Decimal | null {
   238→   238→  if (fills.length === 0) return null;
   239→   239→  let pq = 0; // sum of price*qty
   240→   240→  let q = 0;  // sum of qty
   241→   241→  for (const f of fills) {
   242→   242→    const p = dToNumber(f.price);
   243→   243→    const qt = dToNumber(f.quantity);
   244→   244→    if (!Number.isFinite(p) || !Number.isFinite(qt)) continue;
   245→   245→    pq += p * qt;
   246→   246→    q += qt;
   247→   247→  }
   248→   248→  if (q === 0) return null;
   249→   249→  return dNormalize(String(+(pq / q).toFixed(10)));
   250→   250→}
   251→   251→
   252→   252→function sumQty(fills: ExecutionFill[]): Decimal {
   253→   253→  let s = 0;
   254→   254→  for (const f of fills) s += dToNumber(f.quantity);
   255→   255→  return dNormalize(String(+s.toFixed(10)));
   256→   256→}
   257→   257→
   258→   258→function deriveStatus(netPnl: Cents, entryQty: Decimal, exitQty: Decimal): TradeStatus {
   259→   259→  if (dIsZero(entryQty)) return "open";
   260→   260→  if (dIsZero(exitQty)) return "open";
   261→   261→  const cmp = dCmp(exitQty, entryQty);
   262→   262→  if (cmp < 0) {
   263→   263→    return netPnl > 0 ? "partial_win" : netPnl < 0 ? "partial_loss" : "breakeven";
   264→   264→  }
   265→   265→  if (netPnl > 0) return "win";
   266→   266→  if (netPnl < 0) return "loss";
   267→   267→  return "breakeven";
   268→   268→}

   291→   291→export function calculatePositionSize(input: PositionSizeInput): PositionSizeResult {
   292→   292→  const riskAmountCents = Math.round((input.accountBalanceCents * dToNumber(input.riskPct)) / 100) * 100 / 100;
   293→   293→  // careful: Math.round(balance * pct) gives cents directly
   294→   294→  const riskCents = Math.round((input.accountBalanceCents / 100) * dToNumber(input.riskPct) * 100);
   295→   295→  const stopDistance = dAbs(dSub(input.entryPrice, input.stopPrice));
   296→   296→  // position size = riskAmount / (stopDistance * contractSize * pointValue)
   297→   297→  let positionSize: Decimal | null = null;
   298→   298→  if (!dIsZero(stopDistance)) {
   299→   299→    const denom = dToNumber(dMul(dMul(stopDistance, input.contractSize), dNormalize(String(input.pointValueCents / 100))));
   300→   300→    if (denom > 0) {
   301→   301→      positionSize = dDiv(String(riskCents / 100), String(denom));
   302→   302→    }
   303→   303→  }
   304→   304→  return {
   305→   305→    riskAmountCents: riskCents,
   306→   306→    stopDistance,
   307→   307→    positionSize,
   308→   308→    plannedLossCents: negCents(riskCents),
   309→   309→    plannedRewardCents: null,
   310→   310→  };
   311→   311→}

   356→   356→export function computeAggregateMetrics(trades: TradeMetricRow[]): AggregateMetrics {
   357→   357→  // CANONICAL CLOSED SET (spec §17): only WIN + LOSS + BREAKEVEN.
   358→   358→  // Partial trades (partial_win, partial_loss) are EXCLUDED from closed metrics.
   359→   359→  const closed = trades.filter((t) => t.status === "win" || t.status === "loss" || t.status === "breakeven");
   360→   360→  const wins = closed.filter((t) => t.netPnlCents > 0);
   361→   361→  const losses = closed.filter((t) => t.netPnlCents < 0);
   362→   362→  const bes = closed.filter((t) => t.netPnlCents === 0);
   363→   363→
   364→   364→  const totalPnl = closed.reduce((s, t) => s + t.netPnlCents, 0);
   365→   365→
   366→   366→  const winRate = closed.length > 0 ? dNormalize(String(wins.length / closed.length)) : null;
   367→   367→
   368→   368→  const avgPnl = closed.length > 0 ? Math.round(totalPnl / closed.length) : null;
   369→   369→
   370→   370→  const avgWin = wins.length > 0 ? Math.round(wins.reduce((s, t) => s + t.netPnlCents, 0) / wins.length) : null;
   371→   371→  const avgLoss = losses.length > 0 ? Math.round(losses.reduce((s, t) => s + t.netPnlCents, 0) / losses.length) : null;
   372→   372→
   373→   373→  // Avg R over trades with valid R
   374→   374→  const rValues = closed.map((t) => t.actualR).filter((r): r is Decimal => r != null && Number.isFinite(dToNumber(r)));
   375→   375→  const avgR = rValues.length > 0 ? dNormalize(String(+(rValues.reduce((s, r) => s + dToNumber(r), 0) / rValues.length).toFixed(10))) : null;
   376→   376→
   377→   377→  // Expectancy = avg realized R (per spec section 73)
   378→   378→  const expectancyR = avgR;
   379→   379→
   380→   380→  // Profit factor (spec section 72): gross profits / abs(gross losses)
   381→   381→  const grossProfit = wins.reduce((s, t) => s + t.netPnlCents, 0);
   382→   382→  const grossLoss = Math.abs(losses.reduce((s, t) => s + t.netPnlCents, 0));
   383→   383→  const profitFactor = grossLoss > 0 ? dNormalize(String(+(grossProfit / grossLoss).toFixed(10))) : null;
   384→   384→
   385→   385→  // Drawdown (spec §23): based on realized equity, ordered by EXIT timestamp
   386→   386→  const sorted = [...closed].sort((a, b) => {
   387→   387→    const ad = a.exitTime ? new Date(a.exitTime).getTime() : (a.entryTime ? new Date(a.entryTime).getTime() : 0);
   388→   388→    const bd = b.exitTime ? new Date(b.exitTime).getTime() : (b.entryTime ? new Date(b.entryTime).getTime() : 0);
   389→   389→    return ad - bd;
   390→   390→  });
   391→   391→  let equity = 0;
   392→   392→  let peak = 0;
   393→   393→  let maxDD = 0;
   394→   394→  let currentDD = 0;
   395→   395→  let recoveryStart: number | null = null;
   396→   396→  let recoveryPeriod: number | null = null;
   397→   397→  let inDD = false;
   398→   398→  let ddStartIdx = -1;
   399→   399→  for (let i = 0; i < sorted.length; i++) {
   400→   400→    equity += sorted[i].netPnlCents;
   401→   401→    if (equity > peak) {
   402→   402→      peak = equity;
   403→   403→      if (inDD) {
   404→   404→        // recovered
   405→   405→        inDD = false;
   406→   406→        recoveryPeriod = recoveryStart != null ? i - recoveryStart : null;
   407→   407→        recoveryStart = null;
   408→   408→      }
   409→   409→    }
   410→   410→    const dd = peak - equity;
   411→   411→    if (dd > maxDD) maxDD = dd;
   412→   412→    currentDD = dd;
   413→   413→    if (dd > 0 && !inDD) {
   414→   414→      inDD = true;
   415→   415→      recoveryStart = i;
   416→   416→    }
   417→   417→  }
   418→   418→
   419→   419→  const largestWin = wins.length > 0 ? Math.max(...wins.map((t) => t.netPnlCents)) : 0;
   420→   420→  const largestLoss = losses.length > 0 ? Math.min(...losses.map((t) => t.netPnlCents)) : 0;
   421→   421→
   422→   422→  // Streaks
   423→   423→  let cw = 0, cl = 0, maxCw = 0, maxCl = 0;
   424→   424→  for (const t of sorted) {
   425→   425→    if (t.netPnlCents > 0) { cw++; cl = 0; maxCw = Math.max(maxCw, cw); }
   426→   426→    else if (t.netPnlCents < 0) { cl++; cw = 0; maxCl = Math.max(maxCl, cl); }
   427→   427→    else { cw = 0; cl = 0; }
   428→   428→  }
   429→   429→

   438→   438→    avgPnlCents: avgPnl,
   439→   439→    avgWinCents: avgWin,
   440→   440→    avgLossCents: avgLoss,
   441→   441→    avgR,
   442→   442→    expectancyR,
   443→   443→    profitFactor,
   444→   444→    maxDrawdownCents: maxDD,
   445→   445→    currentDrawdownCents: currentDD,

   458→   458→export interface EquityPoint {
   459→   459→  date: string; // ISO date
   460→   460→  label: string;

   466→   466→export function buildEquityCurve(trades: TradeMetricRow[], startingBalanceCents: Cents = 0): EquityPoint[] {
   467→   467→  // Use only closed trades (WIN/LOSS/BREAKEVEN), ordered by EXIT timestamp
   468→   468→  const sorted = [...trades]
   469→   469→    .filter((t) => (t.status === "win" || t.status === "loss" || t.status === "breakeven") && (t.exitTime || t.entryTime))
   470→   470→    .sort((a, b) => {
   471→   471→      const ad = a.exitTime ? new Date(a.exitTime).getTime() : (a.entryTime ? new Date(a.entryTime).getTime() : 0);
   472→   472→      const bd = b.exitTime ? new Date(b.exitTime).getTime() : (b.entryTime ? new Date(b.entryTime).getTime() : 0);
   473→   473→      return ad - bd;
   474→   474→    });
   475→   475→  let cumulative = startingBalanceCents;
   476→   476→  return sorted.map((t) => {
   477→   477→    cumulative += t.netPnlCents;
   478→   478→    const d = new Date(t.exitTime || t.entryTime!);
   479→   479→    return {
   480→   480→      date: d.toISOString(),
   481→   481→      label: d.toLocaleDateString("en-US", { month: "short", day: "numeric" }),
   482→   482→      pnlCents: t.netPnlCents,
   483→   483→      cumulativeCents: cumulative,
   484→   484→      tradeId: t.id,
   485→   485→    };
   486→   486→  });
   487→   487→}

   493→   493→export interface DailyPnlPoint {
   494→   494→  date: string;
   495→   495→  label: string;

   497→   497→  tradeCount: number;

   499→   499→
   500→   500→export function buildDailyPnl(trades: TradeMetricRow[]): DailyPnlPoint[] {
   501→   501→  // Use only closed trades (WIN/LOSS/BREAKEVEN), grouped by EXIT timestamp
   502→   502→  const map = new Map<string, { pnl: Cents; count: number; ts: number }>();
   503→   503→  for (const t of trades) {
   504→   504→    if (t.status !== "win" && t.status !== "loss" && t.status !== "breakeven") continue;
   505→   505→    const ts = t.exitTime || t.entryTime;
   506→   506→    if (!ts) continue;
   507→   507→    const d = new Date(ts);
   508→   508→    const key = d.toISOString().slice(0, 10);
   509→   509→    const entry = map.get(key) ?? { pnl: 0, count: 0, ts: d.getTime() };
   510→   510→    entry.pnl += t.netPnlCents;
   511→   511→    entry.count += 1;
   512→   512→    map.set(key, entry);
   513→   513→  }
   514→   514→  return [...map.entries()]
   515→   515→    .sort((a, b) => a[1].ts - b[1].ts)
   516→   516→    .map(([date, e]) => ({
   517→   517→      date,
   518→   518→      label: new Date(date).toLocaleDateString("en-US", { month: "short", day: "numeric" }),
   519→   519→      pnlCents: e.pnl,
   520→   520→      tradeCount: e.count,
   521→   521→    }));
   522→   522→}

   528→   528→export interface RDistributionBucket {
   529→   529→  bucket: string;
   530→   530→  count: number;
   531→   531→  minR: number;
   532→   532→  maxR: number;
   533→   533→}
   534→   534→
   535→   535→export function buildRDistribution(trades: TradeMetricRow[]): RDistributionBucket[] {
   536→   536→  const buckets: { min: number; max: number; label: string }[] = [
   537→   537→    { min: -Infinity, max: -2, label: "≤ -2R" },
   538→   538→    { min: -2, max: -1, label: "-2 to -1R" },
   539→   539→    { min: -1, max: 0, label: "-1 to 0R" },
   540→   540→    { min: 0, max: 0.0001, label: "BE" },
   541→   541→    { min: 0.0001, max: 1, label: "0 to +1R" },
   542→   542→    { min: 1, max: 2, label: "+1 to +2R" },
   543→   543→    { min: 2, max: Infinity, label: "≥ +2R" },
   544→   544→  ];
   545→   545→  return buckets.map((b) => {
   546→   546→    const count = trades.filter((t) => {
   547→   547→      if (t.actualR == null) return false;
   548→   548→      const r = dToNumber(t.actualR);
   549→   549→      if (b.min === 0 && b.max === 0.0001) return r === 0;
   550→   550→      if (b.min === -Infinity) return r <= b.max;
   551→   551→      if (b.max === Infinity) return r > b.min;
   552→   552→      return r > b.min && r <= b.max;
   553→   553→    }).length;
   554→   554→    return { bucket: b.label, count, minR: b.min, maxR: b.max };
   555→   555→  });
   556→   556→}

   562→   562→export interface GroupBreakdown {
   563→   563→  key: string;
   564→   564→  label: string;
   565→   565→  trades: number;
   566→   566→  wins: number;
   567→   567→  winRate: Decimal | null;

   569→   569→  expectancyR: Decimal | null;
   570→   570→  pnlCents: Cents;

   573→   573→export function groupBy<T extends TradeMetricRow>(
   574→   574→  trades: T[],
   575→   575→  keyFn: (t: T) => string | null | undefined,
   576→   576→  labelFn: (k: string) => string = (k) => k,
   577→   577→): GroupBreakdown[] {
   578→   578→  const groups = new Map<string, T[]>();
   579→   579→  for (const t of trades) {
   580→   580→    const k = keyFn(t);
   581→   581→    if (k == null) continue;
   582→   582→    const arr = groups.get(k) ?? [];
   583→   583→    arr.push(t);
   584→   584→    groups.set(k, arr);
   585→   585→  }
   586→   586→  return [...groups.entries()].map(([key, arr]) => {
   587→   587→    const m = computeAggregateMetrics(arr);
   588→   588→    return {
   589→   589→      key,
   590→   590→      label: labelFn(key),
   591→   591→      trades: m.totalTrades,
   592→   592→      wins: m.wins,
   593→   593→      winRate: m.winRate,
   594→   594→      avgR: m.avgR,
   595→   595→      expectancyR: m.expectancyR,
   596→   596→      pnlCents: m.totalPnlCents,
   597→   597→    };
   598→   598→  }).sort((a, b) => b.trades - a.trades);
   599→   599→}

   605→   605→export interface Insight {
   606→   606→  id: string;
   607→   607→  title: string;
   608→   608→  body: string;
   609→   609→  sampleSize: number;
   610→   610→  metric: string;
   611→   611→  bucketLabel: string;
   612→   612→  comparison?: string;
   613→   613→  tradeIds: string[];
   614→   614→  warning?: "low_sample" | "missing_data";
   615→   615→}
   616→   616→
   617→   617→export function generateInsights(trades: TradeMetricRow[], minSample = 10): Insight[] {
   618→   618→  const insights: Insight[] = [];
   619→   619→  const closed = trades.filter((t) => t.status !== "open" && t.status !== "cancelled" && t.status !== "planned");
   620→   620→  if (closed.length < minSample) {
   621→   621→    if (closed.length > 0) {
   622→   622→      insights.push({
   623→   623→        id: "low-sample",
   624→   624→        title: "Not enough data yet",
   625→   625→        body: `You have ${closed.length} closed trade${closed.length === 1 ? "" : "s"}. DnD starts generating statistical insights once you have at least ${minSample} closed trades, so patterns you see now may not be reliable.`,
   626→   626→        sampleSize: closed.length,
   627→   627→        metric: "samples",
   628→   628→        bucketLabel: "all",
   629→   629→        tradeIds: closed.map((t) => t.id),
   630→   630→        warning: "low_sample",
   631→   631→      });
   632→   632→    }
   633→   633→    return insights;
   634→   634→  }
   635→   635→
   636→   636→  // Behavior flag insights
   637→   637→  const allFlags = new Set<string>();
   638→   638→  closed.forEach((t) => t.behaviorFlags?.forEach((f) => allFlags.add(f)));
   639→   639→  for (const flag of allFlags) {
   640→   640→    const withFlag = closed.filter((t) => t.behaviorFlags?.includes(flag));
   641→   641→    const withoutFlag = closed.filter((t) => !t.behaviorFlags?.includes(flag));
   642→   642→    if (withFlag.length < minSample) continue;
   643→   643→    const withR = withFlag.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   644→   644→    const withoutR = withoutFlag.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   645→   645→    if (withR.length === 0) continue;
   646→   646→    const withAvg = withR.reduce((s, r) => s + dToNumber(r), 0) / withR.length;
   647→   647→    const withoutAvg = withoutR.length > 0 ? withoutR.reduce((s, r) => s + dToNumber(r), 0) / withoutR.length : null;
   648→   648→    const flagLabel = humanizeFlag(flag);
   649→   649→    if (withoutAvg != null && Math.abs(withAvg - withoutAvg) > 0.05) {
   650→   650→      const direction = withAvg < withoutAvg ? "lower" : "higher";
   651→   651→      insights.push({
   652→   652→        id: `flag-${flag}`,
   653→   653→        title: `Trades tagged "${flagLabel}"`,
   654→   654→        body: `Trades tagged "${flagLabel}" have historically produced ${direction} average R (${withAvg.toFixed(2)}R across ${withFlag.length} trades) compared to ${withoutAvg.toFixed(2)}R across ${withoutFlag.length} trades without that flag.`,
   655→   655→        sampleSize: withFlag.length,
   656→   656→        metric: "avgR",
   657→   657→        bucketLabel: flagLabel,
   658→   658→        comparison: `vs ${withoutAvg.toFixed(2)}R`,
   659→   659→        tradeIds: withFlag.map((t) => t.id),
   660→   660→      });
   661→   661→    } else {
   662→   662→      insights.push({
   663→   663→        id: `flag-${flag}`,
   664→   664→        title: `Trades tagged "${flagLabel}"`,
   665→   665→        body: `Trades tagged "${flagLabel}" have an average realized R of ${withAvg.toFixed(2)}R across ${withFlag.length} trades.`,
   666→   666→        sampleSize: withFlag.length,
   667→   667→        metric: "avgR",
   668→   668→        bucketLabel: flagLabel,
   669→   669→        tradeIds: withFlag.map((t) => t.id),
   670→   670→      });
   671→   671→    }
   672→   672→  }
   673→   673→
   674→   674→  // Session insights
   675→   675→  const sessions = groupBy(closed, (t) => t.session, (k) => sessionLabel(k));
   676→   676→  for (const s of sessions) {
   677→   677→    if (s.trades < minSample) continue;
   678→   678→    if (s.avgR == null) continue;
   679→   679→    insights.push({
   680→   680→      id: `session-${s.key}`,
   681→   681→      title: `${s.label} session`,
   682→   682→      body: `${s.label} trades have produced an average of ${dToNumber(s.avgR).toFixed(2)}R across ${s.trades} trades, with a ${dToNumber(s.winRate).toFixed(1)}% win rate.`,
   683→   683→      sampleSize: s.trades,
   684→   684→      metric: "avgR",
   685→   685→      bucketLabel: s.label,
   686→   686→      tradeIds: closed.filter((t) => t.session === s.key).map((t) => t.id),
   687→   687→    });
   688→   688→  }
   689→   689→
   690→   690→  // A+ vs non-A+
   691→   691→  const aplus = closed.filter((t) => t.setupGrade === "A+");
   692→   692→  const nonAplus = closed.filter((t) => t.setupGrade && t.setupGrade !== "A+" && t.setupGrade !== "Invalid");
   693→   693→  if (aplus.length >= minSample && nonAplus.length >= minSample) {
   694→   694→    const aR = aplus.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   695→   695→    const nR = nonAplus.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   696→   696→    if (aR.length > 0 && nR.length > 0) {
   697→   697→      const aAvg = aR.reduce((s, r) => s + dToNumber(r), 0) / aR.length;
   698→   698→      const nAvg = nR.reduce((s, r) => s + dToNumber(r), 0) / nR.length;
   699→   699→      insights.push({
   700→   700→        id: "aplus-vs-non",
   701→   701→        title: "A+ vs non-A+ setups",
   702→   702→        body: `A+ setups have produced ${aAvg.toFixed(2)}R on average across ${aplus.length} trades, compared to ${nAvg.toFixed(2)}R across ${nonAplus.length} non-A+ trades.`,
   703→   703→        sampleSize: aplus.length + nonAplus.length,
   704→   704→        metric: "avgR",
   705→   705→        bucketLabel: "A+ vs non-A+",
   706→   706→        tradeIds: [...aplus, ...nonAplus].map((t) => t.id),
   707→   707→      });
   708→   708→    }
   709→   709→  }
   710→   710→
   711→   711→  return insights.sort((a, b) => b.sampleSize - a.sampleSize);
   712→   712→}

   732→   732→// ---------------------------------------------------------------------------
   733→   733→// Plan adherence (spec section 23 — planned vs actual execution)
   734→   734→// ---------------------------------------------------------------------------
   735→   735→
   736→   736→/**
   737→   737→ * Result of comparing a trade's planned values against its actual execution.
   738→   738→ *
   739→   739→ * `adherenceScore` is a 0–1 number representing the fraction of binary checks
   740→   740→ * (session match, target hit, risk-within-tolerance, positive R achieved) that
   741→   741→ * passed. Checks that can't be evaluated (e.g. no planned session recorded, no
   742→   742→ * target price, no account balance) are excluded from the denominator — so a
   743→   743→ * trade with only one check evaluated and that check passed scores 1.0, not
   744→   744→ * 0.25. If no checks can be evaluated at all, the score is 0.
   745→   745→ */
   746→   746→export interface PlanAdherenceCheck {
   747→   747→  label: string;
   748→   748→  passed: boolean | null;
   749→   749→  detail?: string;
   750→   750→}
   751→   751→
   752→   752→export interface PlanAdherence {
   753→   753→  plannedRR: Decimal | null;
   754→   754→  actualRR: Decimal | null;
   755→   755→  plannedRiskPct: Decimal | null;
   756→   756→  actualRiskPct: Decimal | null;
   757→   757→  sessionMatch: boolean | null;
   758→   758→  targetMatch: boolean | null;
   759→   759→  adherenceScore: number;
   760→   760→  adherencePct: number;
   761→   761→  checks: PlanAdherenceCheck[];
   762→   762→}
   763→   763→
   764→   764→/**
   765→   765→ * Trade-shape accepted by `calculatePlanAdherence`. Accepts the loose shape
   766→   766→ * produced by Prisma's `Trade` row (decimal strings + nullable fields) plus an
   767→   767→ * optional `accountBalanceCents` so we can derive `actualRiskPct` from the
   768→   768→ * risk amount vs the account balance.
   769→   769→ */
   770→   770→export interface PlanAdherenceTrade {
   771→   771→  direction?: string | null;
   772→   772→  /** Actual session the trade was taken in (from `Trade.session`). */
   773→   773→  session?: string | null;
   774→   774→  /** Planned session (from `setupJson.session` or a daily plan). */
   775→   775→  plannedSession?: string | null;
   776→   776→  plannedEntryPrice?: Decimal | null;
   777→   777→  plannedStopPrice?: Decimal | null;
   778→   778→  plannedTargetPrice?: Decimal | null;
   779→   779→  plannedRiskPct?: Decimal | null;
   780→   780→  entryPriceAvg?: Decimal | null;
   781→   781→  exitPriceAvg?: Decimal | null;
   782→   782→  /** Pre-computed actual R (decimal string). */
   783→   783→  actualR?: Decimal | null;
   784→   784→  /** Risk amount in cents — typically from `calculateTradePnl.riskAmountCents`. */
   785→   785→  riskAmountCents?: number | null;
   786→   786→  /** Account balance in cents — used to derive `actualRiskPct`. */
   787→   787→  accountBalanceCents?: number | null;
   788→   788→}
   789→   789→
   790→   790→/**
   791→   791→ * Compute plan adherence for a single trade. Pure function — no I/O.
   792→   792→ *
   793→   793→ * Binary checks (each 0/1 or null when undeterminable):
   794→   794→ *   - `sessionMatch`: planned session === actual session
   795→   795→ *   - `targetMatch`: |exitPrice - target| / |target| <= 5% (within 5% of target)
   796→   796→ *   - `riskWithinTolerance`: actual risk is within 50% of planned risk
   797→   797→ *     (ratio of actual/planned between 0.5 and 1.5)
   798→   798→ *   - `rrAchieved`: actualR > 0 (the trade was profitable)
   799→   799→ *
   800→   800→ * `adherenceScore` = mean of the non-null checks (or 0 if all null).
   801→   801→ */
   802→   802→export function calculatePlanAdherence(trade: PlanAdherenceTrade): PlanAdherence {
   803→   803→  // Planned R:R = |target - entry| / |entry - stop|
   804→   804→  let plannedRR: Decimal | null = null;
   805→   805→  if (trade.plannedEntryPrice && trade.plannedStopPrice && trade.plannedTargetPrice) {
   806→   806→    const stopDist = dAbs(dSub(trade.plannedEntryPrice, trade.plannedStopPrice));
   807→   807→    const targetDist = dAbs(dSub(trade.plannedTargetPrice, trade.plannedEntryPrice));
   808→   808→    if (!dIsZero(stopDist)) {
   809→   809→      plannedRR = dDiv(targetDist, stopDist);
   810→   810→    }
   811→   811→  }
   812→   812→
   813→   813→  // Actual R: already computed server-side — normalize and pass through.
   814→   814→  const actualRR: Decimal | null =
   815→   815→    trade.actualR != null && trade.actualR !== "" ? dNormalize(trade.actualR) : null;
   816→   816→
   817→   817→  // Planned risk %
   818→   818→  const plannedRiskPct: Decimal | null =
   819→   819→    trade.plannedRiskPct != null && trade.plannedRiskPct !== ""
   820→   820→      ? dNormalize(trade.plannedRiskPct)
   821→   821→      : null;
   822→   822→
   823→   823→  // Actual risk %: riskAmount / accountBalance (if both available and balance > 0)
   824→   824→  let actualRiskPct: Decimal | null = null;
   825→   825→  if (
   826→   826→    trade.riskAmountCents != null &&
   827→   827→    trade.accountBalanceCents != null &&
   828→   828→    trade.accountBalanceCents > 0 &&
   829→   829→    trade.riskAmountCents >= 0
   830→   830→  ) {
   831→   831→    actualRiskPct = dDiv(
   832→   832→      String(trade.riskAmountCents / 100),
   833→   833→      String(trade.accountBalanceCents / 100),
   834→   834→    );
   835→   835→  }
   836→   836→
   837→   837→  // Session match — null when either planned or actual session is missing.
   838→   838→  const plannedSession = trade.plannedSession ?? null;
   839→   839→  const actualSession = trade.session ?? null;
   840→   840→  const sessionMatch: boolean | null =
   841→   841→    plannedSession && actualSession ? plannedSession === actualSession : null;
   842→   842→
   843→   843→  // Target match — exit price within 5% of planned target.
   844→   844→  let targetMatch: boolean | null = null;
   845→   845→  if (trade.exitPriceAvg && trade.plannedTargetPrice) {
   846→   846→    const exit = dToNumber(trade.exitPriceAvg);
   847→   847→    const target = dToNumber(trade.plannedTargetPrice);
   848→   848→    if (Number.isFinite(exit) && Number.isFinite(target) && target !== 0) {
   849→   849→      const deviation = Math.abs(exit - target) / Math.abs(target);
   850→   850→      targetMatch = deviation <= 0.05;
   851→   851→    }
   852→   852→  }
   853→   853→
   854→   854→  // Risk-within-tolerance — actual risk is within 50% of planned risk
   855→   855→  // (i.e. ratio of actual/planned is between 0.5 and 1.5).
   856→   856→  let riskWithinTolerance: boolean | null = null;
   857→   857→  if (plannedRiskPct && actualRiskPct) {
   858→   858→    const planned = dToNumber(plannedRiskPct);
   859→   859→    const actual = dToNumber(actualRiskPct);
   860→   860→    if (planned > 0) {
   861→   861→      const ratio = actual / planned;
   862→   862→      riskWithinTolerance = ratio >= 0.5 && ratio <= 1.5;
   863→   863→    }
   864→   864→  }
   865→   865→
   866→   866→  // R:R achieved — actualR > 0 (i.e. the trade was profitable).
   867→   867→  const rrAchieved: boolean | null =
   868→   868→    actualRR != null ? dToNumber(actualRR) > 0 : null;
   869→   869→
   870→   870→  const checks: PlanAdherenceCheck[] = [
   871→   871→    {
   872→   872→      label: "Session match",
   873→   873→      passed: sessionMatch,
   874→   874→      detail:
   875→   875→        sessionMatch === null
   876→   876→          ? "No planned session recorded"
   877→   877→          : sessionMatch
   878→   878→            ? "Traded in the planned session"
   879→   879→            : "Traded outside the planned session",
   880→   880→    },
   881→   881→    {
   882→   882→      label: "Target hit",
   883→   883→      passed: targetMatch,
   884→   884→      detail:
   885→   885→        targetMatch === null
   886→   886→          ? "No target or exit price recorded"
   887→   887→          : targetMatch
   888→   888→            ? "Exit price within 5% of planned target"
   889→   889→            : "Exit price deviated more than 5% from target",
   890→   890→    },
   891→   891→    {
   892→   892→      label: "Risk within tolerance",
   893→   893→      passed: riskWithinTolerance,
   894→   894→      detail:
   895→   895→        riskWithinTolerance === null
   896→   896→          ? "Planned or actual risk unavailable"
   897→   897→          : riskWithinTolerance
   898→   898→            ? "Actual risk within 50% of planned risk"
   899→   899→            : "Actual risk outside 50% tolerance of planned risk",
   900→   900→    },
   901→   901→    {
   902→   902→      label: "Positive R achieved",
   903→   903→      passed: rrAchieved,
   904→   904→      detail:
   905→   905→        rrAchieved === null
   906→   906→          ? "No R-multiple computed"
   907→   907→          : rrAchieved
   908→   908→            ? "Trade was profitable (R > 0)"
   909→   909→            : "Trade was not profitable (R ≤ 0)",
   910→   910→    },
   911→   911→  ];
   912→   912→
   913→   913→  // Adherence score: average of non-null binary checks. Each check contributes
   914→   914→  // 1 (passed) or 0 (failed); null checks are excluded from the denominator.
   915→   915→  const evaluatedChecks = checks
   916→   916→    .map((c) => c.passed)
   917→   917→    .filter((v): v is boolean => v !== null);
   918→   918→  const adherenceScore =
   919→   919→    evaluatedChecks.length > 0
   920→   920→      ? evaluatedChecks.reduce((sum, v) => sum + (v ? 1 : 0), 0) / evaluatedChecks.length
   921→   921→      : 0;
   922→   922→  const adherencePct = adherenceScore * 100;
   923→   923→
   924→   924→  return {
   925→   925→    plannedRR,
   926→   926→    actualRR,
   927→   927→    plannedRiskPct,
   928→   928→    actualRiskPct,
   929→   929→    sessionMatch,
   930→   930→    targetMatch,
   931→   931→    adherenceScore,
   932→   932→    adherencePct,
   933→   933→    checks,
   934→   934→  };
   935→   935→}
