   371→  function setField(path: string, value: any) {
   372→    setForm((f: any) => {
   373→      const next = { ...f };
   374→      const parts = path.split(".");
   375→      let cur: any = next;
   376→      for (let i = 0; i < parts.length - 1; i++) {
   377→        cur[parts[i]] = { ...cur[parts[i]] };
   378→        cur = cur[parts[i]];
   379→      }
   380→      cur[parts[parts.length - 1]] = value;
   381→      return next;
   382→    });
   383→  }

   385→  function toggleArrayField(path: string, value: string) {
   386→    setForm((f: any) => {
   387→      const next = { ...f };
   388→      const parts = path.split(".");
   389→      let cur: any = next;
   390→      for (let i = 0; i < parts.length - 1; i++) {
   391→        cur[parts[i]] = { ...cur[parts[i]] };
   392→        cur = cur[parts[i]];
   393→      }
   394→      const key = parts[parts.length - 1];
   395→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   396→      const idx = arr.indexOf(value);
   397→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   398→      cur[key] = arr;
   399→      return next;
   400→    });
   401→  }

   403→  // Update one A+ checklist answer field (`checked` or `note`) for the given
   404→  // item id. Used by the checkbox + evidence-note inputs in the ICT Setup tab.
   405→  function updateChecklistAnswer(
   406→    itemId: string,
   407→    field: "checked" | "note",
   408→    value: boolean | string,
   409→  ) {
   410→    setForm((f: any) => {
   411→      const current = f.checklistAnswers?.[itemId] ?? { checked: false };
   412→      const next = {
   413→        ...f.checklistAnswers,
   414→        [itemId]: {
   415→          ...current,
   416→          [field]: value,
   417→        },
   418→      };
   419→      // If a checked item is being unchecked, drop its note so storage stays clean.
   420→      if (field === "checked" && value === false && next[itemId].note) {
   421→        delete next[itemId].note;
   422→      }
   423→      return { ...f, checklistAnswers: next };
   424→    });

   427→  async function submit(isDraft: boolean = false) {
   428→    if (!form.accountId) {
   429→      toast.error("Please select an account.");
   430→      return;
   431→    }
   432→    if (!form.instrumentSymbol) {
   433→      toast.error("Please enter an instrument symbol.");
   434→      return;
   435→    }
   436→    setSaving(true);
   437→    try {
   438→      const executions = [];
   439→      if (form.entryPrice) {
   440→        executions.push({
   441→          kind: "entry",
   442→          price: form.entryPrice,
   443→          quantity: form.quantity || "1",
   444→          timestamp: form.entryTime ? new Date(form.entryTime).toISOString() : new Date().toISOString(),
   445→        });
   446→      }
   447→      if (form.exitPrice) {
   448→        executions.push({
   449→          kind: "exit",
   450→          price: form.exitPrice,
   451→          quantity: form.quantity || "1",
   452→          timestamp: form.exitTime ? new Date(form.exitTime).toISOString() : new Date().toISOString(),
   453→        });
   454→      }
   455→      const payload = {
   456→        accountId: form.accountId,
   457→        instrumentSymbol: form.instrumentSymbol.toUpperCase(),
   458→        market: meta?.instruments?.find((i: any) => i.symbol === form.instrumentSymbol.toUpperCase())?.market ?? "custom",
   459→        direction: form.direction,
   460→        session: form.session,
   461→        strategyId: form.strategyId || null,
   462→        // Resolve the strategy version: when a strategy is selected, use the
   463→        // latest version (`versions[0]`, fetched via strategyDetail) so the
   464→        // trade is always linked to the user's current rule set. Stays null
   465→        // when no strategy is selected.
   466→        strategyVersionId: form.strategyId ? (strategyDetail?.versions?.[0]?.id ?? null) : null,
   467→        dailyPlanId: form.dailyPlanId || null,
   468→        // Setup grade/score are now derived from the A+ checklist evaluation,
   469→        // not a manual dropdown. The server re-computes and overrides these
   470→        // values from the answers to prevent client tampering (spec §102).
   471→        checklistVersionId: form.checklistVersionId || null,
   472→        checklistAnswers: form.checklistVersionId ? form.checklistAnswers : null,
   473→        setupGrade: form.checklistVersionId ? evaluation.grade : null,
   474→        setupScore: form.checklistVersionId ? String(evaluation.score) : null,
   475→        tags: form.tags,
   476→        plannedEntryPrice: form.plannedEntryPrice || null,
   477→        plannedStopPrice: form.plannedStopPrice || null,
   478→        plannedTargetPrice: form.plannedTargetPrice || null,
   479→        plannedRiskPct: form.plannedRiskPct || null,
   480→        executions,
   481→        feesCents: Number(form.feesCents) || 0,
   482→        commissionCents: Number(form.commissionCents) || 0,
   483→        swapCents: Number(form.swapCents) || 0,
   484→        slippageCents: Number(form.slippageCents) || 0,
   485→        entryTime: form.entryTime ? new Date(form.entryTime).toISOString() : null,
   486→        exitTime: form.exitTime ? new Date(form.exitTime).toISOString() : null,
   487→        tradingTimezone: form.tradingTimezone,
   488→        setup: form.setup,
   489→        thesis: form.thesis,
   490→        notes: form.notes || null,
   491→        lessons: form.lessons || null,
   492→        psychBefore: form.psychBefore,
   493→        psychAfter: form.psychAfter,
   494→        behaviorFlags: form.behaviorFlags,
   495→        ruleCompliance: form.ruleCompliance,
   496→        isDraft,
   497→      };
   498→      const url = editingId ? `/api/trades/${editingId}` : "/api/trades";
   499→      const method = editingId ? "PATCH" : "POST";
   500→      const res = await fetch(url, {
   501→        method,
   502→        headers: { "Content-Type": "application/json" },
   503→        body: JSON.stringify(payload),
   504→      });
   505→      const data = await res.json();
   506→      if (!res.ok) {
   507→        toast.error(data?.error ?? "Failed to save trade.");
   508→        setSaving(false);
   509→        return;
   510→      }
   511→      // Attach any orphaned uploaded media to the newly created trade and save captions
   512→      if (form.uploadedMedia?.length > 0) {
