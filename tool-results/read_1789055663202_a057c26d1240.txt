   102→async function fetchPlans() {
   103→  const res = await fetch("/api/daily-plans", { cache: "no-store" });
   104→  if (!res.ok) throw new Error("Failed to load daily plans");
   105→  return res.json();
   106→}
   107→
   108→function planToForm(plan: DailyPlan): PlanFormState {
   109→  // The date stored in the DB is a UTC ISO string. We extract the YYYY-MM-DD
   110→  // portion to populate the <input type="date"> control without timezone
   111→  // shifting it back/forward a day.
   112→  const dateStr = plan.date ? plan.date.slice(0, 10) : new Date().toISOString().slice(0, 10);
   113→  return {
   114→    date: dateStr,
   115→    weeklyBias: plan.weeklyBias ?? "",
   116→    dailyBias: plan.dailyBias ?? "",
   117→    instruments: plan.instruments ?? "",
   118→    pwh: plan.pwh ?? "",
   119→    pwl: plan.pwl ?? "",
   120→    pdh: plan.pdh ?? "",
   121→    pdl: plan.pdl ?? "",
   122→    htfLevelsJson: plan.htfLevelsJson ?? "",
   123→    liquidityTargets: plan.liquidityTargets ?? "",
   124→    session: plan.session ?? "__none",
   125→    setupConditions: plan.setupConditions ?? "",
   126→    invalidation: plan.invalidation ?? "",
   127→    maxTrades: plan.maxTrades != null ? String(plan.maxTrades) : "",
   128→    maxDailyRiskPct: plan.maxDailyRiskPct ?? "",
   129→    notes: plan.notes ?? "",
   130→  };
   131→}
   132→
   133→function formToPayload(form: PlanFormState) {
   134→  // Numbers / decimals are sent as null when blank to keep the schema
   135→  // optional (no zero-defaulting surprises).
   136→  return {
   137→    date: form.date,
   138→    weeklyBias: form.weeklyBias || undefined,
   139→    dailyBias: form.dailyBias || undefined,
   140→    instruments: form.instruments || undefined,
   141→    pwh: form.pwh || undefined,
   142→    pwl: form.pwl || undefined,
   143→    pdh: form.pdh || undefined,
   144→    pdl: form.pdl || undefined,
   145→    htfLevelsJson: form.htfLevelsJson || undefined,
   146→    liquidityTargets: form.liquidityTargets || undefined,
   147→    session: form.session && form.session !== "__none" ? form.session : undefined,
   148→    setupConditions: form.setupConditions || undefined,
   149→    invalidation: form.invalidation || undefined,
   150→    maxTrades: form.maxTrades === "" ? undefined : Number(form.maxTrades),
   151→    maxDailyRiskPct: form.maxDailyRiskPct || undefined,
   152→    notes: form.notes || undefined,
   153→  };
   154→}

   160→export function PlansView() {
   161→  const qc = useQueryClient();
   162→  const { data, isLoading } = useQuery({ queryKey: ["daily-plans"], queryFn: fetchPlans });
   163→  const { params } = useNav();
   164→  const [selectedId, setSelectedId] = useState<string | null>(null);
   165→  const [dialogOpen, setDialogOpen] = useState(false);
   166→  const [editingPlan, setEditingPlan] = useState<DailyPlan | null>(null);
   167→  // When the user deep-links in with a `?date=YYYY-MM-DD` (e.g. from the
   168→  // calendar's "Create Plan" button), we open the create dialog with that
   169→  // date pre-filled. The value is cleared after the dialog opens to avoid
   170→  // re-opening it on every render.
   171→  const [initialDate, setInitialDate] = useState<string | null>(null);
   172→
   173→  const plans: DailyPlan[] = data?.items ?? [];
   174→
   175→  // Allow deep-linking into a specific plan: when the calendar's day-detail
   176→  // dialog navigates to "plans" with `{ id: <planId> }`, auto-select that
   177→  // plan in the list.
   178→  useEffect(() => {
   179→    if (params?.id && plans.some((p) => p.id === params.id)) {
   180→      setSelectedId(params.id);
   181→    }
   182→  }, [params?.id, plans]);
   183→
   184→  // Deep-link from the calendar's "Create Plan" CTA: when arriving with a
   185→  // `date` param, open the create dialog with that date pre-filled.
   186→  useEffect(() => {
   187→    if (params?.date && !dialogOpen) {
   188→      setEditingPlan(null);
   189→      setInitialDate(params.date);
   190→      setDialogOpen(true);
   191→    }
   192→  }, [params?.date, dialogOpen]);
   193→
   194→  // Sort newest first. The API already returns desc by date, but be defensive
   195→  // and also group by month for visual scanability.
   196→  const sortedPlans = useMemo(() => {
   197→    return [...plans].sort((a, b) => (a.date < b.date ? 1 : a.date > b.date ? -1 : 0));
   198→  }, [plans]);
   199→
   200→  // Group plans by "Month Year" label for the list display.
   201→  const grouped = useMemo(() => {
   202→    const map = new Map<string, DailyPlan[]>();
   203→    for (const p of sortedPlans) {
   204→      const d = new Date(p.date);
   205→      const label = d.toLocaleDateString("en-US", { month: "long", year: "numeric" });
   206→      const arr = map.get(label) ?? [];
   207→      arr.push(p);
   208→      map.set(label, arr);
   209→    }
   210→    return Array.from(map.entries());
   211→  }, [sortedPlans]);
   212→
   213→  const selected = selectedId ? plans.find((p) => p.id === selectedId) ?? null : null;
   214→
   215→  function openCreate() {
   216→    setEditingPlan(null);
   217→    setInitialDate(null);
   218→    setDialogOpen(true);
   219→  }
   220→
   221→  function openEdit(plan: DailyPlan) {
   222→    setEditingPlan(plan);
   223→    setInitialDate(null);
   224→    setDialogOpen(true);
   225→  }
   226→
   227→  function onSaved() {
   228→    qc.invalidateQueries({ queryKey: ["daily-plans"] });
   229→    setDialogOpen(false);
   230→    setEditingPlan(null);
   231→    setInitialDate(null);
   232→  }
   233→
   234→  return (
   235→    <div className="p-4 md:p-6 grid lg:grid-cols-3 gap-4">
   236→      {/* List */}
   237→      <Card className="p-3 lg:col-span-1">
   238→        <div className="flex items-center justify-between mb-3">
   239→          <div>
   240→            <h1 className="font-semibold">Daily Plans</h1>
   241→            <p className="text-xs text-muted-foreground">Pre-market bias, levels &amp; limits.</p>
   242→          </div>
   243→          <Dialog open={dialogOpen} onOpenChange={(o) => { setDialogOpen(o); if (!o) { setEditingPlan(null); setInitialDate(null); } }}>
   244→            <DialogTrigger asChild>
   245→              <Button size="sm" onClick={openCreate}>
   246→                <Plus className="h-4 w-4" />
   247→              </Button>
   248→            </DialogTrigger>
   249→            <PlanFormDialog
   250→              key={editingPlan?.id ?? (initialDate ? `new-${initialDate}` : "new")}
   251→              plan={editingPlan}
   252→              initialDate={initialDate}
   253→              onClose={() => { setDialogOpen(false); setEditingPlan(null); setInitialDate(null); }}
   254→              onSaved={onSaved}
   255→            />
   256→          </Dialog>
   257→        </div>
   258→
   259→        {isLoading ? (
   260→          <div className="space-y-2">
   261→            <Skeleton className="h-12" />
   262→            <Skeleton className="h-12" />
   263→            <Skeleton className="h-12" />
   264→          </div>
   265→        ) : sortedPlans.length === 0 ? (
   266→          <EmptyState
   267→            title="No daily plans yet"
   268→            description="Create your first daily plan to record pre-market bias, key levels, session focus and risk limits for each trading day."
   269→            icon={ClipboardList}
   270→            action={{ label: "New Daily Plan", onClick: openCreate }}
   271→          />
   272→        ) : (
   273→          <div className="space-y-4 max-h-[70vh] overflow-y-auto scroll-thin pr-1">
   274→            {grouped.map(([label, items]) => (
   275→              <div key={label} className="space-y-1">
   276→                <div className="text-[11px] uppercase tracking-wide text-muted-foreground px-1 sticky top-0 bg-card py-1">
   277→                  {label}
   278→                </div>
   279→                {items.map((p) => {
   280→                  const d = new Date(p.date);
   281→                  const dateLabel = d.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" });
   282→                  const isBullish = p.dailyBias === "bullish" || p.weeklyBias === "bullish";
   283→                  const isBearish = p.dailyBias === "bearish" || p.weeklyBias === "bearish";
   284→                  return (
   285→                    <button
   286→                      key={p.id}
   287→                      onClick={() => setSelectedId(p.id)}
   288→                      className={cn(
   289→                        "w-full text-left p-2 rounded-md border border-border hover:bg-muted/40 transition-colors",
   290→                        selectedId === p.id && "ring-2 ring-primary border-primary/30",
   291→                      )}
   292→                    >
   293→                      <div className="flex items-center gap-2">
   294→                        <span className="font-medium text-sm">{dateLabel}</span>
   295→                        {p.session && (
   296→                          <Badge variant="outline" className="text-[10px] uppercase">{p.session.replace("_", " ")}</Badge>
   297→                        )}
   298→                        <div className="ml-auto flex items-center gap-1">
   299→                          {isBullish && <Badge variant="outline" className="text-[10px] text-profit border-profit/40">Bull</Badge>}
   300→                          {isBearish && <Badge variant="outline" className="text-[10px] text-loss border-loss/40">Bear</Badge>}
   301→                        </div>
   302→                      </div>
   303→                      <div className="text-xs text-muted-foreground mt-1 line-clamp-1">
   304→                        {p.instruments || "No instruments"} · {p.setupConditions || "No setup conditions"}
   305→                      </div>
   306→                    </button>
   307→                  );
   308→                })}
   309→              </div>
   310→            ))}
   311→          </div>
   312→        )}
   313→      </Card>
   314→
   315→      {/* Detail */}
   316→      <div className="lg:col-span-2 space-y-4">
   317→        {!selected ? (
   318→          <Card className="p-8 text-center">
   319→            <ClipboardList className="h-8 w-8 mx-auto text-muted-foreground mb-3" />
   320→            <h3 className="font-medium">Select a daily plan</h3>
   321→            <p className="text-sm text-muted-foreground mt-1">
   322→              Daily plans capture your pre-market work: bias, key HTF levels, liquidity targets, session focus, and the rules that invalidate the thesis.
   323→            </p>
   324→          </Card>
   325→        ) : (
   326→          <PlanDetail
   327→            plan={selected}
   328→            onEdit={() => openEdit(selected)}
   329→            onDeleted={() => { setSelectedId(null); qc.invalidateQueries({ queryKey: ["daily-plans"] }); }}
   330→          />
   331→        )}
   332→      </div>
   333→    </div>
   334→  );
   335→}

   341→function PlanDetail({ plan, onEdit, onDeleted }: { plan: DailyPlan; onEdit: () => void; onDeleted: () => void }) {
   342→  const [deleting, setDeleting] = useState(false);
   343→  const { navigate } = useNav();
   344→  const d = new Date(plan.date);
   345→  const dateLabel = d.toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric", year: "numeric" });
   346→
   347→  async function handleDelete() {
   348→    if (!confirm("Delete this daily plan? Trades linked to it will be unlinked but kept.")) return;
   349→    setDeleting(true);
   350→    try {
   351→      const res = await fetch(`/api/daily-plans/${plan.id}`, { method: "DELETE" });
   352→      if (res.ok) {
   353→        toast.success("Daily plan deleted.");
   354→        onDeleted();
   355→      } else {
   356→        const d = await res.json().catch(() => null);
   357→        toast.error(d?.error ?? "Failed to delete plan.");
   358→      }
   359→    } finally {
   360→      setDeleting(false);
   361→    }
   362→  }
   363→
   364→  return (
   365→    <Card className="p-4 md:p-6 space-y-4">
   366→      <div className="flex items-start gap-3">
   367→        <div className="flex-1 min-w-0">
   368→          <div className="text-xs uppercase tracking-wide text-muted-foreground">Daily Plan</div>
   369→          <h2 className="text-xl font-semibold tracking-tight">{dateLabel}</h2>
   370→          <div className="flex flex-wrap items-center gap-2 mt-2">
   371→            {plan.weeklyBias && <BiasBadge label="Weekly" value={plan.weeklyBias} />}
   372→            {plan.dailyBias && <BiasBadge label="Daily" value={plan.dailyBias} />}
   373→            {plan.session && (
   374→              <Badge variant="outline" className="uppercase text-xs">
   375→                {plan.session.replace("_", " ")}
   376→              </Badge>
   377→            )}
   378→            {plan.maxTrades != null && (
   379→              <Badge variant="outline" className="text-xs">Max {plan.maxTrades} trades</Badge>
   380→            )}
   381→            {plan.maxDailyRiskPct && (
   382→              <Badge variant="outline" className="text-xs">Risk {(Number(plan.maxDailyRiskPct) * 100).toFixed(1)}%</Badge>
   383→            )}
   384→          </div>
   385→        </div>
   386→        <div className="flex items-center gap-1">
   387→          <Button
   388→            size="sm"
   389→            variant="default"
   390→            onClick={() => navigate("tradeNew", { dailyPlanId: plan.id })}
   391→          >
   392→            <TrendingUp className="h-3.5 w-3.5" /> Create Trade From Plan
   393→          </Button>
   394→          <Button size="sm" variant="outline" onClick={onEdit}>
   395→            <Pencil className="h-3.5 w-3.5" /> Edit
   396→          </Button>
   397→          <Button
   398→            size="icon"
   399→            variant="ghost"
   400→            className="h-8 w-8 text-muted-foreground hover:text-loss"
   401→            onClick={handleDelete}
   402→            disabled={deleting}
   403→            aria-label="Delete plan"
   404→          >
   405→            {deleting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Trash2 className="h-3.5 w-3.5" />}
   406→          </Button>
   407→        </div>
   408→      </div>
   409→
   410→      <Separator />
   411→
   412→      <div className="grid sm:grid-cols-2 gap-4">
   413→        <Field label="Instruments" value={plan.instruments} />
   414→        <Field label="Session Focus" value={plan.session ? plan.session.replace("_", " ").toUpperCase() : null} />
   415→      </div>
   416→
   417→      <div>
   418→        <h3 className="text-sm font-medium mb-2">Key Levels</h3>
   419→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   420→          <LevelField label="PWH" value={plan.pwh} />
   421→          <LevelField label="PWL" value={plan.pwl} />
   422→          <LevelField label="PDH" value={plan.pdh} />
   423→          <LevelField label="PDL" value={plan.pdl} />
   424→        </div>
   425→      </div>
   426→
   427→      {plan.liquidityTargets && (
   428→        <div>
   429→          <h3 className="text-sm font-medium mb-1">Liquidity Targets</h3>
   430→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.liquidityTargets}</p>
   431→        </div>
   432→      )}
   433→
   434→      {plan.htfLevelsJson && (
   435→        <div>
   436→          <h3 className="text-sm font-medium mb-1">HTF Levels</h3>
   437→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.htfLevelsJson}</p>
   438→        </div>
   439→      )}
   440→
   441→      {plan.setupConditions && (
   442→        <div>
   443→          <h3 className="text-sm font-medium mb-1">Setup Conditions</h3>
   444→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.setupConditions}</p>
   445→        </div>
   446→      )}
   447→
   448→      {plan.invalidation && (
   449→        <div>
   450→          <h3 className="text-sm font-medium mb-1">Invalidation</h3>
   451→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.invalidation}</p>
   452→        </div>
   453→      )}
   454→
   455→      {plan.notes && (
   456→        <div>
   457→          <h3 className="text-sm font-medium mb-1">Notes</h3>
   458→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.notes}</p>
   459→        </div>
   460→      )}
   461→
   462→      <div className="text-xs text-muted-foreground pt-2 border-t border-border">
   463→        Created {new Date(plan.createdAt).toLocaleString()} · Updated {new Date(plan.updatedAt).toLocaleString()}
   464→      </div>
   465→    </Card>
   466→  );
   467→}

   469→function BiasBadge({ label, value }: { label: string; value: string }) {
   470→  const v = value.toLowerCase();
   471→  const tone = v === "bullish" ? "text-profit border-profit/40" : v === "bearish" ? "text-loss border-loss/40" : "";
   472→  return (
   473→    <Badge variant="outline" className={cn("text-xs", tone)}>
   474→      {label}: <span className="capitalize ml-1">{value}</span>
   475→    </Badge>
   476→  );
   477→}

   509→function PlanFormDialog({
   510→  plan,
   511→  initialDate,
   512→  onClose,
   513→  onSaved,
   514→}: {
   515→  plan: DailyPlan | null;
   516→  initialDate?: string | null;
   517→  onClose: () => void;
   518→  onSaved: () => void;
   519→}) {
   520→  const [form, setForm] = useState<PlanFormState>(EMPTY_FORM);
   521→  const [saving, setSaving] = useState(false);
   522→
   523→  useEffect(() => {
   524→    if (plan) {
   525→      setForm(planToForm(plan));
   526→    } else {
   527→      // Pre-fill the date with the deep-linked `initialDate` (e.g. when the
   528→      // calendar's "Create Plan" button passes `?date=YYYY-MM-DD`); fall
   529→      // back to today's date when none was provided.
   530→      const date = initialDate ?? new Date().toISOString().slice(0, 10);
   531→      setForm({ ...EMPTY_FORM, date });
   532→    }
   533→  }, [plan, initialDate]);
   534→
   535→  function setField<K extends keyof PlanFormState>(key: K, value: PlanFormState[K]) {
   536→    setForm((f) => ({ ...f, [key]: value }));
   537→  }
   538→
   539→  async function submit() {
   540→    if (!form.date) {
   541→      toast.error("Date is required.");
