    71→async function fetchSettings() {
    72→  const res = await fetch("/api/settings", { cache: "no-store" });
    73→  if (!res.ok) throw new Error("Failed to fetch settings");
    74→  return res.json();
    75→}

   591→// ---------------------------------------------------------------------------
   592→// LegalDoc — preserved from prior task (reads /api/legal/[doc], allows
   593→// acceptance / re-acceptance on version changes).
   594→// ---------------------------------------------------------------------------
   595→
   596→function LegalDoc({ doc, label }: { doc: string; label: string }) {
   597→  const [state, setState] = useState<any>(null);
   598→  const [open, setOpen] = useState(false);
   599→
   600→  useEffect(() => {
   601→    fetch(`/api/legal/${doc}`).then((r) => r.json()).then(setState);
   602→  }, [doc]);
   603→
   604→  async function accept() {
   605→    const res = await fetch(`/api/legal/${doc}`, {
   606→      method: "POST",
   607→      headers: { "Content-Type": "application/json" },
   608→      body: JSON.stringify({}),
   609→    });
   610→    if (res.ok) {
   611→      toast.success(`${label} accepted.`);
   612→      const data = await res.json();
   613→      setState({
   614→        ...state,
   615→        accepted: true,
   616→        acceptedAt: data.acceptedAt,
   617→        isCurrent: true,
   618→        version: data.version,
   619→      });
   620→    }
   621→  }
   622→
   623→  const needsAccept = state && (!state.accepted || state.isCurrent === false);
   624→
   625→  return (
   626→    <div className="flex items-center justify-between gap-3 py-2 rounded-lg hover:bg-muted/50 px-2 -mx-2">
   627→      <div className="min-w-0">
   628→        <div className="font-medium text-sm text-foreground">{label}</div>
   629→        <div className="text-xs text-muted-foreground">
   630→          {state ? (
   631→            <>
   632→              Version {state.version}
   633→              {" · "}
   634→              {state.accepted
   635→                ? `Accepted ${new Date(state.acceptedAt).toLocaleDateString()}`
   636→                : "Not yet accepted"}
   637→              {state.isCurrent === false && (
   638→                <span className="text-warning"> · outdated (v{state.acceptedVersion || "?"})</span>
   639→              )}
   640→            </>
   641→          ) : (
   642→            "Loading…"
   643→          )}
   644→        </div>
   645→      </div>
   646→      <div className="flex gap-2 shrink-0">
   647→        <Button
   648→          size="sm"
   649→          variant="ghost"
   650→          onClick={() => setOpen(true)}
   651→          className="text-muted-foreground hover:text-foreground hover:bg-background"
   652→        >
   653→          View
   654→        </Button>
   655→        {needsAccept && (
   656→          <Button
   657→            size="sm"
   658→            onClick={accept}
   659→            className="bg-primary text-primary-foreground hover:bg-primary/90"
   660→          >
   661→            Accept
   662→          </Button>
   663→        )}
   664→      </div>
   665→      <Dialog open={open} onOpenChange={setOpen}>
   666→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
   667→          <DialogHeader>
   668→            <DialogTitle>{label}</DialogTitle>
   669→            {state && <DialogDescription>Version {state.version}</DialogDescription>}
   670→          </DialogHeader>
   671→          <div className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">
   672→            {state?.content ?? "Document text."}
   673→          </div>
   674→        </DialogContent>
   675→      </Dialog>
   676→    </div>
   677→  );
   678→}
