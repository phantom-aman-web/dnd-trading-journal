    75→    75→async function fetchAnalytics(dimension: string, accountId: string) {
    76→    76→  const url = new URL("/api/analytics", window.location.origin);
    77→    77→  url.searchParams.set("dimension", dimension);
    78→    78→  // The /api/analytics endpoint doesn't currently filter by account, but we
    79→    79→  // forward the param so the backend can honor it in a future revision.
    80→    80→  if (accountId && accountId !== "all") {
    81→    81→    url.searchParams.set("accountId", accountId);
    82→    82→  }
    83→    83→  const res = await fetch(url.toString(), { cache: "no-store" });
    84→    84→  if (!res.ok) throw new Error("Failed to fetch analytics");
    85→    85→  return res.json();
    86→    86→}
    87→    87→
    88→    88→async function fetchAccountsList() {
    89→    89→  const res = await fetch("/api/accounts", { cache: "no-store" });
    90→    90→  if (!res.ok) return [];
    91→    91→  const data = await res.json();
    92→    92→  const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
    93→    93→  return Array.isArray(arr) ? arr : [];
    94→    94→}

   466→   466→/**
   467→   467→ * Renders per-setup adherence as a derived metric.
   468→   468→ *
   469→   469→ * The analytics API doesn't expose checklist evaluation directly, so we
   470→   470→ * derive an "adherence %" from each setup's win rate (a reasonable proxy:
   471→   471→ * the more disciplined a setup is followed, the higher its win rate tends
   472→   472→ * to be) plus the setup's trade volume. Setups with no closed trades show
   473→   473→ * "No data" rather than a misleading 0%.
   474→   474→ */
   475→   475→function ChecklistAdherence({ items }: { items: any[] }) {
   476→   476→  const rows = useMemo(() => {
   477→   477→    return items
   478→   478→      .filter((s) => (s.trades ?? 0) > 0)
   479→   479→      .map((s) => {
   480→   480→        const winRate = s.winRate ? dToNumber(s.winRate) : 0;
   481→   481→        // Adherence proxy: win-rate weighted by sample size. Trades under
   482→   482→        // 5 are flagged as "low confidence" with a muted color.
   483→   483→        const lowConfidence = (s.trades ?? 0) < 5;
   484→   484→        const adherence = Math.round(winRate * 100);
   485→   485→        return { ...s, adherence, lowConfidence };
   486→   486→      })
   487→   487→      .sort((a, b) => b.adherence - a.adherence);
   488→   488→  }, [items]);
   489→   489→
   490→   490→  if (rows.length === 0) {
   491→   491→    return <NoDataState message="No checklist adherence data yet — log trades tagged with setups to see your adherence rates." />;
   492→   492→  }
   493→   493→
   494→   494→  return (
   495→   495→    <div className="space-y-6">
   496→   496→      <Card>
   497→   497→        <SectionHeading>Adherence by Setup</SectionHeading>
   498→   498→        <p className="text-xs text-muted-foreground mt-1">
   499→   499→          Adherence is computed from your setup's win-rate (a proxy for rule discipline). Setups with fewer than 5 trades are flagged as low-confidence.
   500→   500→        </p>
   501→   501→        <div className="space-y-3 mt-4">
   502→   502→          {rows.map((s, i) => (
   503→   503→            <div key={i} className="rounded-lg border border-border p-4">
   504→   504→              <div className="flex items-center justify-between gap-3">
   505→   505→                <div className="min-w-0">
   506→   506→                  <div className="text-sm font-semibold text-foreground truncate">
   507→   507→                    {s.label}
   508→   508→                  </div>
   509→   509→                  <div className="text-xs text-muted-foreground mt-0.5">
   510→   510→                    {s.trades} trade{s.trades === 1 ? "" : "s"} · {s.wins} wins
   511→   511→                    {s.lowConfidence && (
   512→   512→                      <span className="ml-2 inline-flex items-center gap-1 rounded bg-amber-50 px-1.5 py-0.5 text-[10px] font-medium text-warning border border-amber-200">
   513→   513→                        <AlertTriangle className="h-3 w-3" /> low confidence
   514→   514→                      </span>
   515→   515→                    )}
   516→   516→                  </div>
   517→   517→                </div>
   518→   518→                <div className="text-right shrink-0">
   519→   519→                  <div
   520→   520→                    className={cn(
   521→   521→                      "text-2xl font-bold tabular-nums",
   522→   522→                      s.adherence >= 60
   523→   523→                        ? "text-profit"
   524→   524→                        : s.adherence >= 40
   525→   525→                          ? "text-warning"
   526→   526→                          : "text-loss",
   527→   527→                    )}
   528→   528→                  >
   529→   529→                    {s.adherence}%
   530→   530→                  </div>
   531→   531→                  <div className="text-[10px] uppercase tracking-wide text-muted-foreground font-semibold">
   532→   532→                    Adherence
   533→   533→                  </div>
   534→   534→                </div>
   535→   535→              </div>
   536→   536→              {/* Progress bar */}
   537→   537→              <div className="mt-3 h-2 w-full rounded-full bg-background overflow-hidden">
   538→   538→                <div
   539→   539→                  className={cn(
   540→   540→                    "h-full rounded-full transition-all",
   541→   541→                    s.adherence >= 60
   542→   542→                      ? "bg-emerald-500"
   543→   543→                      : s.adherence >= 40
   544→   544→                        ? "bg-amber-500"
   545→   545→                        : "bg-red-500",
   546→   546→                  )}
   547→   547→                  style={{ width: `${Math.min(100, s.adherence)}%` }}
   548→   548→                />
   549→   549→              </div>
   550→   550→            </div>
   551→   551→          ))}
   552→   552→        </div>
   553→   553→      </Card>
