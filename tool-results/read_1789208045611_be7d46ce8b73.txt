   105→   704→  /* ---------- File upload (non-blocking, concurrent, with progress) ----------
   106→   705→   *
   107→   706→   * Each selected file is added to `form.uploadedMedia` immediately as a

   110→   709→   * file without blocking the form (the old implementation awaited each
   111→   710→   * upload in a sequential for-loop). The File + XHR for each tempId live in
   112→   711→   * the `fileRefs` ref Map (not state), so Cancel can abort and Retry can

   145→   744→  function uploadFile(file: File, tempId: string) {
   146→   745→    const formData = new FormData();
   147→   746→    formData.append("file", file);
   148→   747→    if (editingId) formData.append("tradeId", editingId);
   149→   748→
   150→   749→    const xhr = new XMLHttpRequest();
   151→   750→    xhr.open("POST", "/api/media");
   152→   751→    fileRefs.current.set(tempId, { file, xhr });
   153→   752→
   154→   753→    // Real upload progress — only XMLHttpRequest exposes this; fetch()
   155→   754→    // ReadableStream request bodies don't have progress events yet.
   156→   755→    xhr.upload.onprogress = (e) => {
   157→   756→      if (e.lengthComputable) {
   158→   757→        const progress = Math.round((e.loaded / e.total) * 100);
   159→   758→        setForm((f: any) => ({
   160→   759→          ...f,
   161→   760→          uploadedMedia: f.uploadedMedia.map((m: any) =>
   162→   761→            m.tempId === tempId ? { ...m, progress } : m,
   163→   762→          ),
   164→   763→        }));
   165→   764→      }
   166→   765→    };
   167→   766→
   168→   767→    xhr.onload = () => {
   169→   768→      if (xhr.status >= 200 && xhr.status < 300) {
   170→   769→        try {
   171→   770→          const data = JSON.parse(xhr.responseText);
   172→   771→          // Fetch the signed URL so the preview can render immediately.
   173→   772→          fetch(`/api/media/${data.id}`)
   174→   773→            .then((r) => r.json())
   175→   774→            .then((detail) => {
   176→   775→              setForm((f: any) => ({
   177→   776→                ...f,
   178→   777→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   179→   778→                  m.tempId === tempId
   180→   779→                    ? {
   181→   780→                        ...m,
   182→   781→                        ...data,
   183→   782→                        id: data.id,
   184→   783→                        url: detail.url,
   185→   784→                        status: "uploaded",
   186→   785→                        progress: 100,
   187→   786→                      }
   188→   787→                    : m,
   189→   788→                ),
   190→   789→              }));
   191→   790→            })
   192→   791→            .catch(() => {
   193→   792→              // Signed-URL fetch failed — still mark uploaded with raw data.
   194→   793→              setForm((f: any) => ({
   195→   794→                ...f,
   196→   795→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   197→   796→                  m.tempId === tempId
   198→   797→                    ? {
   199→   798→                        ...m,
   200→   799→                        ...data,
   201→   800→                        id: data.id,
   202→   801→                        status: "uploaded",
   203→   802→                        progress: 100,
   204→   803→                      }
   205→   804→                    : m,
   206→   805→                ),
   207→   806→              }));
   208→   807→            });
   209→   808→          // Upload finished — drop the File/XHR ref (Retry not needed).
   210→   809→          fileRefs.current.delete(tempId);
   211→   810→          toast.success(`${file.name} uploaded.`);
   212→   811→        } catch {
   213→   812→          markUploadFailed(tempId, file.name);
   214→   813→        }
   215→   814→      } else {
   216→   815→        // Surface the server's error message when present (e.g. "File too
   217→   816→        // large. Max 20 MB.").
   218→   817→        let msg = `Failed to upload ${file.name}`;
   219→   818→        try {
   220→   819→          const err = JSON.parse(xhr.responseText);
   221→   820→          if (err?.error) msg = err.error;
   222→   821→        } catch {}
   223→   822→        markUploadFailed(tempId, file.name, msg);
   224→   823→      }
   225→   824→    };
   226→   825→
   227→   826→    xhr.onerror = () => markUploadFailed(tempId, file.name);
   228→   827→    // onabort is fired when cancelUpload calls xhr.abort(); cancelUpload
   229→   828→    // already removes the item from state and clears the ref, so this is
   230→   829→    // a no-op kept for completeness.
   231→   830→    xhr.onabort = () => {};
   232→   831→
   233→   832→    xhr.send(formData);
   234→   833→  }

   338→   937→  /* ---------- Risk & sizing calculations ---------- */
   339→   938→
   340→   939→  const account = accounts.find((a: any) => a.id === form.accountId);
   341→   940→  const accountCurrency = account?.currency ?? "USD";
   342→   941→  const balanceCents = account?.currentBalanceCents ?? account?.startingBalanceCents ?? 0;
   343→   942→
   344→   943→  const entryNum = Number(form.plannedEntryPrice);
   345→   944→  const stopNum = Number(form.plannedStopPrice);
   346→   945→  const targetNum = Number(form.plannedTargetPrice);
   347→   946→  const lotNum = Number(form.lotSize) || 1;
   348→   947→  const riskPctNum = Number(form.riskPct);
   349→   948→
   350→   949→  const hasEntry = Number.isFinite(entryNum) && form.plannedEntryPrice !== "";
   351→   950→  const hasStop = Number.isFinite(stopNum) && form.plannedStopPrice !== "";
   352→   951→  const hasTarget = Number.isFinite(targetNum) && form.plannedTargetPrice !== "";
   353→   952→
   354→   953→  // Stop distance (always positive).
   355→   954→  const stopDistance = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   356→   955→
   357→   956→  // Risk per unit / reward per unit (per 1 lot).
   358→   957→  const riskPerUnit = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   359→   958→  const rewardPerUnit =
   360→   959→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;
   361→   960→
   362→   961→  // Planned R:R = reward / risk (1.0 = breakeven, 2.0 = 2R for 1R).
   363→   962→  const plannedRR =
   364→   963→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
   365→   964→      ? rewardPerUnit / riskPerUnit
   366→   965→      : null;
   367→   966→
   368→   967→  // Risk amount ($) = balance * riskPct.
   369→   968→  const riskAmountDollars =
   370→   969→    Number.isFinite(riskPctNum) && balanceCents > 0
   371→   970→      ? (balanceCents / 100) * riskPctNum
   372→   971→      : null;
   373→   972→
   374→   973→  // Planned profit ($) = riskAmount * RR (if RR is known).
   375→   974→  const plannedProfitDollars =
   376→   975→    riskAmountDollars != null && plannedRR != null
   377→   976→      ? riskAmountDollars * plannedRR
   378→   977→      : null;
   379→   978→
   380→   979→  /* ---------- Instrument-aware risk preview ---------- */
   381→   980→  // Uses pipSize, contractSize, and pointValueCents from the catalog to
   382→   981→  // express the stop distance in the instrument's native unit (pips /
   383→   982→  // ticks / points) and to derive a per-instrument dollar risk that is
   384→   983→  // consistent with the server-side calculateTradePnl formula:
   385→   984→  //   risk$ = stopDistance × qty × contractSize × (pointValueCents/100)
   386→   985→  // (See computePointValueCents in src/lib/calculations.ts for the
   387→   986→  // convention used to derive pointValueCents from cs × tickSize.)
   388→   987→  const stopDistanceUnits = stopDistance != null
   389→   988→    ? distanceInUnits(stopDistance, selectedInstrument.pipSize)
   390→   989→    : 0;
   391→   990→  const contractSizeNum = Number(selectedInstrument.contractSize) || 1;
   392→   991→  const pointValueCents = selectedInstrument.pointValueCents;
   393→   992→  const instrumentRiskDollars =
   394→   993→    stopDistance != null
   395→   994→      ? stopDistance * lotNum * contractSizeNum * (pointValueCents / 100)
   396→   995→      : null;
   397→   996→
   398→   997→  /* ---------- Partial exits calculations ---------- */
   399→   998→
   400→   999→  const validExits = form.partialExits.filter(
   401→  1000→    (e: any) => e.price !== "" && Number.isFinite(Number(e.price)),
   402→  1001→  );

   404→  1003→    validExits.length > 0
   405→  1004→      ? validExits.reduce((sum: number, e: any) => sum + Number(e.price), 0) /
   406→  1005→        validExits.length
   407→  1006→      : null;
   408→  1007→  const totalRExits =
   409→  1008→    weightedExitPrice != null && riskPerUnit != null && riskPerUnit > 0
   410→  1009→      ? form.direction === "long"
   411→  1010→        ? (weightedExitPrice - entryNum) / riskPerUnit
   412→  1011→        : (entryNum - weightedExitPrice) / riskPerUnit
   413→  1012→      : null;
   414→  1013→
   415→  1014→  /* ---------- Evidence grouping (by chart timeframe) ---------- */
   416→  1015→
   417→  1016→  // Bucket each uploaded media item into a timeframe group so the user can
   418→  1017→  // see at a glance whether their HTF context, execution chart, and entry
   419→  1018→  // trigger are all captured. Items with no timeframe selected land in an
   420→  1019→  // "Unassigned" bucket first to nudge the user toward picking one.
   421→  1020→  const groupedMedia = useMemo(() => {
   422→  1021→    const items: Array<{ m: any; i: number }> = (form.uploadedMedia ?? []).map(
   423→  1022→      (m: any, i: number) => ({ m, i }),
   424→  1023→    );
   425→  1024→    const buckets: Record<string, Array<{ m: any; i: number }>> = {};
   426→  1025→    for (const entry of items) {
   427→  1026→      const g = entry.m.timeframe ? getTimeframeGroup(entry.m.timeframe) : null;
   428→  1027→      const key = g ?? "__unassigned";
   429→  1028→      if (!buckets[key]) buckets[key] = [];
   430→  1029→      buckets[key].push(entry);
   431→  1030→    }
   432→  1031→    const ordered: Array<{ group: string; items: Array<{ m: any; i: number }> }> = [];
   433→  1032→    if (buckets["__unassigned"]) {
   434→  1033→      ordered.push({ group: "Unassigned", items: buckets["__unassigned"] });
   435→  1034→    }
   436→  1035→    for (const g of TIMEFRAME_GROUPS) {
   437→  1036→      if (buckets[g]) ordered.push({ group: g, items: buckets[g] });
   438→  1037→    }
   439→  1038→    // Any leftover custom groups (shouldn't normally happen, but defensive).
   440→  1039→    for (const key of Object.keys(buckets)) {
   441→  1040→      if (key === "__unassigned") continue;
   442→  1041→      if (TIMEFRAME_GROUPS.includes(key)) continue;
   443→  1042→      ordered.push({ group: key, items: buckets[key] });
   444→  1043→    }
   445→  1044→    return ordered;
   446→  1045→  }, [form.uploadedMedia]);
