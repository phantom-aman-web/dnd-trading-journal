   116→async function fetchSettings() {
   117→  const res = await fetch("/api/settings", { cache: "no-store" });
   118→  if (!res.ok) throw new Error("Failed to fetch settings");
   119→  return res.json();
   120→}

   122→async function fetchMeta() {
   123→  const res = await fetch("/api/me", { cache: "no-store" });

   125→  return res.json();
   126→}

  1072→function CompactThreshold({
  1073→  label,
  1074→  value,
  1075→  onChange,
  1076→  tone,
  1077→}: {
  1078→  label: string;
  1079→  value: string;
  1080→  onChange: (v: string) => void;
  1081→  tone: "profit" | "warning" | "loss";
  1082→}) {
  1083→  const toneRing =
  1084→    tone === "profit"
  1085→      ? "focus-visible:ring-emerald-500 focus-visible:border-emerald-500"
  1086→      : tone === "warning"
  1087→        ? "focus-visible:ring-amber-500 focus-visible:border-amber-500"

  1120→function LegalDoc({ doc, label }: { doc: string; label: string }) {
  1121→  const [state, setState] = useState<any>(null);
  1122→  const [open, setOpen] = useState(false);
  1123→
  1124→  useEffect(() => {
  1125→    fetch(`/api/legal/${doc}`).then((r) => r.json()).then(setState);
  1126→  }, [doc]);
  1127→
  1128→  async function accept() {
  1129→    const res = await fetch(`/api/legal/${doc}`, {
  1130→      method: "POST",
  1131→      headers: { "Content-Type": "application/json" },
  1132→      body: JSON.stringify({}),
  1133→    });
  1134→    if (res.ok) {
  1135→      toast.success(`${label} accepted.`);
  1136→      const data = await res.json();
  1137→      setState({
  1138→        ...state,
  1139→        accepted: true,
  1140→        acceptedAt: data.acceptedAt,
  1141→        isCurrent: true,
  1142→        version: data.version,
  1143→      });
  1144→    }
  1145→  }
  1146→
  1147→  const needsAccept = state && (!state.accepted || state.isCurrent === false);
  1148→
  1149→  return (
  1150→    <div className="flex items-center justify-between gap-3 py-2 rounded-lg hover:bg-muted/50 px-2 -mx-2">
  1151→      <div className="min-w-0">
  1152→        <div className="font-medium text-sm text-foreground">{label}</div>
  1153→        <div className="text-xs text-muted-foreground">
  1154→          {state ? (
  1155→            <>
  1156→              Version {state.version}
  1157→              {" · "}
  1158→              {state.accepted
  1159→                ? `Accepted ${new Date(state.acceptedAt).toLocaleDateString()}`
  1160→                : "Not yet accepted"}
  1161→              {state.isCurrent === false && (
  1162→                <span className="text-warning"> · outdated (v{state.acceptedVersion || "?"})</span>
  1163→              )}
  1164→            </>
  1165→          ) : (
  1166→            "Loading…"
  1167→          )}
  1168→        </div>
  1169→      </div>
  1170→      <div className="flex gap-2 shrink-0">
  1171→        <Button
  1172→          size="sm"
  1173→          variant="ghost"
  1174→          onClick={() => setOpen(true)}
  1175→          className="text-muted-foreground hover:text-foreground hover:bg-background"
  1176→        >
  1177→          View
  1178→        </Button>
  1179→        {needsAccept && (
  1180→          <Button
  1181→            size="sm"
  1182→            onClick={accept}
  1183→            className="bg-primary text-primary-foreground hover:bg-primary/90"
  1184→          >
  1185→            Accept
  1186→          </Button>
  1187→        )}
  1188→      </div>
  1189→      <Dialog open={open} onOpenChange={setOpen}>
  1190→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
  1191→          <DialogHeader>
  1192→            <DialogTitle>{label}</DialogTitle>
  1193→            {state && <DialogDescription>Version {state.version}</DialogDescription>}
  1194→          </DialogHeader>
  1195→          <div className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">
  1196→            {state?.content ?? "Document text."}
  1197→          </div>
  1198→        </DialogContent>
  1199→      </Dialog>
  1200→    </div>
