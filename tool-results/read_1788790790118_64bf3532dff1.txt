   102→   102→async function fetchPlans() {
   103→   103→  const res = await fetch("/api/daily-plans", { cache: "no-store" });
   104→   104→  if (!res.ok) throw new Error("Failed to load daily plans");
   105→   105→  return res.json();

   108→   108→function planToForm(plan: DailyPlan): PlanFormState {
   109→   109→  // The date stored in the DB is a UTC ISO string. We extract the YYYY-MM-DD
   110→   110→  // portion to populate the <input type="date"> control without timezone
   111→   111→  // shifting it back/forward a day.
   112→   112→  const dateStr = plan.date ? plan.date.slice(0, 10) : new Date().toISOString().slice(0, 10);
   113→   113→  return {
   114→   114→    date: dateStr,
   115→   115→    weeklyBias: plan.weeklyBias ?? "",
   116→   116→    dailyBias: plan.dailyBias ?? "",
   117→   117→    instruments: plan.instruments ?? "",
   118→   118→    pwh: plan.pwh ?? "",
   119→   119→    pwl: plan.pwl ?? "",
   120→   120→    pdh: plan.pdh ?? "",
   121→   121→    pdl: plan.pdl ?? "",
   122→   122→    htfLevelsJson: plan.htfLevelsJson ?? "",
   123→   123→    liquidityTargets: plan.liquidityTargets ?? "",
   124→   124→    session: plan.session ?? "__none",
   125→   125→    setupConditions: plan.setupConditions ?? "",
   126→   126→    invalidation: plan.invalidation ?? "",
   127→   127→    maxTrades: plan.maxTrades != null ? String(plan.maxTrades) : "",
   128→   128→    maxDailyRiskPct: plan.maxDailyRiskPct ?? "",
   129→   129→    notes: plan.notes ?? "",
   130→   130→  };
   131→   131→}

   160→   160→export function PlansView() {
   161→   161→  const qc = useQueryClient();
   162→   162→  const { data, isLoading } = useQuery({ queryKey: ["daily-plans"], queryFn: fetchPlans });
   163→   163→  const { params } = useNav();
   164→   164→  const [selectedId, setSelectedId] = useState<string | null>(null);
   165→   165→  const [dialogOpen, setDialogOpen] = useState(false);
   166→   166→  const [editingPlan, setEditingPlan] = useState<DailyPlan | null>(null);
   167→   167→
   168→   168→  const plans: DailyPlan[] = data?.items ?? [];
   169→   169→
   170→   170→  // Allow deep-linking into a specific plan: when the calendar's day-detail
   171→   171→  // dialog navigates to "plans" with `{ id: <planId> }`, auto-select that
   172→   172→  // plan in the list.
   173→   173→  useEffect(() => {
   174→   174→    if (params?.id && plans.some((p) => p.id === params.id)) {
   175→   175→      setSelectedId(params.id);
   176→   176→    }
   177→   177→  }, [params?.id, plans]);
   178→   178→
   179→   179→  // Sort newest first. The API already returns desc by date, but be defensive
   180→   180→  // and also group by month for visual scanability.
   181→   181→  const sortedPlans = useMemo(() => {
   182→   182→    return [...plans].sort((a, b) => (a.date < b.date ? 1 : a.date > b.date ? -1 : 0));
   183→   183→  }, [plans]);
   184→   184→
   185→   185→  // Group plans by "Month Year" label for the list display.
   186→   186→  const grouped = useMemo(() => {
   187→   187→    const map = new Map<string, DailyPlan[]>();
   188→   188→    for (const p of sortedPlans) {
   189→   189→      const d = new Date(p.date);
   190→   190→      const label = d.toLocaleDateString("en-US", { month: "long", year: "numeric" });
   191→   191→      const arr = map.get(label) ?? [];
   192→   192→      arr.push(p);
   193→   193→      map.set(label, arr);
   194→   194→    }
   195→   195→    return Array.from(map.entries());
   196→   196→  }, [sortedPlans]);
   197→   197→
   198→   198→  const selected = selectedId ? plans.find((p) => p.id === selectedId) ?? null : null;
   199→   199→
   200→   200→  function openCreate() {
   201→   201→    setEditingPlan(null);
   202→   202→    setDialogOpen(true);
   203→   203→  }
   204→   204→
   205→   205→  function openEdit(plan: DailyPlan) {
   206→   206→    setEditingPlan(plan);
   207→   207→    setDialogOpen(true);
   208→   208→  }
   209→   209→
   210→   210→  function onSaved() {
   211→   211→    qc.invalidateQueries({ queryKey: ["daily-plans"] });
   212→   212→    setDialogOpen(false);
   213→   213→    setEditingPlan(null);
   214→   214→  }
   215→   215→
   216→   216→  return (
   217→   217→    <div className="p-4 md:p-6 grid lg:grid-cols-3 gap-4">
   218→   218→      {/* List */}
   219→   219→      <Card className="p-3 lg:col-span-1">
   220→   220→        <div className="flex items-center justify-between mb-3">
   221→   221→          <div>
   222→   222→            <h1 className="font-semibold">Daily Plans</h1>
   223→   223→            <p className="text-xs text-muted-foreground">Pre-market bias, levels &amp; limits.</p>
   224→   224→          </div>
   225→   225→          <Dialog open={dialogOpen} onOpenChange={(o) => { setDialogOpen(o); if (!o) setEditingPlan(null); }}>
   226→   226→            <DialogTrigger asChild>
   227→   227→              <Button size="sm" onClick={openCreate}>
   228→   228→                <Plus className="h-4 w-4" />
   229→   229→              </Button>
   230→   230→            </DialogTrigger>
   231→   231→            <PlanFormDialog
   232→   232→              key={editingPlan?.id ?? "new"}
   233→   233→              plan={editingPlan}
   234→   234→              onClose={() => { setDialogOpen(false); setEditingPlan(null); }}
   235→   235→              onSaved={onSaved}
   236→   236→            />
   237→   237→          </Dialog>
   238→   238→        </div>
   239→   239→
   240→   240→        {isLoading ? (
   241→   241→          <div className="space-y-2">
   242→   242→            <Skeleton className="h-12" />
   243→   243→            <Skeleton className="h-12" />
   244→   244→            <Skeleton className="h-12" />
   245→   245→          </div>
   246→   246→        ) : sortedPlans.length === 0 ? (
   247→   247→          <EmptyState
   248→   248→            title="No daily plans yet"
   249→   249→            description="Create your first daily plan to record pre-market bias, key levels, session focus and risk limits for each trading day."
   250→   250→            icon={ClipboardList}
   251→   251→            action={{ label: "New Daily Plan", onClick: openCreate }}
   252→   252→          />
   253→   253→        ) : (
   254→   254→          <div className="space-y-4 max-h-[70vh] overflow-y-auto scroll-thin pr-1">
   255→   255→            {grouped.map(([label, items]) => (
   256→   256→              <div key={label} className="space-y-1">
   257→   257→                <div className="text-[11px] uppercase tracking-wide text-muted-foreground px-1 sticky top-0 bg-card py-1">
   258→   258→                  {label}
   259→   259→                </div>
   260→   260→                {items.map((p) => {
   261→   261→                  const d = new Date(p.date);
   262→   262→                  const dateLabel = d.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" });
   263→   263→                  const isBullish = p.dailyBias === "bullish" || p.weeklyBias === "bullish";
   264→   264→                  const isBearish = p.dailyBias === "bearish" || p.weeklyBias === "bearish";
   265→   265→                  return (
   266→   266→                    <button
   267→   267→                      key={p.id}
   268→   268→                      onClick={() => setSelectedId(p.id)}
   269→   269→                      className={cn(
   270→   270→                        "w-full text-left p-2 rounded-md border border-border hover:bg-muted/40 transition-colors",
   271→   271→                        selectedId === p.id && "ring-2 ring-primary border-primary/30",
   272→   272→                      )}
   273→   273→                    >
   274→   274→                      <div className="flex items-center gap-2">
   275→   275→                        <span className="font-medium text-sm">{dateLabel}</span>
   276→   276→                        {p.session && (
   277→   277→                          <Badge variant="outline" className="text-[10px] uppercase">{p.session.replace("_", " ")}</Badge>
   278→   278→                        )}
   279→   279→                        <div className="ml-auto flex items-center gap-1">
   280→   280→                          {isBullish && <Badge variant="outline" className="text-[10px] text-profit border-profit/40">Bull</Badge>}
   281→   281→                          {isBearish && <Badge variant="outline" className="text-[10px] text-loss border-loss/40">Bear</Badge>}
   282→   282→                        </div>
   283→   283→                      </div>
   284→   284→                      <div className="text-xs text-muted-foreground mt-1 line-clamp-1">
   285→   285→                        {p.instruments || "No instruments"} · {p.setupConditions || "No setup conditions"}
   286→   286→                      </div>
   287→   287→                    </button>
   288→   288→                  );
   289→   289→                })}
   290→   290→              </div>
   291→   291→            ))}
   292→   292→          </div>
   293→   293→        )}
   294→   294→      </Card>
   295→   295→
   296→   296→      {/* Detail */}
   297→   297→      <div className="lg:col-span-2 space-y-4">
   298→   298→        {!selected ? (
   299→   299→          <Card className="p-8 text-center">
   300→   300→            <ClipboardList className="h-8 w-8 mx-auto text-muted-foreground mb-3" />
   301→   301→            <h3 className="font-medium">Select a daily plan</h3>
   302→   302→            <p className="text-sm text-muted-foreground mt-1">
   303→   303→              Daily plans capture your pre-market work: bias, key HTF levels, liquidity targets, session focus, and the rules that invalidate the thesis.
   304→   304→            </p>
   305→   305→          </Card>
   306→   306→        ) : (
   307→   307→          <PlanDetail
   308→   308→            plan={selected}
   309→   309→            onEdit={() => openEdit(selected)}
   310→   310→            onDeleted={() => { setSelectedId(null); qc.invalidateQueries({ queryKey: ["daily-plans"] }); }}
   311→   311→          />
   312→   312→        )}
   313→   313→      </div>
   314→   314→    </div>
   315→   315→  );
   316→   316→}

   322→   322→function PlanDetail({ plan, onEdit, onDeleted }: { plan: DailyPlan; onEdit: () => void; onDeleted: () => void }) {
   323→   323→  const [deleting, setDeleting] = useState(false);
   324→   324→  const d = new Date(plan.date);
   325→   325→  const dateLabel = d.toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric", year: "numeric" });
   326→   326→
   327→   327→  async function handleDelete() {
   328→   328→    if (!confirm("Delete this daily plan? Trades linked to it will be unlinked but kept.")) return;
   329→   329→    setDeleting(true);
   330→   330→    try {
   331→   331→      const res = await fetch(`/api/daily-plans/${plan.id}`, { method: "DELETE" });
   332→   332→      if (res.ok) {
   333→   333→        toast.success("Daily plan deleted.");
   334→   334→        onDeleted();
   335→   335→      } else {
   336→   336→        const d = await res.json().catch(() => null);
   337→   337→        toast.error(d?.error ?? "Failed to delete plan.");
   338→   338→      }
   339→   339→    } finally {
   340→   340→      setDeleting(false);
   341→   341→    }
   342→   342→  }
   343→   343→
   344→   344→  return (
   345→   345→    <Card className="p-4 md:p-6 space-y-4">
   346→   346→      <div className="flex items-start gap-3">
   347→   347→        <div className="flex-1 min-w-0">
   348→   348→          <div className="text-xs uppercase tracking-wide text-muted-foreground">Daily Plan</div>
   349→   349→          <h2 className="text-xl font-semibold tracking-tight">{dateLabel}</h2>
   350→   350→          <div className="flex flex-wrap items-center gap-2 mt-2">
   351→   351→            {plan.weeklyBias && <BiasBadge label="Weekly" value={plan.weeklyBias} />}
   352→   352→            {plan.dailyBias && <BiasBadge label="Daily" value={plan.dailyBias} />}
   353→   353→            {plan.session && (
   354→   354→              <Badge variant="outline" className="uppercase text-xs">
   355→   355→                {plan.session.replace("_", " ")}
   356→   356→              </Badge>
   357→   357→            )}
   358→   358→            {plan.maxTrades != null && (
   359→   359→              <Badge variant="outline" className="text-xs">Max {plan.maxTrades} trades</Badge>
   360→   360→            )}
   361→   361→            {plan.maxDailyRiskPct && (
   362→   362→              <Badge variant="outline" className="text-xs">Risk {(Number(plan.maxDailyRiskPct) * 100).toFixed(1)}%</Badge>
   363→   363→            )}
   364→   364→          </div>
   365→   365→        </div>
   366→   366→        <div className="flex items-center gap-1">
   367→   367→          <Button size="sm" variant="outline" onClick={onEdit}>
   368→   368→            <Pencil className="h-3.5 w-3.5" /> Edit
   369→   369→          </Button>
   370→   370→          <Button
   371→   371→            size="icon"
   372→   372→            variant="ghost"
   373→   373→            className="h-8 w-8 text-muted-foreground hover:text-loss"
   374→   374→            onClick={handleDelete}
   375→   375→            disabled={deleting}
   376→   376→            aria-label="Delete plan"
   377→   377→          >
   378→   378→            {deleting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Trash2 className="h-3.5 w-3.5" />}
   379→   379→          </Button>
   380→   380→        </div>
   381→   381→      </div>
   382→   382→
   383→   383→      <Separator />
   384→   384→
   385→   385→      <div className="grid sm:grid-cols-2 gap-4">
   386→   386→        <Field label="Instruments" value={plan.instruments} />
   387→   387→        <Field label="Session Focus" value={plan.session ? plan.session.replace("_", " ").toUpperCase() : null} />
   388→   388→      </div>
   389→   389→
   390→   390→      <div>
   391→   391→        <h3 className="text-sm font-medium mb-2">Key Levels</h3>
   392→   392→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   393→   393→          <LevelField label="PWH" value={plan.pwh} />
   394→   394→          <LevelField label="PWL" value={plan.pwl} />
   395→   395→          <LevelField label="PDH" value={plan.pdh} />
   396→   396→          <LevelField label="PDL" value={plan.pdl} />
   397→   397→        </div>
   398→   398→      </div>
   399→   399→
   400→   400→      {plan.liquidityTargets && (
   401→   401→        <div>
   402→   402→          <h3 className="text-sm font-medium mb-1">Liquidity Targets</h3>
   403→   403→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.liquidityTargets}</p>
   404→   404→        </div>
   405→   405→      )}
   406→   406→
   407→   407→      {plan.htfLevelsJson && (
   408→   408→        <div>
   409→   409→          <h3 className="text-sm font-medium mb-1">HTF Levels</h3>
   410→   410→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.htfLevelsJson}</p>
   411→   411→        </div>
   412→   412→      )}
   413→   413→
   414→   414→      {plan.setupConditions && (
   415→   415→        <div>
   416→   416→          <h3 className="text-sm font-medium mb-1">Setup Conditions</h3>
   417→   417→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.setupConditions}</p>
   418→   418→        </div>
   419→   419→      )}
   420→   420→
   421→   421→      {plan.invalidation && (
   422→   422→        <div>
   423→   423→          <h3 className="text-sm font-medium mb-1">Invalidation</h3>
   424→   424→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.invalidation}</p>
   425→   425→        </div>
   426→   426→      )}
   427→   427→
   428→   428→      {plan.notes && (
   429→   429→        <div>
   430→   430→          <h3 className="text-sm font-medium mb-1">Notes</h3>
   431→   431→          <p className="text-sm text-muted-foreground whitespace-pre-wrap">{plan.notes}</p>
   432→   432→        </div>
   433→   433→      )}
   434→   434→
   435→   435→      <div className="text-xs text-muted-foreground pt-2 border-t border-border">
   436→   436→        Created {new Date(plan.createdAt).toLocaleString()} · Updated {new Date(plan.updatedAt).toLocaleString()}
   437→   437→      </div>
   438→   438→    </Card>
   439→   439→  );
   440→   440→}

   442→   442→function BiasBadge({ label, value }: { label: string; value: string }) {
   443→   443→  const v = value.toLowerCase();
   444→   444→  const tone = v === "bullish" ? "text-profit border-profit/40" : v === "bearish" ? "text-loss border-loss/40" : "";
   445→   445→  return (
   446→   446→    <Badge variant="outline" className={cn("text-xs", tone)}>
   447→   447→      {label}: <span className="capitalize ml-1">{value}</span>
   448→   448→    </Badge>
   449→   449→  );
