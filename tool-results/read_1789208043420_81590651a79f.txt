   744→  function uploadFile(file: File, tempId: string) {
   745→    const formData = new FormData();
   746→    formData.append("file", file);
   747→    if (editingId) formData.append("tradeId", editingId);
   748→
   749→    const xhr = new XMLHttpRequest();
   750→    xhr.open("POST", "/api/media");
   751→    fileRefs.current.set(tempId, { file, xhr });
   752→
   753→    // Real upload progress — only XMLHttpRequest exposes this; fetch()
   754→    // ReadableStream request bodies don't have progress events yet.
   755→    xhr.upload.onprogress = (e) => {
   756→      if (e.lengthComputable) {
   757→        const progress = Math.round((e.loaded / e.total) * 100);
   758→        setForm((f: any) => ({
   759→          ...f,
   760→          uploadedMedia: f.uploadedMedia.map((m: any) =>
   761→            m.tempId === tempId ? { ...m, progress } : m,
   762→          ),
   763→        }));
   764→      }
   765→    };
   766→
   767→    xhr.onload = () => {
   768→      if (xhr.status >= 200 && xhr.status < 300) {
   769→        try {
   770→          const data = JSON.parse(xhr.responseText);
   771→          // Fetch the signed URL so the preview can render immediately.
   772→          fetch(`/api/media/${data.id}`)
   773→            .then((r) => r.json())
   774→            .then((detail) => {
   775→              setForm((f: any) => ({
   776→                ...f,
   777→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   778→                  m.tempId === tempId
   779→                    ? {
   780→                        ...m,
   781→                        ...data,
   782→                        id: data.id,
   783→                        url: detail.url,
   784→                        status: "uploaded",
   785→                        progress: 100,
   786→                      }
   787→                    : m,
   788→                ),
   789→              }));
   790→            })
   791→            .catch(() => {
   792→              // Signed-URL fetch failed — still mark uploaded with raw data.
   793→              setForm((f: any) => ({
   794→                ...f,
   795→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   796→                  m.tempId === tempId
   797→                    ? {
   798→                        ...m,
   799→                        ...data,
   800→                        id: data.id,
   801→                        status: "uploaded",
   802→                        progress: 100,
   803→                      }
   804→                    : m,
   805→                ),
   806→              }));
   807→            });
   808→          // Upload finished — drop the File/XHR ref (Retry not needed).
   809→          fileRefs.current.delete(tempId);
   810→          toast.success(`${file.name} uploaded.`);
   811→        } catch {
   812→          markUploadFailed(tempId, file.name);
   813→        }
   814→      } else {
   815→        // Surface the server's error message when present (e.g. "File too
   816→        // large. Max 20 MB.").
   817→        let msg = `Failed to upload ${file.name}`;
   818→        try {
   819→          const err = JSON.parse(xhr.responseText);
   820→          if (err?.error) msg = err.error;
   821→        } catch {}
   822→        markUploadFailed(tempId, file.name, msg);
   823→      }
   824→    };
   825→
   826→    xhr.onerror = () => markUploadFailed(tempId, file.name);
   827→    // onabort is fired when cancelUpload calls xhr.abort(); cancelUpload
   828→    // already removes the item from state and clears the ref, so this is
   829→    // a no-op kept for completeness.
   830→    xhr.onabort = () => {};
   831→
   832→    xhr.send(formData);
   833→  }

   937→  /* ---------- Risk & sizing calculations ---------- */
   938→
   939→  const account = accounts.find((a: any) => a.id === form.accountId);
   940→  const accountCurrency = account?.currency ?? "USD";
   941→  const balanceCents = account?.currentBalanceCents ?? account?.startingBalanceCents ?? 0;
   942→
   943→  const entryNum = Number(form.plannedEntryPrice);
   944→  const stopNum = Number(form.plannedStopPrice);
   945→  const targetNum = Number(form.plannedTargetPrice);
   946→  const lotNum = Number(form.lotSize) || 1;
   947→  const riskPctNum = Number(form.riskPct);
   948→
   949→  const hasEntry = Number.isFinite(entryNum) && form.plannedEntryPrice !== "";
   950→  const hasStop = Number.isFinite(stopNum) && form.plannedStopPrice !== "";
   951→  const hasTarget = Number.isFinite(targetNum) && form.plannedTargetPrice !== "";
   952→
   953→  // Stop distance (always positive).
   954→  const stopDistance = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   955→
   956→  // Risk per unit / reward per unit (per 1 lot).
   957→  const riskPerUnit = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   958→  const rewardPerUnit =
   959→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;
   960→
   961→  // Planned R:R = reward / risk (1.0 = breakeven, 2.0 = 2R for 1R).
   962→  const plannedRR =
   963→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
   964→      ? rewardPerUnit / riskPerUnit
   965→      : null;
   966→
   967→  // Risk amount ($) = balance * riskPct.
   968→  const riskAmountDollars =
   969→    Number.isFinite(riskPctNum) && balanceCents > 0
   970→      ? (balanceCents / 100) * riskPctNum
   971→      : null;
   972→
   973→  // Planned profit ($) = riskAmount * RR (if RR is known).
   974→  const plannedProfitDollars =
   975→    riskAmountDollars != null && plannedRR != null
   976→      ? riskAmountDollars * plannedRR
   977→      : null;
   978→
   979→  /* ---------- Instrument-aware risk preview ---------- */
   980→  // Uses pipSize, contractSize, and pointValueCents from the catalog to
   981→  // express the stop distance in the instrument's native unit (pips /
   982→  // ticks / points) and to derive a per-instrument dollar risk that is
   983→  // consistent with the server-side calculateTradePnl formula:
   984→  //   risk$ = stopDistance × qty × contractSize × (pointValueCents/100)
   985→  // (See computePointValueCents in src/lib/calculations.ts for the
   986→  // convention used to derive pointValueCents from cs × tickSize.)
   987→  const stopDistanceUnits = stopDistance != null
   988→    ? distanceInUnits(stopDistance, selectedInstrument.pipSize)
   989→    : 0;
   990→  const contractSizeNum = Number(selectedInstrument.contractSize) || 1;
   991→  const pointValueCents = selectedInstrument.pointValueCents;
   992→  const instrumentRiskDollars =
   993→    stopDistance != null
   994→      ? stopDistance * lotNum * contractSizeNum * (pointValueCents / 100)
   995→      : null;
   996→
   997→  /* ---------- Partial exits calculations ---------- */
   998→
   999→  const validExits = form.partialExits.filter(
  1000→    (e: any) => e.price !== "" && Number.isFinite(Number(e.price)),
  1001→  );
  1002→  const weightedExitPrice =
  1003→    validExits.length > 0
  1004→      ? validExits.reduce((sum: number, e: any) => sum + Number(e.price), 0) /
  1005→        validExits.length
  1006→      : null;
  1007→  const totalRExits =
  1008→    weightedExitPrice != null && riskPerUnit != null && riskPerUnit > 0
  1009→      ? form.direction === "long"
  1010→        ? (weightedExitPrice - entryNum) / riskPerUnit
  1011→        : (entryNum - weightedExitPrice) / riskPerUnit
  1012→      : null;
  1013→
  1014→  /* ---------- Evidence grouping (by chart timeframe) ---------- */
  1015→
  1016→  // Bucket each uploaded media item into a timeframe group so the user can
  1017→  // see at a glance whether their HTF context, execution chart, and entry
  1018→  // trigger are all captured. Items with no timeframe selected land in an
  1019→  // "Unassigned" bucket first to nudge the user toward picking one.
  1020→  const groupedMedia = useMemo(() => {
  1021→    const items: Array<{ m: any; i: number }> = (form.uploadedMedia ?? []).map(
  1022→      (m: any, i: number) => ({ m, i }),
  1023→    );
  1024→    const buckets: Record<string, Array<{ m: any; i: number }>> = {};
  1025→    for (const entry of items) {
  1026→      const g = entry.m.timeframe ? getTimeframeGroup(entry.m.timeframe) : null;
  1027→      const key = g ?? "__unassigned";
  1028→      if (!buckets[key]) buckets[key] = [];
  1029→      buckets[key].push(entry);
  1030→    }
  1031→    const ordered: Array<{ group: string; items: Array<{ m: any; i: number }> }> = [];
  1032→    if (buckets["__unassigned"]) {
  1033→      ordered.push({ group: "Unassigned", items: buckets["__unassigned"] });
  1034→    }
  1035→    for (const g of TIMEFRAME_GROUPS) {
  1036→      if (buckets[g]) ordered.push({ group: g, items: buckets[g] });
  1037→    }
  1038→    // Any leftover custom groups (shouldn't normally happen, but defensive).
  1039→    for (const key of Object.keys(buckets)) {
  1040→      if (key === "__unassigned") continue;
  1041→      if (TIMEFRAME_GROUPS.includes(key)) continue;
  1042→      ordered.push({ group: key, items: buckets[key] });
  1043→    }
  1044→    return ordered;
