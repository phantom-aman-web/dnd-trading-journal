   150→/** Fetch the user's accounts as a plain array (defensive). Returns [] on
   151→ *  any error or unexpected shape. */
   152→async function fetchAccounts(accountId?: string) {
   153→  try {
   154→    const qs = new URLSearchParams({ ...(accountId ? { accountId } : {}) }).toString();
   155→    const url = `/api/accounts${qs ? `?${qs}` : ""}`;
   156→    const res = await fetch(url, { cache: "no-store" });
   157→    if (!res.ok) return [];
   158→    const data = await res.json();
   159→    const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
   160→    return Array.isArray(arr) ? arr : [];
   161→  } catch {
   162→    return [];
   163→  }
   164→}
   165→
   166→/** Fetch the user's strategies as a plain array (defensive). Returns [] on
   167→ *  any error or unexpected shape. */
   168→async function fetchStrategies() {
   169→  try {
   170→    const res = await fetch("/api/strategies", { cache: "no-store" });
   171→    if (!res.ok) return [];

   180→/** Extract a flat `ChecklistItem[]` from a strategy version's rulesJson.
   181→ *  Supports both:
   182→ *    - flat array:   [{ title, required, description, weight }, ...]
   183→ *                    [{ text,   required, weight }, ...]
   184→ *                    ["plain string rule", ...]
   185→ *    - legacy object: { entry: [...], stop: [...], target: [...],
   186→ *                       management: [...], invalidation: [...] }
   187→ */
   188→function extractChecklist(rulesJson: string | null | undefined): ChecklistItem[] {
   189→  if (!rulesJson) return [];
   190→  let parsed: unknown;
   191→  try {
   192→    parsed = JSON.parse(rulesJson);
   193→  } catch {
   194→    return [];
   195→  }
   196→  const items: ChecklistItem[] = [];
   197→  let idx = 0;
   198→
   199→  const pushRule = (rule: unknown) => {
   200→    if (typeof rule === "string") {
   201→      const text = rule.trim();
   202→      if (!text) return;
   203→      items.push({
   204→        id: `rule_${idx}`,
   205→        text,
   206→        required: true,
   207→        weight: 1,
   208→        evidenceRequired: false,
   209→      });
   210→      idx++;
   211→      return;
   212→    }
   213→    if (rule && typeof rule === "object") {
   214→      const r = rule as Record<string, unknown>;
   215→      const text = String(r.text ?? r.title ?? "").trim();
   216→      if (!text) return;
   217→      const weightNum = typeof r.weight === "number" && Number.isFinite(r.weight) ? r.weight : 1;
   218→      items.push({
   219→        id: `rule_${idx}`,
   220→        text,
   221→        required: r.required !== false,
   222→        weight: Math.max(0, weightNum),
   223→        evidenceRequired: r.evidenceRequired === true,
   224→      });
   225→      idx++;
   226→    }
   227→  };
   228→
   229→  if (Array.isArray(parsed)) {
   230→    parsed.forEach(pushRule);
   231→    return items;
   232→  }
   233→  if (parsed && typeof parsed === "object") {
   234→    const obj = parsed as Record<string, unknown>;
   235→    // New playbooks-view format: { items: [...] }
   236→    if (Array.isArray(obj.items)) {
   237→      obj.items.forEach(pushRule);
   238→      return items;
   239→    }
   240→    // Legacy category object: { entry, stop, target, management, invalidation }
   241→    for (const section of ["entry", "stop", "target", "management", "invalidation"]) {
   242→      const arr = obj[section];
   243→      if (Array.isArray(arr)) arr.forEach(pushRule);
   244→    }
   245→    return items;
   246→  }
   247→  return [];
   248→}

   250→/** Parse a JSON string into the typed fallback; never throws. */
   251→function safeJsonParse(json: string | null | undefined, fallback: any): any {
   252→  if (!json) return fallback;
   253→  try { return JSON.parse(json); } catch { return fallback; }
   254→}
   255→
   256→/** Format a numeric price with the given decimal precision (default 5,
   257→ *  matching the historical forex-major behaviour). Returns "—" when the
   258→ *  value is missing or non-finite. Pass the selected instrument's
   259→ *  `pricePrecision` to render prices with the right number of decimals
   260→ *  (e.g. 2 for gold, 1 for indices, 0 for YM). */
   261→function fmtPrice(
   262→  value: number | string | null | undefined,
   263→  precision: number = 5,
   264→): string {
   265→  if (value == null || value === "") return "—";
   266→  const n = typeof value === "string" ? Number(value) : value;
   267→  if (!Number.isFinite(n)) return "—";
   268→  const p =
   269→    Number.isFinite(precision) && precision >= 0 && precision <= 8
   270→      ? Math.round(precision)
   271→      : 5;
   272→  return n.toFixed(p);
   273→}
   274→
   275→/** Format a money amount given a dollars-number (NOT cents). */
   276→function fmtMoney(dollars: number | null | undefined, currency = "USD"): string {
   277→  if (dollars == null || !Number.isFinite(dollars)) return "—";
   278→  try {
   279→    return new Intl.NumberFormat("en-US", {
   280→      style: "currency",
   281→      currency,
   282→      minimumFractionDigits: 2,
   283→      maximumFractionDigits: 2,
   284→    }).format(dollars);
   285→  } catch {
   286→    return `$${dollars.toFixed(2)}`;
   287→  }
   288→}
   289→
   290→/** Generate a short random id for client-side list items. */
   291→function uid(): string {
   292→  return Math.random().toString(36).slice(2, 10);
   293→}

   830→  const riskPerUnit = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   831→  const rewardPerUnit =
   832→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;

   835→  const plannedRR =
   836→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
   837→      ? rewardPerUnit / riskPerUnit
   838→      : null;

   840→  // Risk amount ($) = balance * riskPct.
   841→  const riskAmountDollars =
   842→    Number.isFinite(riskPctNum) && balanceCents > 0
   843→      ? (balanceCents / 100) * riskPctNum
   844→      : null;
   845→
   846→  // Planned profit ($) = riskAmount * RR (if RR is known).
   847→  const plannedProfitDollars =
   848→    riskAmountDollars != null && plannedRR != null
   849→      ? riskAmountDollars * plannedRR
   850→      : null;

   852→  /* ---------- Instrument-aware risk preview ---------- */
   853→  // Uses pipSize, contractSize, and pointValueCents from the catalog to
   854→  // express the stop distance in the instrument's native unit (pips /
   855→  // ticks / points) and to derive a per-instrument dollar risk that is
   856→  // consistent with the server-side calculateTradePnl formula:
   857→  //   risk$ = stopDistance × qty × contractSize × (pointValueCents/100)
   858→  // (See computePointValueCents in src/lib/calculations.ts for the
   859→  // convention used to derive pointValueCents from cs × tickSize.)
   860→  const stopDistanceUnits = stopDistance != null
   861→    ? distanceInUnits(stopDistance, selectedInstrument.pipSize)
   862→    : 0;
   863→  const contractSizeNum = Number(selectedInstrument.contractSize) || 1;
   864→  const pointValueCents = selectedInstrument.pointValueCents;
   865→  const instrumentRiskDollars =
   866→    stopDistance != null

   876→    validExits.length > 0

   878→        validExits.length

   881→    weightedExitPrice != null && riskPerUnit != null && riskPerUnit > 0

   900→      const key = g ?? "__unassigned";
   901→      if (!buckets[key]) buckets[key] = [];
   902→      buckets[key].push(entry);
   903→    }
   904→    const ordered: Array<{ group: string; items: Array<{ m: any; i: number }> }> = [];
