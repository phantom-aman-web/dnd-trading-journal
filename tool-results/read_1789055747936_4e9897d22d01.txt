    50→   749→   *
    51→   750→   * Each selected file is added to `form.uploadedMedia` immediately as a

    89→   788→  function uploadFile(file: File, tempId: string) {
    90→   789→    const formData = new FormData();
    91→   790→    formData.append("file", file);
    92→   791→    if (editingId) formData.append("tradeId", editingId);
    93→   792→
    94→   793→    const xhr = new XMLHttpRequest();
    95→   794→    xhr.open("POST", "/api/media");
    96→   795→    fileRefs.current.set(tempId, { file, xhr });
    97→   796→
    98→   797→    // Real upload progress — only XMLHttpRequest exposes this; fetch()
    99→   798→    // ReadableStream request bodies don't have progress events yet.
   100→   799→    xhr.upload.onprogress = (e) => {
   101→   800→      if (e.lengthComputable) {
   102→   801→        const progress = Math.round((e.loaded / e.total) * 100);
   103→   802→        setForm((f: any) => ({
   104→   803→          ...f,
   105→   804→          uploadedMedia: f.uploadedMedia.map((m: any) =>
   106→   805→            m.tempId === tempId ? { ...m, progress } : m,
   107→   806→          ),
   108→   807→        }));
   109→   808→      }
   110→   809→    };
   111→   810→
   112→   811→    xhr.onload = () => {
   113→   812→      if (xhr.status >= 200 && xhr.status < 300) {
   114→   813→        try {
   115→   814→          const data = JSON.parse(xhr.responseText);
   116→   815→          // Fetch the signed URL so the preview can render immediately.
   117→   816→          fetch(`/api/media/${data.id}`)
   118→   817→            .then((r) => r.json())
   119→   818→            .then((detail) => {
   120→   819→              setForm((f: any) => ({
   121→   820→                ...f,
   122→   821→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   123→   822→                  m.tempId === tempId
   124→   823→                    ? {
   125→   824→                        ...m,
   126→   825→                        ...data,
   127→   826→                        id: data.id,
   128→   827→                        url: detail.url,
   129→   828→                        status: "uploaded",
   130→   829→                        progress: 100,
   131→   830→                      }
   132→   831→                    : m,
   133→   832→                ),
   134→   833→              }));
   135→   834→            })
   136→   835→            .catch(() => {
   137→   836→              // Signed-URL fetch failed — still mark uploaded with raw data.
   138→   837→              setForm((f: any) => ({
   139→   838→                ...f,
   140→   839→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   141→   840→                  m.tempId === tempId
   142→   841→                    ? {
   143→   842→                        ...m,
   144→   843→                        ...data,
   145→   844→                        id: data.id,
   146→   845→                        status: "uploaded",
   147→   846→                        progress: 100,
   148→   847→                      }
   149→   848→                    : m,
   150→   849→                ),
   151→   850→              }));
   152→   851→            });
   153→   852→          // Upload finished — drop the File/XHR ref (Retry not needed).
   154→   853→          fileRefs.current.delete(tempId);
   155→   854→          toast.success(`${file.name} uploaded.`);
   156→   855→        } catch {
   157→   856→          markUploadFailed(tempId, file.name);
   158→   857→        }
   159→   858→      } else {
   160→   859→        // Surface the server's error message when present (e.g. "File too
   161→   860→        // large. Max 20 MB.").
   162→   861→        let msg = `Failed to upload ${file.name}`;
   163→   862→        try {
   164→   863→          const err = JSON.parse(xhr.responseText);
   165→   864→          if (err?.error) msg = err.error;
   166→   865→        } catch {}
   167→   866→        markUploadFailed(tempId, file.name, msg);
   168→   867→      }
   169→   868→    };
   170→   869→
   171→   870→    xhr.onerror = () => markUploadFailed(tempId, file.name);
   172→   871→    // onabort is fired when cancelUpload calls xhr.abort(); cancelUpload
   173→   872→    // already removes the item from state and clears the ref, so this is
   174→   873→    // a no-op kept for completeness.
   175→   874→    xhr.onabort = () => {};
   176→   875→
   177→   876→    xhr.send(formData);
   178→   877→  }

   295→   994→  const hasStop = Number.isFinite(stopNum) && form.plannedStopPrice !== "";
   296→   995→  const hasTarget = Number.isFinite(targetNum) && form.plannedTargetPrice !== "";
   297→   996→

   300→   999→

   303→  1002→  const rewardPerUnit =
   304→  1003→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;
   305→  1004→

   307→  1006→  const plannedRR =
   308→  1007→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
   309→  1008→      ? rewardPerUnit / riskPerUnit

   313→  1012→  const riskAmountDollars =
   314→  1013→    Number.isFinite(riskPctNum) && balanceCents > 0
   315→  1014→      ? (balanceCents / 100) * riskPctNum
   316→  1015→      : null;

   318→  1017→  // Planned profit ($) = riskAmount * RR (if RR is known).
   319→  1018→  const plannedProfitDollars =
   320→  1019→    riskAmountDollars != null && plannedRR != null
   321→  1020→      ? riskAmountDollars * plannedRR
