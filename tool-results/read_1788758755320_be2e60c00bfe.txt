   371→   371→  function setField(path: string, value: any) {
   372→   372→    setForm((f: any) => {
   373→   373→      const next = { ...f };
   374→   374→      const parts = path.split(".");
   375→   375→      let cur: any = next;
   376→   376→      for (let i = 0; i < parts.length - 1; i++) {
   377→   377→        cur[parts[i]] = { ...cur[parts[i]] };
   378→   378→        cur = cur[parts[i]];
   379→   379→      }
   380→   380→      cur[parts[parts.length - 1]] = value;
   381→   381→      return next;
   382→   382→    });
   383→   383→  }
   384→   384→
   385→   385→  function toggleArrayField(path: string, value: string) {
   386→   386→    setForm((f: any) => {
   387→   387→      const next = { ...f };
   388→   388→      const parts = path.split(".");
   389→   389→      let cur: any = next;
   390→   390→      for (let i = 0; i < parts.length - 1; i++) {
   391→   391→        cur[parts[i]] = { ...cur[parts[i]] };
   392→   392→        cur = cur[parts[i]];
   393→   393→      }
   394→   394→      const key = parts[parts.length - 1];
   395→   395→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   396→   396→      const idx = arr.indexOf(value);
   397→   397→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   398→   398→      cur[key] = arr;
   399→   399→      return next;
   400→   400→    });
   401→   401→  }

   403→   403→  // Update one A+ checklist answer field (`checked` or `note`) for the given

   405→   405→  function updateChecklistAnswer(
   406→   406→    itemId: string,
   407→   407→    field: "checked" | "note",
   408→   408→    value: boolean | string,
   409→   409→  ) {
   410→   410→    setForm((f: any) => {
   411→   411→      const current = f.checklistAnswers?.[itemId] ?? { checked: false };
   412→   412→      const next = {
   413→   413→        ...f.checklistAnswers,
   414→   414→        [itemId]: {
   415→   415→          ...current,
   416→   416→          [field]: value,
   417→   417→        },
   418→   418→      };
   419→   419→      // If a checked item is being unchecked, drop its note so storage stays clean.
   420→   420→      if (field === "checked" && value === false && next[itemId].note) {
   421→   421→        delete next[itemId].note;
   422→   422→      }
   423→   423→      return { ...f, checklistAnswers: next };
   424→   424→    });
