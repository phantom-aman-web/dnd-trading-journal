   265→function RulesView({ rulesJson }: { rulesJson: string }) {
   266→  let rules: any = {};
   267→  try { rules = JSON.parse(rulesJson ?? "{}"); } catch {}
   268→  const sections = ["entry", "stop", "target", "management", "invalidation"];
   269→  const thresholds = rules.gradingThresholds;
   270→  return (
   271→    <div className="space-y-4">
   272→      {thresholds && (
   273→        <div className="flex flex-wrap gap-2">
   274→          {["A+", "A", "B", "C"].map((g) => (
   275→            <Badge key={g} variant="outline" className="text-xs">
   276→              {g} ≥ {Math.round((thresholds[g] ?? 0) * 100)}%
   277→            </Badge>
   278→          ))}
   279→        </div>
   280→      )}
   281→      <div className="grid sm:grid-cols-2 gap-3 text-sm">
   282→        {sections.map((sec) => {
   283→          const sectionRules = Array.isArray(rules[sec]) ? rules[sec] : [];
   284→          if (sectionRules.length === 0) return null;
   285→          return (
   286→            <div key={sec} className="rounded-md border border-border p-3">
   287→              <div className="text-xs uppercase tracking-wide text-muted-foreground mb-1">{sec}</div>
   288→              <ul className="space-y-1">
   289→                {sectionRules.map((r: any, i: number) => {
   290→                  const text = typeof r === "string" ? r : r.text;
   291→                  const weight = typeof r === "object" ? r.weight : 1;
   292→                  const required = typeof r === "object" ? r.required : true;
   293→                  return (
   294→                    <li key={i} className="text-sm flex items-center gap-2">
   295→                      <span>{text}</span>
   296→                      {required && <Badge variant="outline" className="text-[10px] text-loss">Required</Badge>}
   297→                      <span className="text-[10px] text-muted-foreground ml-auto">×{weight}</span>
   298→                    </li>
   299→                  );
   300→                })}
   301→              </ul>
   302→            </div>
   303→          );
   304→        })}
   305→      </div>
   306→    </div>
   307→  );
   308→}

   310→function NewStrategyDialog({ onClose, onCreated }: { onClose: () => void; onCreated: () => void }) {
   311→  const [name, setName] = useState("");

   313→  const [purpose, setPurpose] = useState("");
   314→  const [instruments, setInstruments] = useState("");
   315→  const [market, setMarket] = useState("forex");
   316→  const [timeframe, setTimeframe] = useState("5m-1h");
   317→  const [session, setSession] = useState("london");
   318→  const [checklistConfigId, setChecklistConfigId] = useState("");
   319→  // Rules: each section is a newline-separated list
   320→  const [entryRules, setEntryRules] = useState("");
   321→  const [stopRules, setStopRules] = useState("");
   322→  const [targetRules, setTargetRules] = useState("");
   323→  const [managementRules, setManagementRules] = useState("");
   324→  const [invalidationRules, setInvalidationRules] = useState("");
   325→  // Weight + required configuration
   326→  const [ruleConfig, setRuleConfig] = useState("");
   327→  // Grading thresholds
   328→  const [aplusThreshold, setAplusThreshold] = useState("0.9");
   329→  const [aThreshold, setAThreshold] = useState("0.75");
   330→  const [bThreshold, setBThreshold] = useState("0.6");
   331→  const [cThreshold, setCThreshold] = useState("0.4");
   332→  const [loading, setLoading] = useState(false);
   333→
   334→  // Fetch checklists for the association selector
   335→  const { data: checklistsData } = useQuery({
   336→    queryKey: ["checklists"],
   337→    queryFn: async () => {
   338→      const res = await fetch("/api/checklists", { cache: "no-store" });
   339→      if (!res.ok) return null;
   340→      return res.json();
   341→    },
   342→  });
   343→  const checklists: any[] = checklistsData?.items ?? [];
   344→
   345→  function parseRules(text: string): string[] {
   346→    return text
   347→      .split("\n")
   348→      .map((l) => l.trim())
   349→      .filter((l) => l.length > 0);
   350→  }
   351→
   352→  // Parse the weight/required config text into a map
   353→  function parseRuleConfig(text: string): Map<string, { weight: number; required: boolean }> {
   354→    const map = new Map<string, { weight: number; required: boolean }>();
   355→    for (const line of text.split("\n")) {
   356→      const parts = line.split("|").map((p) => p.trim());
   357→      if (parts.length >= 1 && parts[0]) {
   358→        const ruleText = parts[0];
   359→        const weight = parts[1] ? parseFloat(parts[1]) || 1 : 1;
   360→        const required = parts[2] ? parts[2].toLowerCase() === "required" : true;
   361→        map.set(ruleText, { weight, required });
   362→      }
   363→    }
   364→    return map;
   365→  }
   366→
   367→  // Build rulesJson with weight/required metadata per rule, plus grading thresholds
   368→  function buildRulesJson() {
   369→    const config = parseRuleConfig(ruleConfig);
   370→    const buildSection = (rules: string) => parseRules(rules).map((rule) => {
   371→      const cfg = config.get(rule) ?? { weight: 1, required: true };
   372→      return { text: rule, weight: cfg.weight, required: cfg.required };
   373→    });
   374→    return {
   375→      entry: buildSection(entryRules),
   376→      stop: buildSection(stopRules),
   377→      target: buildSection(targetRules),
   378→      management: buildSection(managementRules),
   379→      invalidation: buildSection(invalidationRules),
   380→      gradingThresholds: {
   381→        "A+": parseFloat(aplusThreshold) || 0.9,
   382→        A: parseFloat(aThreshold) || 0.75,
   383→        B: parseFloat(bThreshold) || 0.6,
   384→        C: parseFloat(cThreshold) || 0.4,
   385→      },
   386→    };
   387→  }

   392→    const rules = buildRulesJson();
   393→    const res = await fetch("/api/strategies", {
   394→      method: "POST",
   395→      headers: { "Content-Type": "application/json" },
   396→      body: JSON.stringify({
   397→        name,
   398→        description,
   399→        purpose,
   400→        instruments,
   401→        market,
   402→        timeframe,
   403→        session,
   404→        checklistConfigId: checklistConfigId || null,
   405→        rules,
   406→      }),
   407→    });
   408→    if (res.ok) {
   409→      toast.success("Strategy created.");
   410→      onCreated();
   411→      onClose();
   412→    } else {
   413→      toast.error("Failed to create strategy.");
   414→    }
   415→    setLoading(false);
   416→  }
   417→
   418→  return (
   419→    <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   420→      <DialogHeader>
   421→        <DialogTitle>New Strategy</DialogTitle>
   422→      </DialogHeader>

   424→        {/* Basics */}

   429→        <div className="space-y-2">
   430→          <FieldLabel hint="Brief description of the strategy">Description</FieldLabel>

   432→        </div>
   433→        <div className="space-y-2">
   434→          <FieldLabel hint="Why this strategy exists">Purpose</FieldLabel>
   435→          <Textarea value={purpose} onChange={(e) => setPurpose(e.target.value)} rows={2} />
   436→        </div>
   437→
   438→        {/* Configuration */}
   439→        <div className="grid grid-cols-2 gap-3">
   440→          <div className="space-y-2">
   441→            <FieldLabel hint="Which market this strategy targets">Market</FieldLabel>
   442→            <Select value={market} onValueChange={setMarket}>
   443→              <SelectTrigger><SelectValue /></SelectTrigger>

   545→function NewVersionDialog({ strategyId, latestLabel, latestRulesJson, onClose, onCreated }: { strategyId: string; latestLabel?: string; latestRulesJson?: string; onClose: () => void; onCreated: () => void }) {
   546→  // Pre-fill rules from the latest version
   547→  const initialRules = (() => {
   548→    try {
   549→      const parsed = JSON.parse(latestRulesJson ?? "{}");
   550→      // Handle both old format (string[]) and new format ({text, weight, required}[])
   551→      const extractText = (arr: any[]) => arr.map((r: any) => typeof r === "string" ? r : r.text);
   552→      const extractConfig = (arr: any[]) => arr.map((r: any) => {
   553→        if (typeof r === "string") return `${r} | 1 | required`;
   554→        return `${r.text} | ${r.weight ?? 1} | ${r.required ? "required" : "optional"}`;
   555→      });
   556→      return {
   557→        entry: Array.isArray(parsed.entry) ? extractText(parsed.entry).join("\n") : "",
   558→        stop: Array.isArray(parsed.stop) ? extractText(parsed.stop).join("\n") : "",
   559→        target: Array.isArray(parsed.target) ? extractText(parsed.target).join("\n") : "",
   560→        management: Array.isArray(parsed.management) ? extractText(parsed.management).join("\n") : "",
   561→        invalidation: Array.isArray(parsed.invalidation) ? extractText(parsed.invalidation).join("\n") : "",
   562→        ruleConfig: Array.isArray(parsed.entry) ? extractConfig([...(parsed.entry||[]), ...(parsed.stop||[]), ...(parsed.target||[]), ...(parsed.management||[]), ...(parsed.invalidation||[])]).join("\n") : "",
   563→        thresholds: parsed.gradingThresholds ?? { "A+": 0.9, A: 0.75, B: 0.6, C: 0.4 },
   564→      };
   565→    } catch {
   566→      return { entry: "", stop: "", target: "", management: "", invalidation: "", ruleConfig: "", thresholds: { "A+": 0.9, A: 0.75, B: 0.6, C: 0.4 } };
   567→    }
   568→  })();
   569→
   570→  const [versionLabel, setVersionLabel] = useState(incrementVersion(latestLabel));
   571→  const [changeReason, setChangeReason] = useState("");
   572→  const [changeSummary, setChangeSummary] = useState("");
   573→  const [entryRules, setEntryRules] = useState(initialRules.entry);
   574→  const [stopRules, setStopRules] = useState(initialRules.stop);
   575→  const [targetRules, setTargetRules] = useState(initialRules.target);
   576→  const [managementRules, setManagementRules] = useState(initialRules.management);
   577→  const [invalidationRules, setInvalidationRules] = useState(initialRules.invalidation);
   578→  const [ruleConfig, setRuleConfig] = useState(initialRules.ruleConfig);
   579→  const [aplusThreshold, setAplusThreshold] = useState(String(initialRules.thresholds["A+"] ?? 0.9));
   580→  const [aThreshold, setAThreshold] = useState(String(initialRules.thresholds.A ?? 0.75));
   581→  const [bThreshold, setBThreshold] = useState(String(initialRules.thresholds.B ?? 0.6));
   582→  const [cThreshold, setCThreshold] = useState(String(initialRules.thresholds.C ?? 0.4));
   583→  const [loading, setLoading] = useState(false);
   584→
   585→  function incrementVersion(v?: string): string {
   586→    if (!v) return "1.0";
   587→    const [maj, min] = v.split(".");
   588→    return `${maj}.${Number(min) + 1}`;
   589→  }
   590→
   591→  function parseRules(text: string): string[] {
   592→    return text.split("\n").map((l) => l.trim()).filter((l) => l.length > 0);
   593→  }
   594→
   595→  function parseRuleConfig(text: string): Map<string, { weight: number; required: boolean }> {
   596→    const map = new Map<string, { weight: number; required: boolean }>();
   597→    for (const line of text.split("\n")) {
   598→      const parts = line.split("|").map((p) => p.trim());
   599→      if (parts.length >= 1 && parts[0]) {
   600→        const ruleText = parts[0];
   601→        const weight = parts[1] ? parseFloat(parts[1]) || 1 : 1;
   602→        const required = parts[2] ? parts[2].toLowerCase() === "required" : true;
   603→        map.set(ruleText, { weight, required });
   604→      }
   605→    }
   606→    return map;
   607→  }
   608→
   609→  function buildRulesJson() {
   610→    const config = parseRuleConfig(ruleConfig);
   611→    const buildSection = (rules: string) => parseRules(rules).map((rule) => {
   612→      const cfg = config.get(rule) ?? { weight: 1, required: true };
   613→      return { text: rule, weight: cfg.weight, required: cfg.required };
   614→    });
   615→    return {
   616→      entry: buildSection(entryRules),
   617→      stop: buildSection(stopRules),
   618→      target: buildSection(targetRules),
   619→      management: buildSection(managementRules),
   620→      invalidation: buildSection(invalidationRules),
   621→      gradingThresholds: {
   622→        "A+": parseFloat(aplusThreshold) || 0.9,
   623→        A: parseFloat(aThreshold) || 0.75,
   624→        B: parseFloat(bThreshold) || 0.6,
   625→        C: parseFloat(cThreshold) || 0.4,
   626→      },
   627→    };
   628→  }
   629→
   630→  async function submit() {

   632→    setLoading(true);
   633→    const rules = buildRulesJson();

   635→      method: "PUT",
   636→      headers: { "Content-Type": "application/json" },
   637→      body: JSON.stringify({ versionLabel, changeReason, changeSummary, rules }),
   638→    });
   639→    if (res.ok) {

   672→
