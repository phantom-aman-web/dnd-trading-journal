    93→  // Seed pseudo-random with a fixed seed for reproducibility

   100→  // Generate 70 trades over ~70 days
   101→  for (let i = 0; i < 70; i++) {

   153→      ? psychAfterSets[0]

   160→    if (strat.flags.includes("revenge_trade")) tags.push("revenge");

   163→    out.push({
   164→      daysAgo,
   165→      hour,
   166→      minute,
   167→      symbol: instr.symbol,
   168→      direction,
   169→      session,
   170→      entryPrice: entry,
   171→      stopPrice: stop,
   172→      targetPrice: target,
   173→      exitPrice: exit,
   174→      qty,
   175→      feesCents: Math.round((rand() * 4 + 1) * 100),
   176→      commissionCents: Math.round((rand() * 6 + 2) * 100),
   177→      swapCents: Math.round(rand() * 100),
   178→      strategyName: strat.name,
   179→      grade: strat.grade,
   180→      setupScore: strat.score,
   181→      behaviorFlags: strat.flags,
   182→      psychBefore: psychB,
   183→      psychAfter: psychA,
   184→      confidence,
   185→      thesis: isWin
   186→        ? "Clear liquidity sweep followed by displacement and MSS confirmation. Entry inside the FVG with defined risk to recent swing low."
   187→        : "Setup appeared valid but market structure did not confirm; displacement stalled at first mitigation. Lesson recorded.",
   188→      lessons: !isWin ? "Need to wait for explicit MSS confirmation before committing to entry." : undefined,
   189→      tags,
   190→    });
   191→  }
   192→  // sort oldest first
   193→  return out.sort((a, b) => b.daysAgo - a.daysAgo);
   194→}

   196→/**
   197→ * Generate plausible checklist answers for a trade based on its intended
   198→ * grade. The patterns follow the spec's remediation guidance:
   199→ *   - A+ trades:        all 8 items checked
   200→ *   - A trades:         6-7 items checked (one required unchecked)
   201→ *   - B trades:         5 items checked
   202→ *   - C trades:         3-4 items checked
   203→ *   - Invalid trades:   2 items checked
   204→ *
   205→ * The actual evaluated grade/score is computed via `evaluateChecklist`
   206→ * (which applies the required-item cap), so the stored `setupGrade` may
   207→ * differ from the intended grade when a required item is unchecked —
   208→ * that's the realistic behaviour the spec mandates.
   209→ *
   210→ * Items are expected in the order:
   211→ *   [htf(req,2), liq(req,2), disp(req,2), mss(req,1.5),
   212→ *    fvg(opt,1), zone(req,1.5), risk(req,1), session(req,1)]
   213→ */
   214→function generateChecklistAnswers(
   215→  grade: string,
   216→  items: ChecklistItem[],
   217→): ChecklistAnswer {
   218→  const answers: ChecklistAnswer = {};
   219→  // Default everything to unchecked.
   220→  for (const item of items) {
   221→    answers[item.id] = { checked: false };
   222→  }
   223→  if (items.length === 0) return answers;
   224→
   225→  // Helper: check the first N items in declared order.
   226→  const checkFirst = (n: number) => {
   227→    for (let i = 0; i < Math.min(n, items.length); i++) {
   228→      answers[items[i].id] = { checked: true };
   229→    }
   230→  };
   231→
   232→  switch (grade) {
   233→    case "A+":
   234→      // All items checked.
   235→      checkFirst(items.length);
   236→      break;
   237→    case "A":
   238→      // 6-7 items checked (one required unchecked). Check everything,
   239→      // then uncheck the 4th item (mss — a "minor" required item being
   240→      // missed is a plausible trader slip that turns an A-quality setup
   241→      // into a capped-C).
   242→      checkFirst(items.length);
   243→      if (items[3]) {
   244→        answers[items[3].id] = { checked: false };
   245→      }
   246→      break;
   247→    case "B":
   248→      // 5 items checked.
   249→      checkFirst(5);
   250→      break;
   251→    case "C":
   252→      // 3-4 items checked. Use 4 to leave the score non-trivial.
   253→      checkFirst(4);
   254→      break;
   255→    case "Invalid":
   256→    default:
   257→      // 2 items checked.
   258→      checkFirst(2);
   259→      break;
   260→  }
   261→  return answers;
   262→}
