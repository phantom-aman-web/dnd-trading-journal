   100→   100→export function SetupWizard() {
   101→   101→  const {
   102→   102→    state,
   103→   103→    setupStep,
   104→   104→    nextSetupStep,
   105→   105→    prevSetupStep,
   106→   106→    completeStep,
   107→   107→    setState,
   108→   108→  } = useOnboarding();
   109→   109→  const visible = state === "setup";
   110→   110→
   111→   111→  // Escape key skips the wizard.
   112→   112→  useEffect(() => {
   113→   113→    if (!visible) return;
   114→   114→    const handler = (e: KeyboardEvent) => {
   115→   115→      if (e.key === "Escape") {
   116→   116→        e.preventDefault();
   117→   117→        setState("skipped");
   118→   118→      }
   119→   119→    };
   120→   120→    window.addEventListener("keydown", handler);
   121→   121→    return () => window.removeEventListener("keydown", handler);
   122→   122→  }, [visible, setState]);
   123→   123→
   124→   124→  if (!visible) return null;
   125→   125→
   126→   126→  const step = STEP_META[setupStep];
   127→   127→  const Icon = step.icon;
   128→   128→  const progress = ((setupStep + 1) / TOTAL_SETUP_STEPS) * 100;
   129→   129→
   130→   130→  function handleSkip() {
   131→   131→    completeStep(step.key);
   132→   132→    if (setupStep === TOTAL_SETUP_STEPS - 1) {
   133→   133→      setState("tour");
   134→   134→    } else {
   135→   135→      nextSetupStep();
   136→   136→    }
   137→   137→  }
   138→   138→
   139→   139→  return (
   140→   140→    <div
   141→   141→      role="dialog"
   142→   142→      aria-modal="true"
   143→   143→      aria-labelledby="setup-title"
   144→   144→      className="fixed inset-0 z-[100] flex items-center justify-center bg-black/60 backdrop-blur-sm p-4 animate-in fade-in duration-200"
   145→   145→    >
   146→   146→      <div className="w-full max-w-2xl max-h-[92vh] flex flex-col rounded-xl border border-border bg-card text-card-foreground shadow-xl animate-in zoom-in-95 duration-200">
   147→   147→        {/* Header */}
   148→   148→        <div className="flex items-start justify-between gap-3 border-b border-border p-5 sm:p-6">
   149→   149→          <div className="flex items-start gap-3 min-w-0">
   150→   150→            <div className="mt-0.5 flex h-9 w-9 shrink-0 items-center justify-center rounded-md bg-primary/10 text-primary">
   151→   151→              <Icon className="h-5 w-5" />
   152→   152→            </div>
   153→   153→            <div className="min-w-0">
   154→   154→              <div className="flex items-center gap-2">
   155→   155→                <Badge variant="outline" className="text-[10px] uppercase tracking-wider">
   156→   156→                  Step {setupStep + 1} of {TOTAL_SETUP_STEPS}
   157→   157→                </Badge>
   158→   158→              </div>
   159→   159→              <h2 id="setup-title" className="mt-1 text-lg font-semibold tracking-tight">
   160→   160→                {step.title}
   161→   161→              </h2>
   162→   162→              <p className="mt-1 text-sm text-muted-foreground">
   163→   163→                {step.description}
   164→   164→              </p>
   165→   165→            </div>
   166→   166→          </div>
   167→   167→          <button
   168→   168→            aria-label="Close setup wizard"
   169→   169→            onClick={() => setState("skipped")}
   170→   170→            className="inline-flex h-8 w-8 items-center justify-center rounded-md text-muted-foreground hover:bg-muted hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   171→   171→          >
   172→   172→            <X className="h-4 w-4" />
   173→   173→          </button>
   174→   174→        </div>
   175→   175→
   176→   176→        {/* Progress */}
   177→   177→        <div className="px-5 sm:px-6 pt-4">
   178→   178→          <Progress value={progress} className="h-1.5" />
   179→   179→        </div>
   180→   180→
   181→   181→        {/* Step body */}
   182→   182→        <div className="flex-1 overflow-y-auto scroll-thin px-5 sm:px-6 py-5">
   183→   183→          {setupStep === 0 && <AccountStep onDone={() => advance(setupStep, nextSetupStep, completeStep, step.key, setState)} />}
   184→   184→          {setupStep === 1 && <InstrumentStep onDone={() => advance(setupStep, nextSetupStep, completeStep, step.key, setState)} />}
   185→   185→          {setupStep === 2 && <StrategyStep onDone={() => advance(setupStep, nextSetupStep, completeStep, step.key, setState)} />}
   186→   186→          {setupStep === 3 && <StrategyRulesStep onDone={() => advance(setupStep, nextSetupStep, completeStep, step.key, setState)} />}
   187→   187→          {setupStep === 4 && <PlanStep onDone={() => advance(setupStep, nextSetupStep, completeStep, step.key, setState)} />}
   188→   188→          {setupStep === 5 && <TradeStep onDone={() => advance(setupStep, nextSetupStep, completeStep, step.key, setState)} />}
   189→   189→        </div>
   190→   190→
   191→   191→        {/* Footer */}
   192→   192→        <div className="flex items-center justify-between gap-2 border-t border-border p-4 sm:px-6">
   193→   193→          <Button
   194→   194→            variant="ghost"
   195→   195→            size="sm"
   196→   196→            onClick={() => prevSetupStep()}
   197→   197→            disabled={setupStep === 0}
   198→   198→            className="text-muted-foreground"
   199→   199→          >
   200→   200→            <ArrowLeft className="h-4 w-4" /> Back
   201→   201→          </Button>
   202→   202→          <Button
   203→   203→            variant="ghost"
   204→   204→            size="sm"
   205→   205→            onClick={handleSkip}
   206→   206→            className="text-muted-foreground"
   207→   207→          >
   208→   208→            Skip <SkipForward className="h-4 w-4" />
   209→   209→          </Button>
   210→   210→        </div>
   211→   211→      </div>
   212→   212→    </div>
   213→   213→  );
   214→   214→}
   215→   215→
   216→   216→function advance(
   217→   217→  currentStep: number,
   218→   218→  next: () => void,
   219→   219→  complete: (s: SetupStep) => void,
   220→   220→  key: SetupStep,
   221→   221→  setState: (s: "tour" | "skipped") => void,
   222→   222→) {
   223→   223→  complete(key);
   224→   224→  if (currentStep === TOTAL_SETUP_STEPS - 1) {
   225→   225→    setState("tour");
   226→   226→  } else {
   227→   227→    next();
   228→   228→  }
   229→   229→}

   235→   235→function AccountStep({ onDone }: { onDone: () => void }) {
   236→   236→  const qc = useQueryClient();
   237→   237→  const [name, setName] = useState("Primary");
   238→   238→  const [currency, setCurrency] = useState("USD");
   239→   239→  const [balance, setBalance] = useState("10000");
   240→   240→  const [loading, setLoading] = useState(false);
   241→   241→
   242→   242→  async function submit() {
   243→   243→    if (!name.trim()) {
   244→   244→      toast.error("Account name is required");
   245→   245→      return;
   246→   246→    }
   247→   247→    setLoading(true);
   248→   248→    try {
   249→   249→      const bal = parseFloat(balance);
   250→   250→      const startingBalanceCents = Number.isFinite(bal)
   251→   251→        ? Math.round(bal * 100)
   252→   252→        : 0;
   253→   253→      const res = await fetch("/api/accounts", {
   254→   254→        method: "POST",
   255→   255→        headers: { "Content-Type": "application/json" },
   256→   256→        body: JSON.stringify({
   257→   257→          name: name.trim(),
   258→   258→          currency,
   259→   259→          startingBalanceCents,
   260→   260→          isDefault: true,
   261→   261→        }),
   262→   262→      });
   263→   263→      if (!res.ok) {
   264→   264→        const data = await res.json().catch(() => null);
   265→   265→        toast.error(data?.error ?? "Failed to create account");
   266→   266→        setLoading(false);
   267→   267→        return;
   268→   268→      }
   269→   269→      qc.invalidateQueries({ queryKey: ["accounts"] });
   270→   270→      qc.invalidateQueries({ queryKey: ["me"] });
   271→   271→      toast.success("Trading account created.");
   272→   272→      onDone();
   273→   273→    } catch {
   274→   274→      toast.error("Network error");
   275→   275→      setLoading(false);
   276→   276→    }
   277→   277→  }
   278→   278→
   279→   279→  return (
   280→   280→    <form
   281→   281→      className="space-y-4"
   282→   282→      onSubmit={(e) => {
   283→   283→        e.preventDefault();
   284→   284→        submit();
   285→   285→      }}
   286→   286→    >
   287→   287→      <div className="space-y-2">
   288→   288→        <FieldLabel required>Account Name</FieldLabel>
   289→   289→        <Input
   290→   290→          value={name}
   291→   291→          onChange={(e) => setName(e.target.value)}
   292→   292→          placeholder="e.g. Primary, FTMO 100k"
   293→   293→          autoFocus
   294→   294→        />
   295→   295→      </div>
   296→   296→      <div className="grid grid-cols-2 gap-3">
   297→   297→        <div className="space-y-2">
   298→   298→          <FieldLabel>Currency</FieldLabel>
   299→   299→          <Select value={currency} onValueChange={setCurrency}>
   300→   300→            <SelectTrigger className="w-full">
   301→   301→              <SelectValue />
   302→   302→            </SelectTrigger>
   303→   303→            <SelectContent>
   304→   304→              {CURRENCIES.map((c) => (
   305→   305→                <SelectItem key={c} value={c}>
   306→   306→                  {c}
   307→   307→                </SelectItem>
   308→   308→              ))}
   309→   309→            </SelectContent>
   310→   310→          </Select>
   311→   311→        </div>
   312→   312→        <div className="space-y-2">
   313→   313→          <FieldLabel required>Starting Balance</FieldLabel>
   314→   314→          <Input
   315→   315→            type="number"
   316→   316→            min="0"
   317→   317→            step="0.01"
   318→   318→            value={balance}
   319→   319→            onChange={(e) => setBalance(e.target.value)}
   320→   320→          />
   321→   321→        </div>
   322→   322→      </div>
   323→   323→      <StepFooter>
   324→   324→        <Button type="submit" disabled={loading}>
   325→   325→          {loading ? (
   326→   326→            <Loader2 className="h-4 w-4 animate-spin" />
   327→   327→          ) : (
   328→   328→            <Check className="h-4 w-4" />
   329→   329→          )}
   330→   330→          Create Account
   331→   331→        </Button>
   332→   332→      </StepFooter>
   333→   333→    </form>
   334→   334→  );
   335→   335→}

   853→   853→function buildRulesJson(entryRules: string, ruleConfig: string) {
   854→   854→  const lines = entryRules
   855→   855→    .split("\n")
   856→   856→    .map((l) => l.trim())
   857→   857→    .filter((l) => l.length > 0);
   858→   858→
   859→   859→  const cfg = new Map<string, { weight: number; required: boolean }>();
   860→   860→  for (const line of ruleConfig.split("\n")) {
   861→   861→    const parts = line.split("|").map((p) => p.trim());
   862→   862→    if (parts.length >= 1 && parts[0]) {
   863→   863→      const ruleText = parts[0];
   864→   864→      const weight = parts[1] ? parseFloat(parts[1]) || 1 : 1;
   865→   865→      const required = parts[2] ? parts[2].toLowerCase() === "required" : true;
   866→   866→      cfg.set(ruleText, { weight, required });
   867→   867→    }
   868→   868→  }
   869→   869→
   870→   870→  const entry = lines.map((rule) => {
   871→   871→    const c = cfg.get(rule) ?? { weight: 1, required: true };
   872→   872→    return { text: rule, weight: c.weight, required: c.required };
   873→   873→  });
   874→   874→
   875→   875→  return {
   876→   876→    entry,
   877→   877→    stop: [],
   878→   878→    target: [],
   879→   879→    management: [],
   880→   880→    invalidation: [],
   881→   881→    gradingThresholds: {
   882→   882→      "A+": 0.9,
   883→   883→      A: 0.75,
   884→   884→      B: 0.6,
   885→   885→      C: 0.4,
   886→   886→    },
   887→   887→  };
   888→   888→}
