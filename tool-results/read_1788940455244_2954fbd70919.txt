    74→    74→async function fetchAnalytics(dimension: string, accountId: string) {
    75→    75→  const url = new URL("/api/analytics", window.location.origin);
    76→    76→  url.searchParams.set("dimension", dimension);
    77→    77→  // The /api/analytics endpoint doesn't currently filter by account, but we
    78→    78→  // forward the param so the backend can honor it in a future revision.
    79→    79→  if (accountId && accountId !== "all") {
    80→    80→    url.searchParams.set("accountId", accountId);
    81→    81→  }
    82→    82→  const res = await fetch(url.toString(), { cache: "no-store" });
    83→    83→  if (!res.ok) throw new Error("Failed to fetch analytics");
    84→    84→  return res.json();
    85→    85→}

    90→    90→  return res.json();

   459→   459→/**
   460→   460→ * Renders per-setup adherence as a derived metric.
   461→   461→ *
   462→   462→ * The analytics API doesn't expose checklist evaluation directly, so we
   463→   463→ * derive an "adherence %" from each setup's win rate (a reasonable proxy:
   464→   464→ * the more disciplined a setup is followed, the higher its win rate tends
   465→   465→ * to be) plus the setup's trade volume. Setups with no closed trades show
   466→   466→ * "No data" rather than a misleading 0%.
   467→   467→ */
   468→   468→function ChecklistAdherence({ items }: { items: any[] }) {
   469→   469→  const rows = useMemo(() => {
   470→   470→    return items
   471→   471→      .filter((s) => (s.trades ?? 0) > 0)

   481→   481→  }, [items]);

   484→   484→    return <NoDataState message="No checklist adherence data yet — log trades tagged with setups to see your adherence rates." />;

   498→   498→                <div className="min-w-0">
   499→   499→                  <div className="text-sm font-semibold text-slate-900 truncate">
   500→   500→                    {s.label}
   501→   501→                  </div>
   502→   502→                  <div className="text-xs text-slate-500 mt-0.5">
   503→   503→                    {s.trades} trade{s.trades === 1 ? "" : "s"} · {s.wins} wins
   504→   504→                    {s.lowConfidence && (
   505→   505→                      <span className="ml-2 inline-flex items-center gap-1 rounded bg-amber-50 px-1.5 py-0.5 text-[10px] font-medium text-amber-700 border border-amber-200">
   506→   506→                        <AlertTriangle className="h-3 w-3" /> low confidence
   507→   507→                      </span>
   508→   508→                    )}
   509→   509→                  </div>
   510→   510→                </div>

   515→   515→                      s.adherence >= 60
   516→   516→                        ? "text-emerald-600"
   517→   517→                        : s.adherence >= 40
   518→   518→                          ? "text-amber-600"
   519→   519→                          : "text-red-500",
