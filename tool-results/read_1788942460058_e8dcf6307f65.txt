    37→async function fetchMe() {
    38→  const res = await fetch("/api/me", { cache: "no-store" });
    39→  if (!res.ok) return null;
    40→  return res.json();
    41→}
    42→
    43→/** Fetch the user's accounts as a plain array, defensively handling both
    44→ *  the standard { items: [...] } envelope and the bare-array response
    45→ *  shapes the /api/accounts endpoint may return. Returns [] on any error. */
    46→async function fetchAccounts() {
    47→  const res = await fetch("/api/accounts", { cache: "no-store" });
    48→  if (!res.ok) return [];
    49→  const data = await res.json();
    50→  const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
    51→  return Array.isArray(arr) ? arr : [];
    52→}
    53→
    54→/** Fetch the user's strategies as a plain array (defensive, see fetchAccounts). */
    55→async function fetchStrategies() {
    56→  const res = await fetch("/api/strategies", { cache: "no-store" });
    57→  if (!res.ok) return [];
    58→  const data = await res.json();
    59→  const arr = data?.items ?? data?.strategies ?? (Array.isArray(data) ? data : []);
    60→  return Array.isArray(arr) ? arr : [];
    61→}

   445→  async function handleFileUpload(files: FileList | File[]) {
   446→    const fileList = Array.from(files);
   447→    for (const file of fileList) {
   448→      const formData = new FormData();
   449→      formData.append("file", file);
   450→      if (editingId) formData.append("tradeId", editingId);
   451→      // Detect evidence kind from the file's MIME type so the form
   452→      // immediately knows whether to render an <img> (annotation-able) or a
   453→      // <video controls> element — the server may not always echo back the
   454→      // kind field reliably.
   455→      const clientKind = file.type.startsWith("video/") ? "video" : "image";
   456→      try {
   457→        const res = await fetch("/api/media", { method: "POST", body: formData });
   458→        if (res.ok) {
   459→          const data = await res.json();
   460→          // Fetch signed URL
   461→          const detailRes = await fetch(`/api/media/${data.id}`);
   462→          const detail = await detailRes.json();
   463→          setForm((f: any) => ({
   464→            ...f,
   465→            uploadedMedia: [...(f.uploadedMedia ?? []), { ...data, kind: data.kind ?? clientKind, url: detail.url }],
   466→          }));
   467→          toast.success(`${file.name} uploaded.`);
   468→        } else {
   469→          const err = await res.json().catch(() => ({}));
   470→          toast.error(err?.error ?? `Failed to upload ${file.name}`);
   471→        }
   472→      } catch {
   473→        toast.error(`Failed to upload ${file.name}`);
   474→      }
   475→    }
   476→  }

   478→  function setField(path: string, value: any) {
   479→    setForm((f: any) => {
   480→      const next = { ...f };
   481→      const parts = path.split(".");
   482→      let cur: any = next;
   483→      for (let i = 0; i < parts.length - 1; i++) {
   484→        cur[parts[i]] = { ...cur[parts[i]] };
   485→        cur = cur[parts[i]];
   486→      }
   487→      cur[parts[parts.length - 1]] = value;
   488→      return next;
   489→    });
   490→  }

   497→      for (let i = 0; i < parts.length - 1; i++) {
   498→        cur[parts[i]] = { ...cur[parts[i]] };
   499→        cur = cur[parts[i]];
   500→      }
   501→      const key = parts[parts.length - 1];
   502→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   503→      const idx = arr.indexOf(value);
   504→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   505→      cur[key] = arr;
   506→      return next;
   507→    });
   508→  }

   510→  // Update one A+ checklist answer field (`checked` or `note`) for the given
   511→  // item id. Used by the checkbox + evidence-note inputs in the ICT Setup tab.
   512→  function updateChecklistAnswer(
   513→    itemId: string,
   514→    field: "checked" | "note",
   515→    value: boolean | string,
   516→  ) {
   517→    setForm((f: any) => {
   518→      const current = f.checklistAnswers?.[itemId] ?? { checked: false };
   519→      const next = {
   520→        ...f.checklistAnswers,
   521→        [itemId]: {
   522→          ...current,
   523→          [field]: value,
   524→        },
   525→      };
   526→      // If a checked item is being unchecked, drop its note so storage stays clean.
   527→      if (field === "checked" && value === false && next[itemId].note) {
   528→        delete next[itemId].note;
   529→      }
   530→      return { ...f, checklistAnswers: next };
   531→    });
   532→  }
