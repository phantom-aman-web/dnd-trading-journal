    41→async function fetchTrade(id: string) {
    42→  const res = await fetch(`/api/trades/${id}`, { cache: "no-store" });
    43→  if (!res.ok) throw new Error("Failed to load trade");
    44→  return res.json();
    45→}

   603→function SetupSection({ title, data, isList }: { title: string; data: any; isList?: boolean }) {
   604→  if (!data) return null;
   605→  return (
   606→    <div>
   607→      <div className="text-xs text-muted-foreground uppercase tracking-wide">{title}</div>
   608→      {isList && Array.isArray(data) ? (
   609→        <div className="flex flex-wrap gap-1 mt-1">
   610→          {data.map((d: string, i: number) => <Badge key={i} variant="secondary" className="text-xs">{d}</Badge>)}
   611→        </div>
   612→      ) : typeof data === "object" ? (
   613→        <div className="mt-1 text-sm">
   614→          {Object.entries(data).map(([k, v]) => (
   615→            <div key={k}>
   616→              <span className="text-muted-foreground">{k}:</span> {String(v)}
   617→            </div>
   618→          ))}
   619→        </div>
   620→      ) : (
   621→        <div className="font-medium mt-1">{String(data)}</div>
   622→      )}
   623→    </div>
   624→  );
   625→}
