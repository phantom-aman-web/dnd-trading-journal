    38→    38→  contractSize?: Decimal; // e.g. 100000 for standard forex lot
    39→    39→  pipSize?: Decimal; // e.g. 0.0001 for forex
    40→    40→  /**
    41→    41→   * Value per 1.0 price unit per 1.0 quantity, in cents.
    42→    42→   * Defaults to 100 (i.e. $1 per 1.0 price move per 1.0 qty) — backward compatible
    43→    43→   * with the original hardcoded behaviour for stocks/crypto where price * qty
    44→    44→   * directly equals the dollar P&L. Callers may override this with an
    45→    45→   * instrument-derived value (see `computePointValueCents` below) so that
    46→    46→   * forex/metals/indices/futures whose "1.0 price move" semantics differ from
    47→    47→   * a 1-share stock are scaled correctly.
    48→    48→   */
    49→    49→  pointValueCents?: Cents;

    51→    51→
    52→    52→/**
    53→    53→ * Compute an instrument-aware `pointValueCents` from the Instrument's stored
    54→    54→ * spec fields. The returned value represents the dollar value (in cents) of a
    55→    55→ * single tick (or pip, when no tickSize is set) per 1.0 contract.
    56→    56→ *
    57→    57→ * - Forex (e.g. 100000 * 0.0001 * 100 = 1000 cents = $10 per pip per lot)
    58→    58→ * - Gold/metals, indices, futures: contractSize * tickSize * 100
    59→    59→ * - Crypto/stocks (contractSize 1, tickSize 0.01): falls back to 100 to keep
    60→    60→ *   the 1:1 price-to-dollar semantics that the default 100 already provides.
    61→    61→ *
    62→    62→ * All inputs are decimal strings (per spec §7); non-numeric values are
    63→    63→ * coerced to 0 so the result is always a finite, non-NaN integer.
    64→    64→ */
    65→    65→export function computePointValueCents(
    66→    66→  contractSize: string | number | null | undefined,
    67→    67→  tickSize: string | number | null | undefined,
    68→    68→  pipSize: string | number | null | undefined,
    69→    69→): Cents {
    70→    70→  const cs = Number(contractSize ?? "1");
    71→    71→  const ts = Number(tickSize ?? pipSize ?? "0.01");
    72→    72→  if (!Number.isFinite(cs) || !Number.isFinite(ts)) return 100;
    73→    73→  const cents = Math.round(cs * ts * 100);
    74→    74→  // For instruments where contractSize * tickSize rounds to ≤1 cent (e.g.
    75→    75→  // 1-share stocks with a 0.01 tick → 1 cent), preserve the historical
    76→    76→  // 1:1-with-price default of 100 cents so existing stock/crypto P&L math is
    77→    77→  // unaffected. Only non-trivial contracts (forex/metals/indices/futures)
    78→    78→  // produce a derived value > 100 and therefore change behaviour.
    79→    79→  if (!Number.isFinite(cents) || cents <= 1) return 100;
    80→    80→  return cents;
    81→    81→}

   103→   103→/**
   104→   104→ * Extract mood-tag strings from a `psychBeforeJson` (or `psychAfterJson`)
   105→   105→ * column. The column is stored as a JSON **object** whose `moodTags` field is
   106→   106→ * an array of strings (per spec section 38), but legacy rows may store a
   107→   107→ * bare array — both shapes are supported. Any other shape yields `[]`.
   108→   108→ *
   109→   109→ * Never throws: malformed JSON, null, or unexpected shapes degrade to `[]`.
   110→   110→ */
   111→   111→export function safeParsePsychTags(json: string | null | undefined): string[] {
   112→   112→  if (!json) return [];
   113→   113→  let v: unknown;
   114→   114→  try {
   115→   115→    v = JSON.parse(json);
   116→   116→  } catch {
   117→   117→    return [];
   118→   118→  }
   119→   119→  if (Array.isArray(v)) {
   120→   120→    // Backward-compat: legacy rows stored a bare array of tags.
   121→   121→    return v.filter((x): x is string => typeof x === "string");
   122→   122→  }
   123→   123→  if (v && typeof v === "object") {
   124→   124→    const tags = (v as { moodTags?: unknown }).moodTags;
   125→   125→    if (Array.isArray(tags)) {
   126→   126→      return tags.filter((x): x is string => typeof x === "string");
   127→   127→    }
   128→   128→  }
   129→   129→  return [];
   130→   130→}

   132→   132→// ---------------------------------------------------------------------------
   133→   133→// Core: compute trade P&L from execution ledger (spec section 33)
   134→   134→// ---------------------------------------------------------------------------
   135→   135→
   136→   136→/**
   137→   137→ * Calculate the gross P&L in cents from a list of fills.
   138→   138→ * Method: volume-weighted average entry and exit prices, signed by direction.
   139→   139→ * gross = (exitAvg - entryAvg) * qty * contractSize * pointValue  (long)
   140→   140→ * gross = (entryAvg - exitAvg) * qty * contractSize * pointValue  (short)
   141→   141→ * Only matches qty between entries and exits.
   142→   142→ */
   143→   143→export function calculateTradePnl(input: TradeCalcInput): TradeCalcResult {
   144→   144→  const contractSize = input.contractSize ?? "1";
   145→   145→  const pointValueCents = input.pointValueCents ?? 100; // default 1.00 per 1.0 qty per 1.0 price
   146→   146→
   147→   147→  const entries = input.fills.filter((f) => f.kind === "entry");
   148→   148→  const exits = input.fills.filter((f) => f.kind === "exit");
   149→   149→
   150→   150→  // Volume-weighted average entry price
   151→   151→  const entryAvg = vwap(entries);
   152→   152→  const exitAvg = vwap(exits);
   153→   153→
   154→   154→  // Matched quantity (min of total entry qty and total exit qty)
   155→   155→  const entryQty = sumQty(entries);
   156→   156→  const exitQty = sumQty(exits);
   157→   157→  const matchedQty = dCmp(entryQty, exitQty) <= 0 ? entryQty : exitQty;
   158→   158→
   159→   159→  let grossPnlCents: Cents = 0;
   160→   160→  if (!dIsZero(matchedQty) && entryAvg && exitAvg) {
   161→   161→    // price diff * matchedQty * contractSize * pointValue
   162→   162→    const diff = input.direction === "long"
   163→   163→      ? dSub(exitAvg, entryAvg)
   164→   164→      : dSub(entryAvg, exitAvg);
   165→   165→    const grossDecimal = dMul(dMul(dMul(diff, matchedQty), contractSize), dNormalize(String(pointValueCents / 100)));
   166→   166→    // convert decimal dollars to cents
   167→   167→    grossPnlCents = toCents(dToNumber(grossDecimal));
   168→   168→  }
   169→   169→
   170→   170→  // Costs
   171→   171→  const fees = input.feesCents ?? 0;
   172→   172→  const commission = input.commissionCents ?? 0;
   173→   173→  const swap = input.swapCents ?? 0;
   174→   174→  const slippage = input.slippageCents ?? 0;
   175→   175→  const totalCostsCents = addCents(addCents(addCents(fees, commission), swap), slippage);
   176→   176→
   177→   177→  // Net = gross - costs (costs always reduce PnL)
   178→   178→  const netPnlCents = subCents(grossPnlCents, totalCostsCents);
   179→   179→
   180→   180→  // Stop & target distances
   181→   181→  const stopDistance = (input.plannedStopPrice && input.plannedEntryPrice && !dIsZero(dSub(input.plannedEntryPrice, input.plannedStopPrice)))
   182→   182→    ? dAbs(dSub(input.plannedEntryPrice, input.plannedStopPrice))
   183→   183→    : null;
   184→   184→  const targetDistance = (input.plannedTargetPrice && input.plannedEntryPrice && !dIsZero(dSub(input.plannedTargetPrice, input.plannedEntryPrice)))
   185→   185→    ? dAbs(dSub(input.plannedTargetPrice, input.plannedEntryPrice))
   186→   186→    : null;
   187→   187→
   188→   188→  // Planned R:R
   189→   189→  const plannedRR = (stopDistance && targetDistance)
   190→   190→    ? dDiv(targetDistance, stopDistance)
   191→   191→    : null;
   192→   192→
   193→   193→  // Risk amount: stopDistance * plannedQty * contractSize * pointValue
   194→   194→  // We use entryQty as plannedQty (or matchedQty fallback)
   195→   195→  const plannedQty = entryQty;
   196→   196→  let riskAmountCents: Cents | null = null;
   197→   197→  let rewardAmountCents: Cents | null = null;
   198→   198→  if (stopDistance && !dIsZero(plannedQty)) {
   199→   199→    const riskDecimal = dMul(dMul(dMul(stopDistance, plannedQty), contractSize), dNormalize(String(pointValueCents / 100)));
   200→   200→    riskAmountCents = toCents(dToNumber(riskDecimal));
   201→   201→  }
   202→   202→  if (targetDistance && !dIsZero(plannedQty)) {
   203→   203→    const rewardDecimal = dMul(dMul(dMul(targetDistance, plannedQty), contractSize), dNormalize(String(pointValueCents / 100)));
   204→   204→    rewardAmountCents = toCents(dToNumber(rewardDecimal));
   205→   205→  }
   206→   206→
   207→   207→  // Actual R: net / risk
   208→   208→  let actualR: Decimal | null = null;
   209→   209→  if (riskAmountCents && riskAmountCents > 0) {
   210→   210→    actualR = dDiv(String(netPnlCents / 100), String(riskAmountCents / 100));
   211→   211→  }
   212→   212→
   213→   213→  // Status
   214→   214→  const status = deriveStatus(netPnlCents, entryQty, exitQty);
   215→   215→
   216→   216→  return {
   217→   217→    entryPriceAvg: entryAvg,
   218→   218→    exitPriceAvg: exitAvg,
   219→   219→    totalQuantity: plannedQty,
   220→   220→    grossPnlCents,
   221→   221→    totalCostsCents,
   222→   222→    netPnlCents,
   223→   223→    stopDistance,
   224→   224→    targetDistance,
   225→   225→    plannedRR,
   226→   226→    actualR,
   227→   227→    riskAmountCents,
   228→   228→    rewardAmountCents,
   229→   229→    status,
   230→   230→  };
   231→   231→}
   232→   232→
   233→   233→function vwap(fills: ExecutionFill[]): Decimal | null {
   234→   234→  if (fills.length === 0) return null;
   235→   235→  let pq = 0; // sum of price*qty
   236→   236→  let q = 0;  // sum of qty
   237→   237→  for (const f of fills) {
   238→   238→    const p = dToNumber(f.price);
   239→   239→    const qt = dToNumber(f.quantity);
   240→   240→    if (!Number.isFinite(p) || !Number.isFinite(qt)) continue;
   241→   241→    pq += p * qt;
   242→   242→    q += qt;
   243→   243→  }
   244→   244→  if (q === 0) return null;
   245→   245→  return dNormalize(String(+(pq / q).toFixed(10)));
   246→   246→}
   247→   247→
   248→   248→function sumQty(fills: ExecutionFill[]): Decimal {
   249→   249→  let s = 0;
   250→   250→  for (const f of fills) s += dToNumber(f.quantity);
   251→   251→  return dNormalize(String(+s.toFixed(10)));
   252→   252→}
   253→   253→
   254→   254→function deriveStatus(netPnl: Cents, entryQty: Decimal, exitQty: Decimal): TradeStatus {
   255→   255→  if (dIsZero(entryQty)) return "open";
   256→   256→  if (dIsZero(exitQty)) return "open";
   257→   257→  const cmp = dCmp(exitQty, entryQty);
   258→   258→  if (cmp < 0) {
   259→   259→    return netPnl > 0 ? "partial_win" : netPnl < 0 ? "partial_loss" : "breakeven";
   260→   260→  }
   261→   261→  if (netPnl > 0) return "win";
   262→   262→  if (netPnl < 0) return "loss";
   263→   263→  return "breakeven";
   264→   264→}

   282→   282→  positionSize: Decimal | null;

   287→   287→export function calculatePositionSize(input: PositionSizeInput): PositionSizeResult {
   288→   288→  const riskAmountCents = Math.round((input.accountBalanceCents * dToNumber(input.riskPct)) / 100) * 100 / 100;
   289→   289→  // careful: Math.round(balance * pct) gives cents directly
   290→   290→  const riskCents = Math.round((input.accountBalanceCents / 100) * dToNumber(input.riskPct) * 100);
   291→   291→  const stopDistance = dAbs(dSub(input.entryPrice, input.stopPrice));
   292→   292→  // position size = riskAmount / (stopDistance * contractSize * pointValue)
   293→   293→  let positionSize: Decimal | null = null;
   294→   294→  if (!dIsZero(stopDistance)) {
   295→   295→    const denom = dToNumber(dMul(dMul(stopDistance, input.contractSize), dNormalize(String(input.pointValueCents / 100))));
   296→   296→    if (denom > 0) {
   297→   297→      positionSize = dDiv(String(riskCents / 100), String(denom));
   298→   298→    }
   299→   299→  }
   300→   300→  return {
   301→   301→    riskAmountCents: riskCents,
   302→   302→    stopDistance,
   303→   303→    positionSize,
   304→   304→    plannedLossCents: negCents(riskCents),
   305→   305→    plannedRewardCents: null,
   306→   306→  };
   307→   307→}

   352→   352→export function computeAggregateMetrics(trades: TradeMetricRow[]): AggregateMetrics {
   353→   353→  const closed = trades.filter((t) => t.status !== "open" && t.status !== "cancelled" && t.status !== "planned");
   354→   354→  const wins = closed.filter((t) => t.netPnlCents > 0);
   355→   355→  const losses = closed.filter((t) => t.netPnlCents < 0);
   356→   356→  const bes = closed.filter((t) => t.netPnlCents === 0);
   357→   357→
   358→   358→  const totalPnl = closed.reduce((s, t) => s + t.netPnlCents, 0);
   359→   359→
   360→   360→  const winRate = closed.length > 0 ? dNormalize(String(wins.length / closed.length)) : null;
   361→   361→
   362→   362→  const avgPnl = closed.length > 0 ? Math.round(totalPnl / closed.length) : null;
   363→   363→
   364→   364→  const avgWin = wins.length > 0 ? Math.round(wins.reduce((s, t) => s + t.netPnlCents, 0) / wins.length) : null;
   365→   365→  const avgLoss = losses.length > 0 ? Math.round(losses.reduce((s, t) => s + t.netPnlCents, 0) / losses.length) : null;
   366→   366→
   367→   367→  // Avg R over trades with valid R
   368→   368→  const rValues = closed.map((t) => t.actualR).filter((r): r is Decimal => r != null && Number.isFinite(dToNumber(r)));
   369→   369→  const avgR = rValues.length > 0 ? dNormalize(String(+(rValues.reduce((s, r) => s + dToNumber(r), 0) / rValues.length).toFixed(10))) : null;
   370→   370→
   371→   371→  // Expectancy = avg realized R (per spec section 73)
   372→   372→  const expectancyR = avgR;
   373→   373→
   374→   374→  // Profit factor (spec section 72): gross profits / abs(gross losses)
   375→   375→  const grossProfit = wins.reduce((s, t) => s + t.netPnlCents, 0);
   376→   376→  const grossLoss = Math.abs(losses.reduce((s, t) => s + t.netPnlCents, 0));
   377→   377→  const profitFactor = grossLoss > 0 ? dNormalize(String(+(grossProfit / grossLoss).toFixed(10))) : null;
   378→   378→
   379→   379→  // Drawdown (spec section 71): from equity curve
   380→   380→  const sorted = [...closed].sort((a, b) => {
   381→   381→    const ad = a.entryTime ? new Date(a.entryTime).getTime() : 0;
   382→   382→    const bd = b.entryTime ? new Date(b.entryTime).getTime() : 0;
   383→   383→    return ad - bd;
   384→   384→  });
   385→   385→  let equity = 0;
   386→   386→  let peak = 0;
   387→   387→  let maxDD = 0;
   388→   388→  let currentDD = 0;
   389→   389→  let recoveryStart: number | null = null;
   390→   390→  let recoveryPeriod: number | null = null;
   391→   391→  let inDD = false;
   392→   392→  let ddStartIdx = -1;
   393→   393→  for (let i = 0; i < sorted.length; i++) {
   394→   394→    equity += sorted[i].netPnlCents;
   395→   395→    if (equity > peak) {
   396→   396→      peak = equity;
   397→   397→      if (inDD) {
   398→   398→        // recovered
   399→   399→        inDD = false;
   400→   400→        recoveryPeriod = recoveryStart != null ? i - recoveryStart : null;
   401→   401→        recoveryStart = null;
   402→   402→      }
   403→   403→    }
   404→   404→    const dd = peak - equity;
   405→   405→    if (dd > maxDD) maxDD = dd;
   406→   406→    currentDD = dd;
   407→   407→    if (dd > 0 && !inDD) {
   408→   408→      inDD = true;
   409→   409→      recoveryStart = i;
   410→   410→    }
   411→   411→  }
   412→   412→
   413→   413→  const largestWin = wins.length > 0 ? Math.max(...wins.map((t) => t.netPnlCents)) : 0;
   414→   414→  const largestLoss = losses.length > 0 ? Math.min(...losses.map((t) => t.netPnlCents)) : 0;
   415→   415→
   416→   416→  // Streaks
   417→   417→  let cw = 0, cl = 0, maxCw = 0, maxCl = 0;
   418→   418→  for (const t of sorted) {
   419→   419→    if (t.netPnlCents > 0) { cw++; cl = 0; maxCw = Math.max(maxCw, cw); }
   420→   420→    else if (t.netPnlCents < 0) { cl++; cw = 0; maxCl = Math.max(maxCl, cl); }
   421→   421→    else { cw = 0; cl = 0; }
   422→   422→  }
   423→   423→
   424→   424→  return {
   425→   425→    totalTrades: trades.length,
   426→   426→    closedTrades: closed.length,
   427→   427→    wins: wins.length,
   428→   428→    losses: losses.length,
   429→   429→    breakevens: bes.length,
   430→   430→    winRate,
   431→   431→    totalPnlCents: totalPnl,
   432→   432→    avgPnlCents: avgPnl,
   433→   433→    avgWinCents: avgWin,
   434→   434→    avgLossCents: avgLoss,
   435→   435→    avgR,

   514→   514→export interface RDistributionBucket {
   515→   515→  bucket: string;
   516→   516→  count: number;
   517→   517→  minR: number;
   518→   518→  maxR: number;
   519→   519→}
   520→   520→
   521→   521→export function buildRDistribution(trades: TradeMetricRow[]): RDistributionBucket[] {
   522→   522→  const buckets: { min: number; max: number; label: string }[] = [
   523→   523→    { min: -Infinity, max: -2, label: "≤ -2R" },
   524→   524→    { min: -2, max: -1, label: "-2 to -1R" },
   525→   525→    { min: -1, max: 0, label: "-1 to 0R" },
   526→   526→    { min: 0, max: 0.0001, label: "BE" },
   527→   527→    { min: 0.0001, max: 1, label: "0 to +1R" },
   528→   528→    { min: 1, max: 2, label: "+1 to +2R" },
   529→   529→    { min: 2, max: Infinity, label: "≥ +2R" },
   530→   530→  ];
   531→   531→  return buckets.map((b) => {
   532→   532→    const count = trades.filter((t) => {
   533→   533→      if (t.actualR == null) return false;
   534→   534→      const r = dToNumber(t.actualR);
   535→   535→      if (b.min === 0 && b.max === 0.0001) return r === 0;
   536→   536→      if (b.min === -Infinity) return r <= b.max;
   537→   537→      if (b.max === Infinity) return r > b.min;
   538→   538→      return r > b.min && r <= b.max;
   539→   539→    }).length;
   540→   540→    return { bucket: b.label, count, minR: b.min, maxR: b.max };
   541→   541→  });
   542→   542→}

   548→   548→export interface GroupBreakdown {
   549→   549→  key: string;
   550→   550→  label: string;
   551→   551→  trades: number;
   552→   552→  wins: number;
   553→   553→  winRate: Decimal | null;
   554→   554→  avgR: Decimal | null;
   555→   555→  expectancyR: Decimal | null;
   556→   556→  pnlCents: Cents;
   557→   557→}
   558→   558→
   559→   559→export function groupBy<T extends TradeMetricRow>(
   560→   560→  trades: T[],
   561→   561→  keyFn: (t: T) => string | null | undefined,
   562→   562→  labelFn: (k: string) => string = (k) => k,
   563→   563→): GroupBreakdown[] {
   564→   564→  const groups = new Map<string, T[]>();
   565→   565→  for (const t of trades) {
   566→   566→    const k = keyFn(t);
   567→   567→    if (k == null) continue;
   568→   568→    const arr = groups.get(k) ?? [];
   569→   569→    arr.push(t);
   570→   570→    groups.set(k, arr);
   571→   571→  }
   572→   572→  return [...groups.entries()].map(([key, arr]) => {
   573→   573→    const m = computeAggregateMetrics(arr);
   574→   574→    return {
   575→   575→      key,
   576→   576→      label: labelFn(key),
   577→   577→      trades: m.totalTrades,
   578→   578→      wins: m.wins,
   579→   579→      winRate: m.winRate,
   580→   580→      avgR: m.avgR,
   581→   581→      expectancyR: m.expectancyR,
   582→   582→      pnlCents: m.totalPnlCents,
   583→   583→    };
   584→   584→  }).sort((a, b) => b.trades - a.trades);
   585→   585→}

   591→   591→export interface Insight {
   592→   592→  id: string;
   593→   593→  title: string;
   594→   594→  body: string;
   595→   595→  sampleSize: number;
   596→   596→  metric: string;
   597→   597→  bucketLabel: string;
   598→   598→  comparison?: string;
   599→   599→  tradeIds: string[];
   600→   600→  warning?: "low_sample" | "missing_data";
   601→   601→}
   602→   602→
   603→   603→export function generateInsights(trades: TradeMetricRow[], minSample = 10): Insight[] {
   604→   604→  const insights: Insight[] = [];
   605→   605→  const closed = trades.filter((t) => t.status !== "open" && t.status !== "cancelled" && t.status !== "planned");
   606→   606→  if (closed.length < minSample) {
   607→   607→    if (closed.length > 0) {
   608→   608→      insights.push({
   609→   609→        id: "low-sample",
   610→   610→        title: "Not enough data yet",
   611→   611→        body: `You have ${closed.length} closed trade${closed.length === 1 ? "" : "s"}. DnD starts generating statistical insights once you have at least ${minSample} closed trades, so patterns you see now may not be reliable.`,
   612→   612→        sampleSize: closed.length,
   613→   613→        metric: "samples",
   614→   614→        bucketLabel: "all",
   615→   615→        tradeIds: closed.map((t) => t.id),
   616→   616→        warning: "low_sample",
   617→   617→      });
   618→   618→    }
   619→   619→    return insights;
   620→   620→  }
   621→   621→
   622→   622→  // Behavior flag insights
   623→   623→  const allFlags = new Set<string>();
   624→   624→  closed.forEach((t) => t.behaviorFlags?.forEach((f) => allFlags.add(f)));
   625→   625→  for (const flag of allFlags) {
   626→   626→    const withFlag = closed.filter((t) => t.behaviorFlags?.includes(flag));
   627→   627→    const withoutFlag = closed.filter((t) => !t.behaviorFlags?.includes(flag));
   628→   628→    if (withFlag.length < minSample) continue;
   629→   629→    const withR = withFlag.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   630→   630→    const withoutR = withoutFlag.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   631→   631→    if (withR.length === 0) continue;
   632→   632→    const withAvg = withR.reduce((s, r) => s + dToNumber(r), 0) / withR.length;
   633→   633→    const withoutAvg = withoutR.length > 0 ? withoutR.reduce((s, r) => s + dToNumber(r), 0) / withoutR.length : null;
   634→   634→    const flagLabel = humanizeFlag(flag);
   635→   635→    if (withoutAvg != null && Math.abs(withAvg - withoutAvg) > 0.05) {
   636→   636→      const direction = withAvg < withoutAvg ? "lower" : "higher";
   637→   637→      insights.push({
   638→   638→        id: `flag-${flag}`,
   639→   639→        title: `Trades tagged "${flagLabel}"`,
   640→   640→        body: `Trades tagged "${flagLabel}" have historically produced ${direction} average R (${withAvg.toFixed(2)}R across ${withFlag.length} trades) compared to ${withoutAvg.toFixed(2)}R across ${withoutFlag.length} trades without that flag.`,
   641→   641→        sampleSize: withFlag.length,
   642→   642→        metric: "avgR",
   643→   643→        bucketLabel: flagLabel,
   644→   644→        comparison: `vs ${withoutAvg.toFixed(2)}R`,
   645→   645→        tradeIds: withFlag.map((t) => t.id),
   646→   646→      });
   647→   647→    } else {
   648→   648→      insights.push({
   649→   649→        id: `flag-${flag}`,
   650→   650→        title: `Trades tagged "${flagLabel}"`,
   651→   651→        body: `Trades tagged "${flagLabel}" have an average realized R of ${withAvg.toFixed(2)}R across ${withFlag.length} trades.`,
   652→   652→        sampleSize: withFlag.length,
   653→   653→        metric: "avgR",
   654→   654→        bucketLabel: flagLabel,
   655→   655→        tradeIds: withFlag.map((t) => t.id),
   656→   656→      });
   657→   657→    }
   658→   658→  }
   659→   659→
   660→   660→  // Session insights
   661→   661→  const sessions = groupBy(closed, (t) => t.session, (k) => sessionLabel(k));
   662→   662→  for (const s of sessions) {
   663→   663→    if (s.trades < minSample) continue;
   664→   664→    if (s.avgR == null) continue;
   665→   665→    insights.push({
   666→   666→      id: `session-${s.key}`,
   667→   667→      title: `${s.label} session`,
   668→   668→      body: `${s.label} trades have produced an average of ${dToNumber(s.avgR).toFixed(2)}R across ${s.trades} trades, with a ${dToNumber(s.winRate).toFixed(1)}% win rate.`,
   669→   669→      sampleSize: s.trades,
   670→   670→      metric: "avgR",
   671→   671→      bucketLabel: s.label,
   672→   672→      tradeIds: closed.filter((t) => t.session === s.key).map((t) => t.id),
   673→   673→    });
   674→   674→  }
   675→   675→
   676→   676→  // A+ vs non-A+
   677→   677→  const aplus = closed.filter((t) => t.setupGrade === "A+");
   678→   678→  const nonAplus = closed.filter((t) => t.setupGrade && t.setupGrade !== "A+" && t.setupGrade !== "Invalid");
   679→   679→  if (aplus.length >= minSample && nonAplus.length >= minSample) {
   680→   680→    const aR = aplus.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   681→   681→    const nR = nonAplus.map((t) => t.actualR).filter((r): r is Decimal => r != null);
   682→   682→    if (aR.length > 0 && nR.length > 0) {
   683→   683→      const aAvg = aR.reduce((s, r) => s + dToNumber(r), 0) / aR.length;
   684→   684→      const nAvg = nR.reduce((s, r) => s + dToNumber(r), 0) / nR.length;
   685→   685→      insights.push({
   686→   686→        id: "aplus-vs-non",
   687→   687→        title: "A+ vs non-A+ setups",
   688→   688→        body: `A+ setups have produced ${aAvg.toFixed(2)}R on average across ${aplus.length} trades, compared to ${nAvg.toFixed(2)}R across ${nonAplus.length} non-A+ trades.`,
   689→   689→        sampleSize: aplus.length + nonAplus.length,
   690→   690→        metric: "avgR",
   691→   691→        bucketLabel: "A+ vs non-A+",
   692→   692→        tradeIds: [...aplus, ...nonAplus].map((t) => t.id),
   693→   693→      });
   694→   694→    }
   695→   695→  }
   696→   696→
   697→   697→  return insights.sort((a, b) => b.sampleSize - a.sampleSize);
   698→   698→}

   718→   718→// ---------------------------------------------------------------------------
   719→   719→// Plan adherence (spec section 23 — planned vs actual execution)
   720→   720→// ---------------------------------------------------------------------------
   721→   721→
   722→   722→/**
   723→   723→ * Result of comparing a trade's planned values against its actual execution.
   724→   724→ *
   725→   725→ * `adherenceScore` is a 0–1 number representing the fraction of binary checks
   726→   726→ * (session match, target hit, risk-within-tolerance, positive R achieved) that
   727→   727→ * passed. Checks that can't be evaluated (e.g. no planned session recorded, no
   728→   728→ * target price, no account balance) are excluded from the denominator — so a
   729→   729→ * trade with only one check evaluated and that check passed scores 1.0, not
   730→   730→ * 0.25. If no checks can be evaluated at all, the score is 0.
   731→   731→ */
   732→   732→export interface PlanAdherenceCheck {
   733→   733→  label: string;
   734→   734→  passed: boolean | null;
   735→   735→  detail?: string;
   736→   736→}
   737→   737→
   738→   738→export interface PlanAdherence {
   739→   739→  plannedRR: Decimal | null;
   740→   740→  actualRR: Decimal | null;
   741→   741→  plannedRiskPct: Decimal | null;
   742→   742→  actualRiskPct: Decimal | null;
   743→   743→  sessionMatch: boolean | null;
   744→   744→  targetMatch: boolean | null;
   745→   745→  adherenceScore: number;
   746→   746→  adherencePct: number;
   747→   747→  checks: PlanAdherenceCheck[];
   748→   748→}
   749→   749→
   750→   750→/**
   751→   751→ * Trade-shape accepted by `calculatePlanAdherence`. Accepts the loose shape
   752→   752→ * produced by Prisma's `Trade` row (decimal strings + nullable fields) plus an
   753→   753→ * optional `accountBalanceCents` so we can derive `actualRiskPct` from the
   754→   754→ * risk amount vs the account balance.
   755→   755→ */
   756→   756→export interface PlanAdherenceTrade {
   757→   757→  direction?: string | null;
   758→   758→  /** Actual session the trade was taken in (from `Trade.session`). */
   759→   759→  session?: string | null;
   760→   760→  /** Planned session (from `setupJson.session` or a daily plan). */
   761→   761→  plannedSession?: string | null;
   762→   762→  plannedEntryPrice?: Decimal | null;
   763→   763→  plannedStopPrice?: Decimal | null;
   764→   764→  plannedTargetPrice?: Decimal | null;
   765→   765→  plannedRiskPct?: Decimal | null;
   766→   766→  entryPriceAvg?: Decimal | null;
   767→   767→  exitPriceAvg?: Decimal | null;
   768→   768→  /** Pre-computed actual R (decimal string). */
   769→   769→  actualR?: Decimal | null;
   770→   770→  /** Risk amount in cents — typically from `calculateTradePnl.riskAmountCents`. */
   771→   771→  riskAmountCents?: number | null;
   772→   772→  /** Account balance in cents — used to derive `actualRiskPct`. */
   773→   773→  accountBalanceCents?: number | null;
   774→   774→}
   775→   775→
   776→   776→/**
   777→   777→ * Compute plan adherence for a single trade. Pure function — no I/O.
   778→   778→ *
   779→   779→ * Binary checks (each 0/1 or null when undeterminable):
   780→   780→ *   - `sessionMatch`: planned session === actual session
   781→   781→ *   - `targetMatch`: |exitPrice - target| / |target| <= 5% (within 5% of target)
   782→   782→ *   - `riskWithinTolerance`: actual risk is within 50% of planned risk
   783→   783→ *     (ratio of actual/planned between 0.5 and 1.5)
   784→   784→ *   - `rrAchieved`: actualR > 0 (the trade was profitable)
   785→   785→ *
   786→   786→ * `adherenceScore` = mean of the non-null checks (or 0 if all null).
   787→   787→ */
   788→   788→export function calculatePlanAdherence(trade: PlanAdherenceTrade): PlanAdherence {
   789→   789→  // Planned R:R = |target - entry| / |entry - stop|
   790→   790→  let plannedRR: Decimal | null = null;
   791→   791→  if (trade.plannedEntryPrice && trade.plannedStopPrice && trade.plannedTargetPrice) {
   792→   792→    const stopDist = dAbs(dSub(trade.plannedEntryPrice, trade.plannedStopPrice));
   793→   793→    const targetDist = dAbs(dSub(trade.plannedTargetPrice, trade.plannedEntryPrice));
   794→   794→    if (!dIsZero(stopDist)) {
   795→   795→      plannedRR = dDiv(targetDist, stopDist);
   796→   796→    }
   797→   797→  }
   798→   798→
   799→   799→  // Actual R: already computed server-side — normalize and pass through.
   800→   800→  const actualRR: Decimal | null =
   801→   801→    trade.actualR != null && trade.actualR !== "" ? dNormalize(trade.actualR) : null;
   802→   802→
   803→   803→  // Planned risk %
   804→   804→  const plannedRiskPct: Decimal | null =
   805→   805→    trade.plannedRiskPct != null && trade.plannedRiskPct !== ""
   806→   806→      ? dNormalize(trade.plannedRiskPct)
   807→   807→      : null;
   808→   808→
   809→   809→  // Actual risk %: riskAmount / accountBalance (if both available and balance > 0)
   810→   810→  let actualRiskPct: Decimal | null = null;
   811→   811→  if (
   812→   812→    trade.riskAmountCents != null &&
   813→   813→    trade.accountBalanceCents != null &&
   814→   814→    trade.accountBalanceCents > 0 &&
   815→   815→    trade.riskAmountCents >= 0
   816→   816→  ) {
   817→   817→    actualRiskPct = dDiv(
   818→   818→      String(trade.riskAmountCents / 100),
   819→   819→      String(trade.accountBalanceCents / 100),
   820→   820→    );
   821→   821→  }
   822→   822→
   823→   823→  // Session match — null when either planned or actual session is missing.
   824→   824→  const plannedSession = trade.plannedSession ?? null;
   825→   825→  const actualSession = trade.session ?? null;
   826→   826→  const sessionMatch: boolean | null =
   827→   827→    plannedSession && actualSession ? plannedSession === actualSession : null;
   828→   828→
   829→   829→  // Target match — exit price within 5% of planned target.
   830→   830→  let targetMatch: boolean | null = null;
   831→   831→  if (trade.exitPriceAvg && trade.plannedTargetPrice) {
   832→   832→    const exit = dToNumber(trade.exitPriceAvg);
   833→   833→    const target = dToNumber(trade.plannedTargetPrice);
   834→   834→    if (Number.isFinite(exit) && Number.isFinite(target) && target !== 0) {
   835→   835→      const deviation = Math.abs(exit - target) / Math.abs(target);
   836→   836→      targetMatch = deviation <= 0.05;
   837→   837→    }
   838→   838→  }
   839→   839→
   840→   840→  // Risk-within-tolerance — actual risk is within 50% of planned risk
   841→   841→  // (i.e. ratio of actual/planned is between 0.5 and 1.5).
   842→   842→  let riskWithinTolerance: boolean | null = null;
   843→   843→  if (plannedRiskPct && actualRiskPct) {
   844→   844→    const planned = dToNumber(plannedRiskPct);
   845→   845→    const actual = dToNumber(actualRiskPct);
   846→   846→    if (planned > 0) {
   847→   847→      const ratio = actual / planned;
   848→   848→      riskWithinTolerance = ratio >= 0.5 && ratio <= 1.5;
   849→   849→    }
   850→   850→  }
   851→   851→
   852→   852→  // R:R achieved — actualR > 0 (i.e. the trade was profitable).
   853→   853→  const rrAchieved: boolean | null =
   854→   854→    actualRR != null ? dToNumber(actualRR) > 0 : null;
   855→   855→
   856→   856→  const checks: PlanAdherenceCheck[] = [
   857→   857→    {
   858→   858→      label: "Session match",
   859→   859→      passed: sessionMatch,
   860→   860→      detail:
   861→   861→        sessionMatch === null
   862→   862→          ? "No planned session recorded"
   863→   863→          : sessionMatch
   864→   864→            ? "Traded in the planned session"
   865→   865→            : "Traded outside the planned session",
   866→   866→    },
   867→   867→    {
   868→   868→      label: "Target hit",
   869→   869→      passed: targetMatch,
   870→   870→      detail:

   899→   899→  // Adherence score: average of non-null binary checks. Each check contributes

   901→   901→  const evaluatedChecks = checks

   904→   904→  const adherenceScore =

   907→   907→      : 0;
   908→   908→  const adherencePct = adherenceScore * 100;
