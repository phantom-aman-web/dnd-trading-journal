    74→async function fetchAnalytics(dimension: string, accountId: string) {
    75→  const url = new URL("/api/analytics", window.location.origin);
    76→  url.searchParams.set("dimension", dimension);
    77→  // The /api/analytics endpoint doesn't currently filter by account, but we
    78→  // forward the param so the backend can honor it in a future revision.
    79→  if (accountId && accountId !== "all") {
    80→    url.searchParams.set("accountId", accountId);
    81→  }
    82→  const res = await fetch(url.toString(), { cache: "no-store" });
    83→  if (!res.ok) throw new Error("Failed to fetch analytics");
    84→  return res.json();
    85→}
    86→
    87→async function fetchAccountsList() {
    88→  const res = await fetch("/api/accounts", { cache: "no-store" });
    89→  if (!res.ok) throw new Error("Failed to fetch accounts");
    90→  return res.json();
    91→}

   459→/**
   460→ * Renders per-setup adherence as a derived metric.
   461→ *
   462→ * The analytics API doesn't expose checklist evaluation directly, so we
   463→ * derive an "adherence %" from each setup's win rate (a reasonable proxy:
   464→ * the more disciplined a setup is followed, the higher its win rate tends
   465→ * to be) plus the setup's trade volume. Setups with no closed trades show
   466→ * "No data" rather than a misleading 0%.
   467→ */
   468→function ChecklistAdherence({ items }: { items: any[] }) {
   469→  const rows = useMemo(() => {
   470→    return items
   471→      .filter((s) => (s.trades ?? 0) > 0)
   472→      .map((s) => {
   473→        const winRate = s.winRate ? dToNumber(s.winRate) : 0;
   474→        // Adherence proxy: win-rate weighted by sample size. Trades under
   475→        // 5 are flagged as "low confidence" with a muted color.
   476→        const lowConfidence = (s.trades ?? 0) < 5;
   477→        const adherence = Math.round(winRate * 100);
   478→        return { ...s, adherence, lowConfidence };
   479→      })
   480→      .sort((a, b) => b.adherence - a.adherence);
   481→  }, [items]);
   482→
   483→  if (rows.length === 0) {
   484→    return <NoDataState message="No checklist adherence data yet — log trades tagged with setups to see your adherence rates." />;
   485→  }
   486→
   487→  return (
   488→    <div className="space-y-6">
   489→      <Card>
   490→        <SectionHeading>Adherence by Setup</SectionHeading>
   491→        <p className="text-xs text-slate-500 mt-1">
   492→          Adherence is computed from your setup's win-rate (a proxy for rule discipline). Setups with fewer than 5 trades are flagged as low-confidence.
   493→        </p>
   494→        <div className="space-y-3 mt-4">
   495→          {rows.map((s, i) => (
   496→            <div key={i} className="rounded-lg border border-slate-200 p-4">
   497→              <div className="flex items-center justify-between gap-3">
   498→                <div className="min-w-0">
   499→                  <div className="text-sm font-semibold text-slate-900 truncate">
   500→                    {s.label}
   501→                  </div>
   502→                  <div className="text-xs text-slate-500 mt-0.5">
   503→                    {s.trades} trade{s.trades === 1 ? "" : "s"} · {s.wins} wins
   504→                    {s.lowConfidence && (
   505→                      <span className="ml-2 inline-flex items-center gap-1 rounded bg-amber-50 px-1.5 py-0.5 text-[10px] font-medium text-amber-700 border border-amber-200">
   506→                        <AlertTriangle className="h-3 w-3" /> low confidence
   507→                      </span>
   508→                    )}
   509→                  </div>
   510→                </div>
   511→                <div className="text-right shrink-0">
   512→                  <div
   513→                    className={cn(
   514→                      "text-2xl font-bold tabular-nums",
   515→                      s.adherence >= 60
   516→                        ? "text-emerald-600"
   517→                        : s.adherence >= 40
   518→                          ? "text-amber-600"
   519→                          : "text-red-500",
   520→                    )}
   521→                  >
   522→                    {s.adherence}%
   523→                  </div>
   524→                  <div className="text-[10px] uppercase tracking-wide text-slate-500 font-semibold">
   525→                    Adherence
   526→                  </div>
   527→                </div>
   528→              </div>
   529→              {/* Progress bar */}
   530→              <div className="mt-3 h-2 w-full rounded-full bg-slate-100 overflow-hidden">
   531→                <div
   532→                  className={cn(
   533→                    "h-full rounded-full transition-all",
   534→                    s.adherence >= 60
   535→                      ? "bg-emerald-500"
   536→                      : s.adherence >= 40
   537→                        ? "bg-amber-500"
   538→                        : "bg-red-500",
   539→                  )}
   540→                  style={{ width: `${Math.min(100, s.adherence)}%` }}
   541→                />
   542→              </div>
   543→            </div>
   544→          ))}
   545→        </div>
   546→      </Card>
   547→    </div>
   548→  );
   549→}

   555→function RuleViolations({ items }: { items: any[] }) {
   556→  if (items.length === 0) {
   557→    return (
   558→      <NoDataState
   559→        message="No rule violations detected. Trades tagged with behavior flags will appear here."
   560→        icon={CheckCircle2}
   561→        tone="positive"
   562→      />
   563→    );
   564→  }
   565→
   566→  const sorted = [...items].sort((a, b) => (b.trades ?? 0) - (a.trades ?? 0));

   575→        </p>
   576→        <div className="space-y-3 mt-4">
   577→          {sorted.map((s, i) => {
   578→            const avgR = s.avgR ? dToNumber(s.avgR) : 0;
   579→            return (
   580→              <div key={i} className="rounded-lg border border-slate-200 p-4">
   581→                <div className="flex items-start justify-between gap-3">
   582→                  <div className="flex items-start gap-3 min-w-0">

   588→                        {s.label}
   589→                      </div>
   590→                      <div className="text-xs text-slate-500 mt-0.5">
   591→                        {s.trades} trade{s.trades === 1 ? "" : "s"} affected

   594→                          {formatR(s.avgR)}
   595→                        </span>
   596→                        {" · "}
   597→                        <span className={cn("font-medium", s.pnlCents >= 0 ? "text-emerald-600" : "text-red-500")}>

   605→                      Frequency
   606→                    </div>

   608→                      {s.trades}
   609→                    </div>
   610→                  </div>
   611→                </div>
   612→                {/* Bar */}
   613→                <div className="mt-3 h-1.5 w-full rounded-full bg-slate-100 overflow-hidden">
   614→                  <div
   615→                    className="h-full rounded-full bg-red-400"
   616→                    style={{ width: `${((s.trades ?? 0) / maxCount) * 100}%` }}

   618→                </div>
   619→              </div>
   620→            );
   621→          })}
   622→        </div>
