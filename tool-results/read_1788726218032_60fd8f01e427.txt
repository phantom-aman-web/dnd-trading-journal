   296→  function setField(path: string, value: any) {
   297→    setForm((f: any) => {
   298→      const next = { ...f };
   299→      const parts = path.split(".");
   300→      let cur: any = next;
   301→      for (let i = 0; i < parts.length - 1; i++) {
   302→        cur[parts[i]] = { ...cur[parts[i]] };
   303→        cur = cur[parts[i]];
   304→      }
   305→      cur[parts[parts.length - 1]] = value;
   306→      return next;
   307→    });
   308→  }

   310→  function toggleArrayField(path: string, value: string) {
   311→    setForm((f: any) => {
   312→      const next = { ...f };
   313→      const parts = path.split(".");
   314→      let cur: any = next;
   315→      for (let i = 0; i < parts.length - 1; i++) {
   316→        cur[parts[i]] = { ...cur[parts[i]] };
   317→        cur = cur[parts[i]];
   318→      }
   319→      const key = parts[parts.length - 1];
   320→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   321→      const idx = arr.indexOf(value);
   322→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   323→      cur[key] = arr;
   324→      return next;
   325→    });
   326→  }

   328→  // Update one A+ checklist answer field (`checked` or `note`) for the given
   329→  // item id. Used by the checkbox + evidence-note inputs in the ICT Setup tab.
   330→  function updateChecklistAnswer(
   331→    itemId: string,
   332→    field: "checked" | "note",
   333→    value: boolean | string,
   334→  ) {
   335→    setForm((f: any) => {
   336→      const current = f.checklistAnswers?.[itemId] ?? { checked: false };
   337→      const next = {
   338→        ...f.checklistAnswers,
   339→        [itemId]: {
   340→          ...current,
   341→          [field]: value,
   342→        },
   343→      };
   344→      // If a checked item is being unchecked, drop its note so storage stays clean.
   345→      if (field === "checked" && value === false && next[itemId].note) {
   346→        delete next[itemId].note;
   347→      }
   348→      return { ...f, checklistAnswers: next };
   349→    });

   390→        checklistVersionId: form.checklistVersionId || null,
   391→        checklistAnswers: form.checklistVersionId ? form.checklistAnswers : null,

   402→        swapCents: Number(form.swapCents) || 0,
   403→        slippageCents: Number(form.slippageCents) || 0,
   404→        entryTime: form.entryTime ? new Date(form.entryTime).toISOString() : null,
   405→        exitTime: form.exitTime ? new Date(form.exitTime).toISOString() : null,
   406→        tradingTimezone: form.tradingTimezone,
   407→        setup: form.setup,
   408→        thesis: form.thesis,
   409→        notes: form.notes || null,

   412→        psychAfter: form.psychAfter,

   415→        isDraft,
   416→      };
   417→      const url = editingId ? `/api/trades/${editingId}` : "/api/trades";
   418→      const method = editingId ? "PATCH" : "POST";
   419→      const res = await fetch(url, {
   420→        method,
   421→        headers: { "Content-Type": "application/json" },
   422→        body: JSON.stringify(payload),
   423→      });

   439→                method: "PATCH",
   440→                headers: { "Content-Type": "application/json" },
   441→                body: JSON.stringify(patchBody),
   442→              }).catch(() => {});
   443→            }
   444→          }
   445→        }
   446→      }
   447→      toast.success(isDraft ? "Draft saved." : "Trade saved.");
   448→      localStorage.removeItem(DRAFT_KEY);
   449→      qc.invalidateQueries({ queryKey: ["trades"] });
   450→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   451→      navigate("tradeDetail", { id: data.id });
