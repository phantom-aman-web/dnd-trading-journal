   210→  function setField(path: string, value: any) {
   211→    setForm((f: any) => {
   212→      const next = { ...f };
   213→      const parts = path.split(".");
   214→      let cur: any = next;
   215→      for (let i = 0; i < parts.length - 1; i++) {
   216→        cur[parts[i]] = { ...cur[parts[i]] };
   217→        cur = cur[parts[i]];
   218→      }
   219→      cur[parts[parts.length - 1]] = value;
   220→      return next;
   221→    });
   222→  }

   224→  function toggleArrayField(path: string, value: string) {
   225→    setForm((f: any) => {
   226→      const next = { ...f };
   227→      const parts = path.split(".");
   228→      let cur: any = next;
   229→      for (let i = 0; i < parts.length - 1; i++) {
   230→        cur[parts[i]] = { ...cur[parts[i]] };
   231→        cur = cur[parts[i]];
   232→      }
   233→      const key = parts[parts.length - 1];
   234→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   235→      const idx = arr.indexOf(value);
   236→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   237→      cur[key] = arr;
   238→      return next;
   239→    });
   240→  }
