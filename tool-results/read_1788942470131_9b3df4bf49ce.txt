   478→   478→  function setField(path: string, value: any) {
   479→   479→    setForm((f: any) => {
   480→   480→      const next = { ...f };
   481→   481→      const parts = path.split(".");
   482→   482→      let cur: any = next;
   483→   483→      for (let i = 0; i < parts.length - 1; i++) {
   484→   484→        cur[parts[i]] = { ...cur[parts[i]] };
   485→   485→        cur = cur[parts[i]];
   486→   486→      }
   487→   487→      cur[parts[parts.length - 1]] = value;
   488→   488→      return next;
   489→   489→    });

   492→   492→  function toggleArrayField(path: string, value: string) {
   493→   493→    setForm((f: any) => {
   494→   494→      const next = { ...f };
   495→   495→      const parts = path.split(".");
   496→   496→      let cur: any = next;
   497→   497→      for (let i = 0; i < parts.length - 1; i++) {
   498→   498→        cur[parts[i]] = { ...cur[parts[i]] };
   499→   499→        cur = cur[parts[i]];
   500→   500→      }
   501→   501→      const key = parts[parts.length - 1];
   502→   502→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   503→   503→      const idx = arr.indexOf(value);
   504→   504→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   505→   505→      cur[key] = arr;
   506→   506→      return next;
   507→   507→    });
   508→   508→  }
