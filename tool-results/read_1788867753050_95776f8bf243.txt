    37→async function fetchSettings() {
    38→  const res = await fetch("/api/settings", { cache: "no-store" });
    39→  return res.json();
    40→}
    41→async function fetchAccounts() {
    42→  const res = await fetch("/api/accounts", { cache: "no-store" });
    43→  return res.json();
    44→}
    45→async function fetchInstruments() {
    46→  const res = await fetch("/api/instruments", { cache: "no-store" });
    47→  return res.json();
    48→}
    49→async function fetchBackups() {
    50→  const res = await fetch("/api/backups", { cache: "no-store" });
    51→  return res.json();

   142→          <TabsTrigger value="instruments">Instruments</TabsTrigger>
   143→          <TabsTrigger value="trading">Trading</TabsTrigger>
   144→          <TabsTrigger value="appearance">Appearance</TabsTrigger>

   174→              <h2 className="font-medium">Trading Accounts</h2>

   176→            </div>

   178→              {(accountsData?.items ?? []).map((a: any) => (

   283→                    <SelectItem value="london">London</SelectItem>
   284→                    <SelectItem value="ny_am">New York AM</SelectItem>
   285→                    <SelectItem value="ny_pm">New York PM</SelectItem>

   305→                DnD flags a trade's planned risk against these percentage thresholds.
   306→                Use them to keep your sizing honest at a glance.
   307→              </p>
   308→            </div>
   309→            <div className="grid sm:grid-cols-3 gap-3">
   310→              <div className="space-y-2">
   311→                <FieldLabel hint="Up to this % is normal">Normal Risk Max %</FieldLabel>

   322→            </div>
   323→            <div className="flex flex-wrap items-center gap-2 text-[11px]">
   324→              <span className="rounded border border-border px-2 py-0.5 text-profit">Normal ≤ {normalRiskMaxPct}%</span>

   328→            <Button onClick={saveSettings}>Save</Button>
   329→          </Card>
   330→        </TabsContent>
   331→
   332→        {/* Appearance */}
   333→        <TabsContent value="appearance">
   334→          <Card className="p-4 space-y-4">
   335→            <div className="space-y-2">
   336→              <FieldLabel required>Theme</FieldLabel>
   337→              <div className="grid grid-cols-3 gap-2">
   338→                {[
   339→                  { key: "nordic", label: "Nordic Clean", desc: "Light, quiet, precise" },
   340→                  { key: "terminal", label: "Sleek Terminal", desc: "Dark, technical" },
   341→                  { key: "institutional", label: "Institutional", desc: "Premium, analytical" },
   342→                ].map((t) => (
   343→                  <button
   344→                    key={t.key}
   345→                    onClick={() => {
   346→                      applyTheme(t.key as any);

   349→                      "p-3 rounded-md border text-left transition-colors",

   369→                </Select>
   370→              </div>
   371→            </div>
   372→            <div className="flex items-center justify-between">

   398→                  When enabled, sensitive fields (private journal notes, psychology notes, strategy notes, review notes)

   400→                  Metadata remains available server-side for analytics and search.
   401→                </p>
   402→              </div>

   800→function ImportCsvDialog({ onImported }: { onImported: () => void }) {
   801→  const [open, setOpen] = useState(false);
   802→  const [step, setStep] = useState(1);
   803→  const [filename, setFilename] = useState("");
   804→  const [content, setContent] = useState("");
   805→  const [importId, setImportId] = useState<string | null>(null);
   806→  const [headers, setHeaders] = useState<string[]>([]);
   807→  const [sampleRows, setSampleRows] = useState<Record<string, string>[]>([]);
   808→  const [mapping, setMapping] = useState<Record<string, string>>({});
   809→  const [validation, setValidation] = useState<any>(null);
   810→
   811→  const FIELDS = ["instrumentSymbol", "direction", "entryPrice", "exitPrice", "stopPrice", "targetPrice", "quantity", "entryTime", "exitTime", "session", "setupGrade", "feesCents", "commissionCents", "swapCents"];
   812→
   813→  async function upload() {
   814→    const res = await fetch("/api/imports", {
   815→      method: "POST",
   816→      headers: { "Content-Type": "application/json" },
   817→      body: JSON.stringify({ filename, source: "csv", content }),
   818→    });
   819→    if (!res.ok) { toast.error("Upload failed."); return; }
   820→    const data = await res.json();
   821→    setImportId(data.id);
   822→    const summary = JSON.parse(data.summaryJson ?? "{}");
   823→    setHeaders(summary.headers ?? []);
   824→    setSampleRows(summary.sampleRows ?? []);
   825→    setStep(2);
   826→  }
   827→
   828→  async function validateImport() {
   829→    const res = await fetch(`/api/imports/${importId}`, {
   830→      method: "PATCH",
   831→      headers: { "Content-Type": "application/json" },
   832→      body: JSON.stringify({ action: "map", mapping }),
   833→    });
   834→    if (!res.ok) { toast.error("Validation failed."); return; }
   835→    const data = await res.json();
   836→    setValidation(data);
   837→    setStep(3);
   838→  }
   839→
   840→  async function confirmImport() {
   841→    const res = await fetch(`/api/imports/${importId}`, {
   842→      method: "PATCH",
   843→      headers: { "Content-Type": "application/json" },
   844→      body: JSON.stringify({ action: "confirm" }),
   845→    });
   846→    if (!res.ok) { toast.error("Import failed."); return; }
   847→    const data = await res.json();
   848→    toast.success(`Imported ${data.imported} trades.`);
   849→    setOpen(false);
   850→    setStep(1);
   851→    onImported();
   852→  }

   874→              A header row is required.

   888→                    value={mapping[h] ?? ""}

   890→                  >
   891→                    <SelectTrigger className="h-7 text-xs"><SelectValue /></SelectTrigger>
   892→                    <SelectContent>
   893→                      {FIELDS.map((f) => <SelectItem key={f} value={f}>{f}</SelectItem>)}
   894→                    </SelectContent>
   895→                  </Select>
   896→                </div>
   897→              ))}
   898→            </div>
   899→            <Button onClick={validateImport}>Validate</Button>

   902→        {step === 3 && validation && (

   905→              <div className="rounded-md border border-border p-3">

   907→                <div className="text-xs text-muted-foreground">Valid</div>
   908→              </div>

   910→                <div className="text-2xl font-semibold text-warning tnum">{validation.duplicates}</div>
   911→                <div className="text-xs text-muted-foreground">Duplicates (skipped)</div>
   912→              </div>
   913→              <div className="rounded-md border border-border p-3">
   914→                <div className="text-2xl font-semibold text-loss tnum">{validation.invalid}</div>
   915→                <div className="text-xs text-muted-foreground">Invalid (skipped)</div>
   916→              </div>
   917→            </div>
   918→            {validation.issues.length > 0 && (
   919→              <div className="rounded-md border border-warning/50 bg-warning/5 p-3 max-h-32 overflow-y-auto">
   920→                <div className="text-xs font-medium text-warning mb-1">Issues:</div>
   921→                <ul className="text-xs text-muted-foreground space-y-0.5">
   922→                  {validation.issues.slice(0, 10).map((iss: string, i: number) => <li key={i}>{iss}</li>)}
   923→                </ul>
   924→              </div>
   925→            )}
   926→            <Button onClick={confirmImport}>Confirm Import ({validation.valid} trades)</Button>
   927→          </div>
   928→        )}
   929→      </DialogContent>
   930→    </Dialog>
   931→  );

   943→    const res = await fetch(`/api/legal/${doc}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({}) });
   944→    if (res.ok) {
   945→      toast.success(`${label} accepted.`);
   946→      const data = await res.json();
   947→      setState({ ...state, accepted: true, acceptedAt: data.acceptedAt, isCurrent: true, version: data.version });
   948→    }
   949→  }
   950→
   951→  // Show the Accept button when: not yet accepted, OR accepted an outdated version.
   952→  const needsAccept = state && (!state.accepted || state.isCurrent === false);
   953→
   954→  return (
   955→    <div className="flex items-center justify-between gap-3 py-2">
   956→      <div>
   957→        <div className="font-medium text-sm">{label}</div>
   958→        <div className="text-xs text-muted-foreground">

   962→              {" · "}
   963→              {state.accepted
   964→                ? `Accepted ${new Date(state.acceptedAt).toLocaleDateString()}`
   965→                : "Not yet accepted"}
   966→              {state.isCurrent === false && (
   967→                <span className="text-warning"> · outdated (v{state.acceptedVersion || "?"})</span>
   968→              )}
   969→            </>
   970→          ) : (
   971→            "Loading…"
   972→          )}
   973→        </div>
   974→      </div>
   975→      <div className="flex gap-2">

   977→        {needsAccept && <Button size="sm" onClick={accept}>Accept</Button>}
   978→      </div>
   979→      <Dialog open={open} onOpenChange={setOpen}>
   980→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
   981→          <DialogHeader>
   982→            <DialogTitle>{label}</DialogTitle>
   983→            {state && (
   984→              <DialogDescription>Version {state.version}</DialogDescription>
   985→            )}
   986→          </DialogHeader>
   987→          <div className="text-sm text-muted-foreground whitespace-pre-wrap leading-relaxed">
   988→            {state?.content ?? "Document text."}
   989→          </div>
   990→        </DialogContent>
   991→      </Dialog>
