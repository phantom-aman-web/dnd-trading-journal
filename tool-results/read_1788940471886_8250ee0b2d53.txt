    92→async function fetchAccounts(): Promise<TradingAccount[]> {
    93→  const res = await fetch("/api/accounts", { cache: "no-store" });
    94→  if (!res.ok) throw new Error("Failed to load accounts");
    95→  const data = await res.json();
    96→  return data.accounts ?? data.items ?? data ?? [];
    97→}
    98→
    99→async function fetchTrades(): Promise<TradeRow[]> {
   100→  const qs = new URLSearchParams({ limit: "200", sortBy: "entryTime", sortDir: "desc" }).toString();
   101→  const res = await fetch(`/api/trades?${qs}`, { cache: "no-store" });
   102→  if (!res.ok) throw new Error("Failed to load trades");
   103→  const data = await res.json();
   104→  return data.items ?? data.trades ?? data ?? [];
   105→}

   135→      const s = map.get(t.accountId) ?? { count: 0, wins: 0, losses: 0, pnlCents: 0 };

   137→      s.pnlCents += t.netPnlCents;
   138→      if (!t.isDraft && t.status !== "open") {
   139→        if (t.netPnlCents > 0) s.wins += 1;
   140→        else if (t.netPnlCents < 0) s.losses += 1;
   141→      }
   142→      map.set(t.accountId, s);
   143→    }

   540→
   541→  const [name, setName] = useState("");

   559→    }

   561→    setName(account?.name ?? "");
   562→    setBroker(account?.broker ?? "");
   563→    setAccountType(account?.accountType ?? "personal");
   564→    setCurrency(account?.currency ?? "USD");

   568→        : "10000",
   569→    );

   571→      account?.consistencyRate != null ? String(account.consistencyRate) : "",
   572→    );

   574→      account?.dailyLossLimitPct != null ? String(account.dailyLossLimitPct) : "",
   575→    );
   576→    setMaxDrawdownPct(
   577→      account?.maxDrawdownPct != null ? String(account.maxDrawdownPct) : "",
   578→    );
   579→    setIsDefault(account?.isDefault ?? false);
   580→    setSeeded(true);
   581→  }, [open, account, seeded]);

   583→  async function submit() {
   584→    if (!name.trim()) {
   585→      toast.error("Account name is required");
   586→      return;
   587→    }
   588→    const balanceNum = Number(balance);
   589→    if (!Number.isFinite(balanceNum)) {
   590→      toast.error("Initial balance must be a valid number");
   591→      return;
   592→    }
   593→    setLoading(true);
   594→    const payload: Record<string, unknown> = {
   595→      name: name.trim(),
   596→      broker: broker.trim() || null,
   597→      accountType,
   598→      currency,
   599→      consistencyRate: consistencyRate === "" ? null : Number(consistencyRate),
   600→      dailyLossLimitPct: dailyLossLimitPct === "" ? null : Number(dailyLossLimitPct),
   601→      maxDrawdownPct: maxDrawdownPct === "" ? null : Number(maxDrawdownPct),
   602→      isDefault,
   603→      startingBalanceCents: Math.round(balanceNum * 100),
   604→    };
   605→
   606→    try {
   607→      const res = isEdit
   608→        ? await fetch(`/api/accounts/${account!.id}`, {
   609→            method: "PATCH",
   610→            headers: { "Content-Type": "application/json" },
   611→            body: JSON.stringify(payload),
   612→          })
   613→        : await fetch("/api/accounts", {
   614→            method: "POST",
   615→            headers: { "Content-Type": "application/json" },
   616→            body: JSON.stringify(payload),
   617→          });
   618→
   619→      if (res.ok) {
   620→        toast.success(isEdit ? "Account updated." : "Account created.");
   621→        onSaved();
   622→      } else {
   623→        const d = await res.json().catch(() => null);
   624→        toast.error(d?.error ?? `Failed to ${isEdit ? "update" : "create"} account.`);
   625→      }
   626→    } catch {
   627→      toast.error("Network error");
   628→    } finally {
   629→      setLoading(false);
   630→    }
   631→  }
