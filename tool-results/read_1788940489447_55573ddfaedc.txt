   414→          // Fetch signed URL
   415→          const detailRes = await fetch(`/api/media/${data.id}`);
   416→          const detail = await detailRes.json();
   417→          setForm((f: any) => ({
   418→            ...f,
   419→            uploadedMedia: [...(f.uploadedMedia ?? []), { ...data, url: detail.url }],
   420→          }));
   421→          toast.success(`${file.name} uploaded.`);

   423→          const err = await res.json().catch(() => ({}));
   424→          toast.error(err?.error ?? `Failed to upload ${file.name}`);

   432→  function setField(path: string, value: any) {
   433→    setForm((f: any) => {
   434→      const next = { ...f };
   435→      const parts = path.split(".");
   436→      let cur: any = next;
   437→      for (let i = 0; i < parts.length - 1; i++) {
   438→        cur[parts[i]] = { ...cur[parts[i]] };
   439→        cur = cur[parts[i]];
   440→      }
   441→      cur[parts[parts.length - 1]] = value;
   442→      return next;
   443→    });
   444→  }

   446→  function toggleArrayField(path: string, value: string) {
   447→    setForm((f: any) => {
   448→      const next = { ...f };
   449→      const parts = path.split(".");
   450→      let cur: any = next;
   451→      for (let i = 0; i < parts.length - 1; i++) {
   452→        cur[parts[i]] = { ...cur[parts[i]] };
   453→        cur = cur[parts[i]];
   454→      }
   455→      const key = parts[parts.length - 1];
   456→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];
   457→      const idx = arr.indexOf(value);
   458→      if (idx >= 0) arr.splice(idx, 1); else arr.push(value);
   459→      cur[key] = arr;
   460→      return next;
   461→    });
   462→  }

   464→  // Update one A+ checklist answer field (`checked` or `note`) for the given
   465→  // item id. Used by the checkbox + evidence-note inputs in the ICT Setup tab.
   466→  function updateChecklistAnswer(
   467→    itemId: string,
   468→    field: "checked" | "note",
   469→    value: boolean | string,
   470→  ) {
   471→    setForm((f: any) => {
   472→      const current = f.checklistAnswers?.[itemId] ?? { checked: false };
   473→      const next = {
   474→        ...f.checklistAnswers,
   475→        [itemId]: {
   476→          ...current,
   477→          [field]: value,
   478→        },
   479→      };
   480→      // If a checked item is being unchecked, drop its note so storage stays clean.
   481→      if (field === "checked" && value === false && next[itemId].note) {
   482→        delete next[itemId].note;
   483→      }
   484→      return { ...f, checklistAnswers: next };
   485→    });

   488→  async function submit(isDraft: boolean = false) {

   490→    if (!form.accountId) {

   499→      setTimeout(() => document.getElementById("field-instrument")?.scrollIntoView({ behavior: "smooth", block: "center" }), 100);

   502→    setSaving(true);
   503→    try {
   504→      const executions = [];
   505→      if (form.entryPrice) {
   506→        executions.push({
   507→          kind: "entry",
   508→          price: form.entryPrice,
   509→          quantity: form.quantity || "1",
   510→          timestamp: form.entryTime ? new Date(form.entryTime).toISOString() : new Date().toISOString(),
   511→        });
   512→      }
   513→      if (form.exitPrice) {
   514→        executions.push({
   515→          kind: "exit",
