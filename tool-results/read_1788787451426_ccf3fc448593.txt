    40→async function fetchTrade(id: string) {
    41→  const res = await fetch(`/api/trades/${id}`, { cache: "no-store" });
    42→  if (!res.ok) throw new Error("Failed to load trade");
    43→  return res.json();
    44→}

   593→function SetupSection({ title, data, isList }: { title: string; data: any; isList?: boolean }) {
   594→  if (!data) return null;
   595→  return (
   596→    <div>
   597→      <div className="text-xs text-muted-foreground uppercase tracking-wide">{title}</div>
   598→      {isList && Array.isArray(data) ? (
   599→        <div className="flex flex-wrap gap-1 mt-1">
   600→          {data.map((d: string, i: number) => <Badge key={i} variant="secondary" className="text-xs">{d}</Badge>)}
   601→        </div>
   602→      ) : typeof data === "object" ? (
   603→        <div className="mt-1 text-sm">
   604→          {Object.entries(data).map(([k, v]) => (
   605→            <div key={k}>
   606→              <span className="text-muted-foreground">{k}:</span> {String(v)}
   607→            </div>
   608→          ))}
   609→        </div>
   610→      ) : (
   611→        <div className="font-medium mt-1">{String(data)}</div>
   612→      )}
   613→    </div>
   614→  );
   615→}
