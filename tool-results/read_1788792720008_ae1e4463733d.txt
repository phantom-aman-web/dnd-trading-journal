   262→function RulesView({ rulesJson }: { rulesJson: string }) {
   263→  let rules: any = {};
   264→  try { rules = JSON.parse(rulesJson ?? "{}"); } catch {}
   265→  const sections = ["entry", "stop", "target", "management", "invalidation"];
   266→  const thresholds = rules.gradingThresholds;
   267→  return (
   268→    <div className="space-y-4">
   269→      {thresholds && (
   270→        <div className="flex flex-wrap gap-2">
   271→          {["A+", "A", "B", "C"].map((g) => (
   272→            <Badge key={g} variant="outline" className="text-xs">
   273→              {g} ≥ {Math.round((thresholds[g] ?? 0) * 100)}%
   274→            </Badge>
   275→          ))}
   276→        </div>
   277→      )}
   278→      <div className="grid sm:grid-cols-2 gap-3 text-sm">
   279→        {sections.map((sec) => {
   280→          const sectionRules = Array.isArray(rules[sec]) ? rules[sec] : [];
   281→          if (sectionRules.length === 0) return null;
   282→          return (
   283→            <div key={sec} className="rounded-md border border-border p-3">
   284→              <div className="text-xs uppercase tracking-wide text-muted-foreground mb-1">{sec}</div>
   285→              <ul className="space-y-1">
   286→                {sectionRules.map((r: any, i: number) => {
   287→                  const text = typeof r === "string" ? r : r.text;
   288→                  const weight = typeof r === "object" ? r.weight : 1;
   289→                  const required = typeof r === "object" ? r.required : true;
   290→                  return (
   291→                    <li key={i} className="text-sm flex items-center gap-2">
   292→                      <span>{text}</span>
   293→                      {required && <Badge variant="outline" className="text-[10px] text-loss">Required</Badge>}
   294→                      <span className="text-[10px] text-muted-foreground ml-auto">×{weight}</span>
   295→                    </li>
   296→                  );
   297→                })}
   298→              </ul>
   299→            </div>
   300→          );
   301→        })}
   302→      </div>
   303→    </div>
   304→  );
   305→}

   307→function NewStrategyDialog({ onClose, onCreated }: { onClose: () => void; onCreated: () => void }) {
   308→  const [name, setName] = useState("");

   310→  const [purpose, setPurpose] = useState("");
   311→  const [instruments, setInstruments] = useState("");
   312→  const [market, setMarket] = useState("forex");
   313→  const [timeframe, setTimeframe] = useState("5m-1h");
   314→  const [session, setSession] = useState("london");
   315→  const [checklistConfigId, setChecklistConfigId] = useState("");
   316→  // Rules: each section is a newline-separated list
   317→  const [entryRules, setEntryRules] = useState("");
   318→  const [stopRules, setStopRules] = useState("");
   319→  const [targetRules, setTargetRules] = useState("");
   320→  const [managementRules, setManagementRules] = useState("");
   321→  const [invalidationRules, setInvalidationRules] = useState("");
   322→  // Weight + required configuration
   323→  const [ruleConfig, setRuleConfig] = useState("");
   324→  // Grading thresholds
   325→  const [aplusThreshold, setAplusThreshold] = useState("0.9");
   326→  const [aThreshold, setAThreshold] = useState("0.75");
   327→  const [bThreshold, setBThreshold] = useState("0.6");
   328→  const [cThreshold, setCThreshold] = useState("0.4");
   329→  const [loading, setLoading] = useState(false);
   330→
   331→  // Fetch checklists for the association selector
   332→  const { data: checklistsData } = useQuery({
   333→    queryKey: ["checklists"],
   334→    queryFn: async () => {
   335→      const res = await fetch("/api/checklists", { cache: "no-store" });
   336→      if (!res.ok) return null;
   337→      return res.json();
   338→    },
   339→  });
   340→  const checklists: any[] = checklistsData?.items ?? [];
   341→
   342→  function parseRules(text: string): string[] {
   343→    return text
   344→      .split("\n")
   345→      .map((l) => l.trim())
   346→      .filter((l) => l.length > 0);
   347→  }
   348→
   349→  // Parse the weight/required config text into a map
   350→  function parseRuleConfig(text: string): Map<string, { weight: number; required: boolean }> {
   351→    const map = new Map<string, { weight: number; required: boolean }>();
   352→    for (const line of text.split("\n")) {
   353→      const parts = line.split("|").map((p) => p.trim());
   354→      if (parts.length >= 1 && parts[0]) {
   355→        const ruleText = parts[0];
   356→        const weight = parts[1] ? parseFloat(parts[1]) || 1 : 1;
   357→        const required = parts[2] ? parts[2].toLowerCase() === "required" : true;
   358→        map.set(ruleText, { weight, required });
   359→      }
   360→    }
   361→    return map;
   362→  }
   363→
   364→  // Build rulesJson with weight/required metadata per rule, plus grading thresholds
   365→  function buildRulesJson() {
   366→    const config = parseRuleConfig(ruleConfig);
   367→    const buildSection = (rules: string) => parseRules(rules).map((rule) => {
   368→      const cfg = config.get(rule) ?? { weight: 1, required: true };
   369→      return { text: rule, weight: cfg.weight, required: cfg.required };
   370→    });
   371→    return {
   372→      entry: buildSection(entryRules),
   373→      stop: buildSection(stopRules),
   374→      target: buildSection(targetRules),
   375→      management: buildSection(managementRules),
   376→      invalidation: buildSection(invalidationRules),
   377→      gradingThresholds: {
   378→        "A+": parseFloat(aplusThreshold) || 0.9,
   379→        A: parseFloat(aThreshold) || 0.75,
   380→        B: parseFloat(bThreshold) || 0.6,
   381→        C: parseFloat(cThreshold) || 0.4,
   382→      },
   383→    };
   384→  }

   389→    const rules = buildRulesJson();
   390→    const res = await fetch("/api/strategies", {
   391→      method: "POST",
   392→      headers: { "Content-Type": "application/json" },
   393→      body: JSON.stringify({
   394→        name,
   395→        description,
   396→        purpose,
   397→        instruments,
   398→        market,
   399→        timeframe,
   400→        session,
   401→        checklistConfigId: checklistConfigId || null,
   402→        rules,
   403→      }),
   404→    });
   405→    if (res.ok) {
   406→      toast.success("Strategy created.");
   407→      onCreated();
   408→      onClose();
   409→    } else {
   410→      toast.error("Failed to create strategy.");
   411→    }
   412→    setLoading(false);
   413→  }
   414→
   415→  return (
   416→    <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   417→      <DialogHeader>
   418→        <DialogTitle>New Strategy</DialogTitle>
   419→      </DialogHeader>

   421→        {/* Basics */}

   430→        <div className="space-y-2">
   431→          <FieldLabel hint="Why this strategy exists">Purpose</FieldLabel>

   433→        </div>
   434→
   435→        {/* Configuration */}
   436→        <div className="grid grid-cols-2 gap-3">

   438→            <FieldLabel hint="Which market this strategy targets">Market</FieldLabel>

   542→function NewVersionDialog({ strategyId, latestLabel, latestRulesJson, onClose, onCreated }: { strategyId: string; latestLabel?: string; latestRulesJson?: string; onClose: () => void; onCreated: () => void }) {
   543→  // Pre-fill rules from the latest version
   544→  const initialRules = (() => {
   545→    try {
   546→      const parsed = JSON.parse(latestRulesJson ?? "{}");
   547→      // Handle both old format (string[]) and new format ({text, weight, required}[])
   548→      const extractText = (arr: any[]) => arr.map((r: any) => typeof r === "string" ? r : r.text);
   549→      const extractConfig = (arr: any[]) => arr.map((r: any) => {
   550→        if (typeof r === "string") return `${r} | 1 | required`;
   551→        return `${r.text} | ${r.weight ?? 1} | ${r.required ? "required" : "optional"}`;
   552→      });
   553→      return {
   554→        entry: Array.isArray(parsed.entry) ? extractText(parsed.entry).join("\n") : "",
   555→        stop: Array.isArray(parsed.stop) ? extractText(parsed.stop).join("\n") : "",
   556→        target: Array.isArray(parsed.target) ? extractText(parsed.target).join("\n") : "",
   557→        management: Array.isArray(parsed.management) ? extractText(parsed.management).join("\n") : "",
   558→        invalidation: Array.isArray(parsed.invalidation) ? extractText(parsed.invalidation).join("\n") : "",
   559→        ruleConfig: Array.isArray(parsed.entry) ? extractConfig([...(parsed.entry||[]), ...(parsed.stop||[]), ...(parsed.target||[]), ...(parsed.management||[]), ...(parsed.invalidation||[])]).join("\n") : "",
   560→        thresholds: parsed.gradingThresholds ?? { "A+": 0.9, A: 0.75, B: 0.6, C: 0.4 },
   561→      };
   562→    } catch {
   563→      return { entry: "", stop: "", target: "", management: "", invalidation: "", ruleConfig: "", thresholds: { "A+": 0.9, A: 0.75, B: 0.6, C: 0.4 } };
   564→    }
   565→  })();
   566→
   567→  const [versionLabel, setVersionLabel] = useState(incrementVersion(latestLabel));
   568→  const [changeReason, setChangeReason] = useState("");
   569→  const [changeSummary, setChangeSummary] = useState("");
   570→  const [entryRules, setEntryRules] = useState(initialRules.entry);
   571→  const [stopRules, setStopRules] = useState(initialRules.stop);
   572→  const [targetRules, setTargetRules] = useState(initialRules.target);
   573→  const [managementRules, setManagementRules] = useState(initialRules.management);
   574→  const [invalidationRules, setInvalidationRules] = useState(initialRules.invalidation);
   575→  const [ruleConfig, setRuleConfig] = useState(initialRules.ruleConfig);
   576→  const [aplusThreshold, setAplusThreshold] = useState(String(initialRules.thresholds["A+"] ?? 0.9));
   577→  const [aThreshold, setAThreshold] = useState(String(initialRules.thresholds.A ?? 0.75));
   578→  const [bThreshold, setBThreshold] = useState(String(initialRules.thresholds.B ?? 0.6));
   579→  const [cThreshold, setCThreshold] = useState(String(initialRules.thresholds.C ?? 0.4));
   580→  const [loading, setLoading] = useState(false);
   581→
   582→  function incrementVersion(v?: string): string {
   583→    if (!v) return "1.0";
   584→    const [maj, min] = v.split(".");
   585→    return `${maj}.${Number(min) + 1}`;
   586→  }
   587→
   588→  function parseRules(text: string): string[] {
   589→    return text.split("\n").map((l) => l.trim()).filter((l) => l.length > 0);
   590→  }
   591→
   592→  function parseRuleConfig(text: string): Map<string, { weight: number; required: boolean }> {
   593→    const map = new Map<string, { weight: number; required: boolean }>();
   594→    for (const line of text.split("\n")) {
   595→      const parts = line.split("|").map((p) => p.trim());
   596→      if (parts.length >= 1 && parts[0]) {
   597→        const ruleText = parts[0];
   598→        const weight = parts[1] ? parseFloat(parts[1]) || 1 : 1;
   599→        const required = parts[2] ? parts[2].toLowerCase() === "required" : true;
   600→        map.set(ruleText, { weight, required });
   601→      }
   602→    }
   603→    return map;
   604→  }
   605→
   606→  function buildRulesJson() {
   607→    const config = parseRuleConfig(ruleConfig);
   608→    const buildSection = (rules: string) => parseRules(rules).map((rule) => {
   609→      const cfg = config.get(rule) ?? { weight: 1, required: true };
   610→      return { text: rule, weight: cfg.weight, required: cfg.required };
   611→    });
   612→    return {
   613→      entry: buildSection(entryRules),
   614→      stop: buildSection(stopRules),
   615→      target: buildSection(targetRules),
   616→      management: buildSection(managementRules),
   617→      invalidation: buildSection(invalidationRules),
   618→      gradingThresholds: {
   619→        "A+": parseFloat(aplusThreshold) || 0.9,
   620→        A: parseFloat(aThreshold) || 0.75,
   621→        B: parseFloat(bThreshold) || 0.6,
   622→        C: parseFloat(cThreshold) || 0.4,
   623→      },
   624→    };
   625→  }

   627→  async function submit() {

   630→    const rules = buildRulesJson();

   632→      method: "PUT",
   633→      headers: { "Content-Type": "application/json" },
   634→      body: JSON.stringify({ versionLabel, changeReason, changeSummary, rules }),
   635→    });
   636→    if (res.ok) {
