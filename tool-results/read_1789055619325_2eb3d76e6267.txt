   152→/** Fetch the user's accounts as a plain array (defensive). Returns [] on
   153→ *  any error or unexpected shape. */
   154→async function fetchAccounts(accountId?: string) {
   155→  try {
   156→    const qs = new URLSearchParams({ ...(accountId ? { accountId } : {}) }).toString();
   157→    const url = `/api/accounts${qs ? `?${qs}` : ""}`;
   158→    const res = await fetch(url, { cache: "no-store" });
   159→    if (!res.ok) return [];
   160→    const data = await res.json();
   161→    const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
   162→    return Array.isArray(arr) ? arr : [];
   163→  } catch {
   164→    return [];
   165→  }
   166→}
   167→
   168→/** Fetch the user's strategies as a plain array (defensive). Returns [] on
   169→ *  any error or unexpected shape. */
   170→async function fetchStrategies() {
   171→  try {
   172→    const res = await fetch("/api/strategies", { cache: "no-store" });
   173→    if (!res.ok) return [];

   182→/** Extract a flat `ChecklistItem[]` from a strategy version's rulesJson.
   183→ *  Supports both:
   184→ *    - flat array:   [{ title, required, description, weight }, ...]
   185→ *                    [{ text,   required, weight }, ...]
   186→ *                    ["plain string rule", ...]
   187→ *    - legacy object: { entry: [...], stop: [...], target: [...],
   188→ *                       management: [...], invalidation: [...] }
   189→ */
   190→function extractChecklist(rulesJson: string | null | undefined): ChecklistItem[] {
   191→  if (!rulesJson) return [];
   192→  let parsed: unknown;
   193→  try {
   194→    parsed = JSON.parse(rulesJson);
   195→  } catch {
   196→    return [];
   197→  }
   198→  const items: ChecklistItem[] = [];
   199→  let idx = 0;
   200→
   201→  const pushRule = (rule: unknown) => {
   202→    if (typeof rule === "string") {
   203→      const text = rule.trim();
   204→      if (!text) return;
   205→      items.push({
   206→        id: `rule_${idx}`,
   207→        text,
   208→        required: true,
   209→        weight: 1,
   210→        evidenceRequired: false,
   211→      });
   212→      idx++;
   213→      return;
   214→    }
   215→    if (rule && typeof rule === "object") {
   216→      const r = rule as Record<string, unknown>;
   217→      const text = String(r.text ?? r.title ?? "").trim();
   218→      if (!text) return;
   219→      const weightNum = typeof r.weight === "number" && Number.isFinite(r.weight) ? r.weight : 1;
   220→      items.push({
   221→        id: `rule_${idx}`,
   222→        text,
   223→        required: r.required !== false,
   224→        weight: Math.max(0, weightNum),
   225→        evidenceRequired: r.evidenceRequired === true,
   226→      });
   227→      idx++;
   228→    }
   229→  };
   230→
   231→  if (Array.isArray(parsed)) {
   232→    parsed.forEach(pushRule);
   233→    return items;
   234→  }
   235→  if (parsed && typeof parsed === "object") {
   236→    const obj = parsed as Record<string, unknown>;
   237→    // New playbooks-view format: { items: [...] }
   238→    if (Array.isArray(obj.items)) {
   239→      obj.items.forEach(pushRule);
   240→      return items;
   241→    }
   242→    // Legacy category object: { entry, stop, target, management, invalidation }
   243→    for (const section of ["entry", "stop", "target", "management", "invalidation"]) {
   244→      const arr = obj[section];
   245→      if (Array.isArray(arr)) arr.forEach(pushRule);
   246→    }
   247→    return items;
   248→  }
   249→  return [];
   250→}

   252→/** Parse a JSON string into the typed fallback; never throws. */
   253→function safeJsonParse(json: string | null | undefined, fallback: any): any {
   254→  if (!json) return fallback;
   255→  try { return JSON.parse(json); } catch { return fallback; }
   256→}
   257→
   258→/** Format a numeric price with the given decimal precision (default 5,
   259→ *  matching the historical forex-major behaviour). Returns "—" when the
   260→ *  value is missing or non-finite. Pass the selected instrument's
   261→ *  `pricePrecision` to render prices with the right number of decimals
   262→ *  (e.g. 2 for gold, 1 for indices, 0 for YM). */
   263→function fmtPrice(
   264→  value: number | string | null | undefined,
   265→  precision: number = 5,
   266→): string {
   267→  if (value == null || value === "") return "—";
   268→  const n = typeof value === "string" ? Number(value) : value;
   269→  if (!Number.isFinite(n)) return "—";
   270→  const p =
   271→    Number.isFinite(precision) && precision >= 0 && precision <= 8
   272→      ? Math.round(precision)
   273→      : 5;
   274→  return n.toFixed(p);
   275→}
   276→
   277→/** Format a money amount given a dollars-number (NOT cents). */
   278→function fmtMoney(dollars: number | null | undefined, currency = "USD"): string {
   279→  if (dollars == null || !Number.isFinite(dollars)) return "—";
   280→  try {
   281→    return new Intl.NumberFormat("en-US", {
   282→      style: "currency",
   283→      currency,
   284→      minimumFractionDigits: 2,
   285→      maximumFractionDigits: 2,
   286→    }).format(dollars);
   287→  } catch {
   288→    return `$${dollars.toFixed(2)}`;
   289→  }
   290→}
   291→
   292→/** Generate a short random id for client-side list items. */
   293→function uid(): string {
   294→  return Math.random().toString(36).slice(2, 10);
   295→}

   748→  /* ---------- File upload (non-blocking, concurrent, with progress) ----------
   749→   *
   750→   * Each selected file is added to `form.uploadedMedia` immediately as a

   754→   * upload in a sequential for-loop). The File + XHR for each tempId live in

   759→  function handleFileUpload(files: FileList | File[]) {
   760→    const fileList = Array.from(files);
   761→    for (const file of fileList) {
   762→      const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   763→      // Detect kind from MIME type so the form immediately knows whether to
   764→      // render an <img> (annotation-able) or a <video controls> element.
   765→      const clientKind = file.type.startsWith("video/") ? "video" : "image";
   766→      const mediaItem: any = {
   767→        tempId,
   768→        filename: file.name,
   769→        originalName: file.name,
   770→        kind: clientKind,
   771→        status: "uploading",
   772→        progress: 0,
   773→        fileSize: file.size,
   774→        annotations: [],
   775→        caption: "",
   776→        timeframe: "",
   777→      };

   779→      // away, then kick off the upload in the background (non-blocking).
   780→      setForm((f: any) => ({
   781→        ...f,
   782→        uploadedMedia: [...(f.uploadedMedia ?? []), mediaItem],
   783→      }));
   784→      uploadFile(file, tempId);
   785→    }

   788→  function uploadFile(file: File, tempId: string) {
   789→    const formData = new FormData();
   790→    formData.append("file", file);
   791→    if (editingId) formData.append("tradeId", editingId);
   792→
   793→    const xhr = new XMLHttpRequest();
   794→    xhr.open("POST", "/api/media");
   795→    fileRefs.current.set(tempId, { file, xhr });
   796→
   797→    // Real upload progress — only XMLHttpRequest exposes this; fetch()
   798→    // ReadableStream request bodies don't have progress events yet.
   799→    xhr.upload.onprogress = (e) => {
   800→      if (e.lengthComputable) {
   801→        const progress = Math.round((e.loaded / e.total) * 100);
   802→        setForm((f: any) => ({
   803→          ...f,
   804→          uploadedMedia: f.uploadedMedia.map((m: any) =>
   805→            m.tempId === tempId ? { ...m, progress } : m,
   806→          ),
   807→        }));
   808→      }
   809→    };
   810→
   811→    xhr.onload = () => {
   812→      if (xhr.status >= 200 && xhr.status < 300) {
   813→        try {
   814→          const data = JSON.parse(xhr.responseText);
   815→          // Fetch the signed URL so the preview can render immediately.
   816→          fetch(`/api/media/${data.id}`)
   817→            .then((r) => r.json())
   818→            .then((detail) => {
   819→              setForm((f: any) => ({
   820→                ...f,
   821→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   822→                  m.tempId === tempId
   823→                    ? {
   824→                        ...m,
   825→                        ...data,
   826→                        id: data.id,
   827→                        url: detail.url,
   828→                        status: "uploaded",
   829→                        progress: 100,
   830→                      }
   831→                    : m,
   832→                ),
   833→              }));
   834→            })
   835→            .catch(() => {
   836→              // Signed-URL fetch failed — still mark uploaded with raw data.
   837→              setForm((f: any) => ({
   838→                ...f,
   839→                uploadedMedia: f.uploadedMedia.map((m: any) =>
   840→                  m.tempId === tempId
   841→                    ? {
   842→                        ...m,
   843→                        ...data,
   844→                        id: data.id,
   845→                        status: "uploaded",
   846→                        progress: 100,
   847→                      }
   848→                    : m,
   849→                ),
   850→              }));
   851→            });
   852→          // Upload finished — drop the File/XHR ref (Retry not needed).
   853→          fileRefs.current.delete(tempId);
   854→          toast.success(`${file.name} uploaded.`);
   855→        } catch {
   856→          markUploadFailed(tempId, file.name);
   857→        }
   858→      } else {
   859→        // Surface the server's error message when present (e.g. "File too
   860→        // large. Max 20 MB.").
   861→        let msg = `Failed to upload ${file.name}`;
   862→        try {
   863→          const err = JSON.parse(xhr.responseText);
   864→          if (err?.error) msg = err.error;
   865→        } catch {}
   866→        markUploadFailed(tempId, file.name, msg);
   867→      }
   868→    };
   869→
   870→    xhr.onerror = () => markUploadFailed(tempId, file.name);
   871→    // onabort is fired when cancelUpload calls xhr.abort(); cancelUpload
   872→    // already removes the item from state and clears the ref, so this is
   873→    // a no-op kept for completeness.
   874→    xhr.onabort = () => {};
   875→
   876→    xhr.send(formData);
   877→  }

   981→  /* ---------- Risk & sizing calculations ---------- */
   982→
   983→  const account = accounts.find((a: any) => a.id === form.accountId);
   984→  const accountCurrency = account?.currency ?? "USD";
   985→  const balanceCents = account?.currentBalanceCents ?? account?.startingBalanceCents ?? 0;
   986→
   987→  const entryNum = Number(form.plannedEntryPrice);
   988→  const stopNum = Number(form.plannedStopPrice);
   989→  const targetNum = Number(form.plannedTargetPrice);
   990→  const lotNum = Number(form.lotSize) || 1;
   991→  const riskPctNum = Number(form.riskPct);
   992→
   993→  const hasEntry = Number.isFinite(entryNum) && form.plannedEntryPrice !== "";
   994→  const hasStop = Number.isFinite(stopNum) && form.plannedStopPrice !== "";
   995→  const hasTarget = Number.isFinite(targetNum) && form.plannedTargetPrice !== "";
   996→
   997→  // Stop distance (always positive).
   998→  const stopDistance = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   999→
  1000→  // Risk per unit / reward per unit (per 1 lot).
  1001→  const riskPerUnit = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
  1002→  const rewardPerUnit =
  1003→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;
  1004→
  1005→  // Planned R:R = reward / risk (1.0 = breakeven, 2.0 = 2R for 1R).
  1006→  const plannedRR =
  1007→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
  1008→      ? rewardPerUnit / riskPerUnit
  1009→      : null;
  1010→
  1011→  // Risk amount ($) = balance * riskPct.
  1012→  const riskAmountDollars =
  1013→    Number.isFinite(riskPctNum) && balanceCents > 0
  1014→      ? (balanceCents / 100) * riskPctNum
  1015→      : null;
  1016→
  1017→  // Planned profit ($) = riskAmount * RR (if RR is known).
  1018→  const plannedProfitDollars =
  1019→    riskAmountDollars != null && plannedRR != null
  1020→      ? riskAmountDollars * plannedRR
  1021→      : null;
  1022→
  1023→  /* ---------- Instrument-aware risk preview ---------- */
  1024→  // Uses pipSize, contractSize, and pointValueCents from the catalog to
  1025→  // express the stop distance in the instrument's native unit (pips /
  1026→  // ticks / points) and to derive a per-instrument dollar risk that is
  1027→  // consistent with the server-side calculateTradePnl formula:
  1028→  //   risk$ = stopDistance × qty × contractSize × (pointValueCents/100)
  1029→  // (See computePointValueCents in src/lib/calculations.ts for the
  1030→  // convention used to derive pointValueCents from cs × tickSize.)
  1031→  const stopDistanceUnits = stopDistance != null
  1032→    ? distanceInUnits(stopDistance, selectedInstrument.pipSize)
  1033→    : 0;
  1034→  const contractSizeNum = Number(selectedInstrument.contractSize) || 1;
  1035→  const pointValueCents = selectedInstrument.pointValueCents;
  1036→  const instrumentRiskDollars =
  1037→    stopDistance != null
  1038→      ? stopDistance * lotNum * contractSizeNum * (pointValueCents / 100)
  1039→      : null;
  1040→
  1041→  /* ---------- Partial exits calculations ---------- */
  1042→
  1043→  const validExits = form.partialExits.filter(
  1044→    (e: any) => e.price !== "" && Number.isFinite(Number(e.price)),
  1045→  );
  1046→  const weightedExitPrice =
  1047→    validExits.length > 0
  1048→      ? validExits.reduce((sum: number, e: any) => sum + Number(e.price), 0) /
  1049→        validExits.length
  1050→      : null;
  1051→  const totalRExits =
  1052→    weightedExitPrice != null && riskPerUnit != null && riskPerUnit > 0
  1053→      ? form.direction === "long"
  1054→        ? (weightedExitPrice - entryNum) / riskPerUnit
  1055→        : (entryNum - weightedExitPrice) / riskPerUnit
  1056→      : null;
  1057→
  1058→  /* ---------- Evidence grouping (by chart timeframe) ---------- */
  1059→
  1060→  // Bucket each uploaded media item into a timeframe group so the user can
  1061→  // see at a glance whether their HTF context, execution chart, and entry
  1062→  // trigger are all captured. Items with no timeframe selected land in an
  1063→  // "Unassigned" bucket first to nudge the user toward picking one.
  1064→  const groupedMedia = useMemo(() => {
  1065→    const items: Array<{ m: any; i: number }> = (form.uploadedMedia ?? []).map(
  1066→      (m: any, i: number) => ({ m, i }),
  1067→    );
  1068→    const buckets: Record<string, Array<{ m: any; i: number }>> = {};
  1069→    for (const entry of items) {
  1070→      const g = entry.m.timeframe ? getTimeframeGroup(entry.m.timeframe) : null;
  1071→      const key = g ?? "__unassigned";
  1072→      if (!buckets[key]) buckets[key] = [];
  1073→      buckets[key].push(entry);
  1074→    }
  1075→    const ordered: Array<{ group: string; items: Array<{ m: any; i: number }> }> = [];
  1076→    if (buckets["__unassigned"]) {
  1077→      ordered.push({ group: "Unassigned", items: buckets["__unassigned"] });
  1078→    }
  1079→    for (const g of TIMEFRAME_GROUPS) {
  1080→      if (buckets[g]) ordered.push({ group: g, items: buckets[g] });
  1081→    }
  1082→    // Any leftover custom groups (shouldn't normally happen, but defensive).
  1083→    for (const key of Object.keys(buckets)) {
  1084→      if (key === "__unassigned") continue;
  1085→      if (TIMEFRAME_GROUPS.includes(key)) continue;
  1086→      ordered.push({ group: key, items: buckets[key] });
  1087→    }
  1088→    return ordered;
  1089→  }, [form.uploadedMedia]);

  1103→    if (!form.instrumentSymbol) {
