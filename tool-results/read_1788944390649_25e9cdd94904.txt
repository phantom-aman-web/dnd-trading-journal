   139→/** Fetch the user's accounts as a plain array (defensive). Returns [] on
   140→ *  any error or unexpected shape. */
   141→async function fetchAccounts(accountId?: string) {
   142→  try {
   143→    const qs = new URLSearchParams({ ...(accountId ? { accountId } : {}) }).toString();
   144→    const url = `/api/accounts${qs ? `?${qs}` : ""}`;
   145→    const res = await fetch(url, { cache: "no-store" });
   146→    if (!res.ok) return [];
   147→    const data = await res.json();
   148→    const arr = data?.items ?? data?.accounts ?? (Array.isArray(data) ? data : []);
   149→    return Array.isArray(arr) ? arr : [];
   150→  } catch {
   151→    return [];
   152→  }
   153→}

   155→/** Fetch the user's strategies as a plain array (defensive). Returns [] on

   169→/** Extract a flat `ChecklistItem[]` from a strategy version's rulesJson.
   170→ *  Supports both:
   171→ *    - flat array:   [{ title, required, description, weight }, ...]
   172→ *                    [{ text,   required, weight }, ...]
   173→ *                    ["plain string rule", ...]
   174→ *    - legacy object: { entry: [...], stop: [...], target: [...],
   175→ *                       management: [...], invalidation: [...] }
   176→ */
   177→function extractChecklist(rulesJson: string | null | undefined): ChecklistItem[] {
   178→  if (!rulesJson) return [];
   179→  let parsed: unknown;
   180→  try {
   181→    parsed = JSON.parse(rulesJson);
   182→  } catch {
   183→    return [];
   184→  }
   185→  const items: ChecklistItem[] = [];
   186→  let idx = 0;
   187→
   188→  const pushRule = (rule: unknown) => {
   189→    if (typeof rule === "string") {
   190→      const text = rule.trim();
   191→      if (!text) return;
   192→      items.push({
   193→        id: `rule_${idx}`,
   194→        text,
   195→        required: true,
   196→        weight: 1,
   197→        evidenceRequired: false,
   198→      });
   199→      idx++;
   200→      return;
   201→    }
   202→    if (rule && typeof rule === "object") {
   203→      const r = rule as Record<string, unknown>;
   204→      const text = String(r.text ?? r.title ?? "").trim();
   205→      if (!text) return;
   206→      const weightNum = typeof r.weight === "number" && Number.isFinite(r.weight) ? r.weight : 1;
   207→      items.push({
   208→        id: `rule_${idx}`,
   209→        text,
   210→        required: r.required !== false,
   211→        weight: Math.max(0, weightNum),
   212→        evidenceRequired: r.evidenceRequired === true,
   213→      });
   214→      idx++;
   215→    }
   216→  };
   217→
   218→  if (Array.isArray(parsed)) {
   219→    parsed.forEach(pushRule);
   220→    return items;
   221→  }
   222→  if (parsed && typeof parsed === "object") {
   223→    const obj = parsed as Record<string, unknown>;
   224→    // New playbooks-view format: { items: [...] }
   225→    if (Array.isArray(obj.items)) {
   226→      obj.items.forEach(pushRule);
   227→      return items;
   228→    }
   229→    // Legacy category object: { entry, stop, target, management, invalidation }
   230→    for (const section of ["entry", "stop", "target", "management", "invalidation"]) {
   231→      const arr = obj[section];
   232→      if (Array.isArray(arr)) arr.forEach(pushRule);
   233→    }
   234→    return items;
   235→  }
   236→  return [];
   237→}

   239→/** Parse a JSON string into the typed fallback; never throws. */
   240→function safeJsonParse(json: string | null | undefined, fallback: any): any {
   241→  if (!json) return fallback;
   242→  try { return JSON.parse(json); } catch { return fallback; }
   243→}
   244→
   245→/** Format a numeric price with 5-dp precision. Returns "—" when not finite. */
   246→function fmtPrice(value: number | string | null | undefined): string {
   247→  if (value == null || value === "") return "—";
   248→  const n = typeof value === "string" ? Number(value) : value;
   249→  if (!Number.isFinite(n)) return "—";
   250→  return n.toFixed(5);
   251→}
   252→
   253→/** Format a money amount given a dollars-number (NOT cents). */
   254→function fmtMoney(dollars: number | null | undefined, currency = "USD"): string {
   255→  if (dollars == null || !Number.isFinite(dollars)) return "—";
   256→  try {
   257→    return new Intl.NumberFormat("en-US", {
   258→      style: "currency",
   259→      currency,
   260→      minimumFractionDigits: 2,
   261→      maximumFractionDigits: 2,
   262→    }).format(dollars);
   263→  } catch {
   264→    return `$${dollars.toFixed(2)}`;
   265→  }
   266→}
   267→
   268→/** Generate a short random id for client-side list items. */
   269→function uid(): string {
   270→  return Math.random().toString(36).slice(2, 10);
   271→}

   621→      let cur: any = next;

   623→        cur[parts[i]] = { ...cur[parts[i]] };
   624→        cur = cur[parts[i]];
   625→      }

   627→      const arr = Array.isArray(cur[key]) ? [...cur[key]] : [];

   739→  /* ---------- Risk & sizing calculations ---------- */
   740→
   741→  const account = accounts.find((a: any) => a.id === form.accountId);
   742→  const accountCurrency = account?.currency ?? "USD";
   743→  const balanceCents = account?.currentBalanceCents ?? account?.startingBalanceCents ?? 0;
   744→
   745→  const entryNum = Number(form.plannedEntryPrice);
   746→  const stopNum = Number(form.plannedStopPrice);
   747→  const targetNum = Number(form.plannedTargetPrice);
   748→  const lotNum = Number(form.lotSize) || 1;
   749→  const riskPctNum = Number(form.riskPct);
   750→
   751→  const hasEntry = Number.isFinite(entryNum) && form.plannedEntryPrice !== "";
   752→  const hasStop = Number.isFinite(stopNum) && form.plannedStopPrice !== "";
   753→  const hasTarget = Number.isFinite(targetNum) && form.plannedTargetPrice !== "";
   754→
   755→  // Stop distance (always positive).
   756→  const stopDistance = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   757→
   758→  // Risk per unit / reward per unit (per 1 lot).
   759→  const riskPerUnit = hasEntry && hasStop ? Math.abs(entryNum - stopNum) : null;
   760→  const rewardPerUnit =
   761→    hasEntry && hasTarget ? Math.abs(targetNum - entryNum) : null;
   762→
   763→  // Planned R:R = reward / risk (1.0 = breakeven, 2.0 = 2R for 1R).
   764→  const plannedRR =
   765→    riskPerUnit && rewardPerUnit && riskPerUnit > 0
   766→      ? rewardPerUnit / riskPerUnit
   767→      : null;
   768→
   769→  // Risk amount ($) = balance * riskPct.
   770→  const riskAmountDollars =
   771→    Number.isFinite(riskPctNum) && balanceCents > 0
   772→      ? (balanceCents / 100) * riskPctNum
   773→      : null;
   774→
   775→  // Planned profit ($) = riskAmount * RR (if RR is known).
   776→  const plannedProfitDollars =
   777→    riskAmountDollars != null && plannedRR != null
   778→      ? riskAmountDollars * plannedRR
   779→      : null;
   780→
   781→  /* ---------- Partial exits calculations ---------- */
   782→
   783→  const validExits = form.partialExits.filter(
   784→    (e: any) => e.price !== "" && Number.isFinite(Number(e.price)),
   785→  );
   786→  const weightedExitPrice =
   787→    validExits.length > 0
   788→      ? validExits.reduce((sum: number, e: any) => sum + Number(e.price), 0) /
   789→        validExits.length
   790→      : null;
   791→  const totalRExits =
   792→    weightedExitPrice != null && riskPerUnit != null && riskPerUnit > 0
   793→      ? form.direction === "long"
   794→        ? (weightedExitPrice - entryNum) / riskPerUnit
   795→        : (entryNum - weightedExitPrice) / riskPerUnit
   796→      : null;
