   112→async function fetchDashboard(preset: string, accountId?: string): Promise<DashboardResponse> {
   113→  const params = new URLSearchParams({ preset });
   114→  if (accountId) params.set("accountId", accountId);
   115→  const res = await fetch(`/api/dashboard?${params.toString()}`, { cache: "no-store" });
   116→  if (!res.ok) throw new Error("Failed to load dashboard");
   117→  return res.json();
   118→}
   119→
   120→async function fetchRecentTrades(accountId?: string): Promise<TradesResponse> {
   121→  const params = new URLSearchParams({ limit: "10", sortBy: "entryTime", sortDir: "desc" });
   122→  if (accountId) params.set("accountId", accountId);
   123→  const res = await fetch(`/api/trades?${params.toString()}`, { cache: "no-store" });
   124→  if (!res.ok) throw new Error("Failed to load trades");
   125→  return res.json();
   126→}

   132→function safeParseArr(json: string | null | undefined): string[] {
   133→  if (!json) return [];
   134→  try {
   135→    const v = JSON.parse(json);
   136→    return Array.isArray(v) ? v : [];
   137→  } catch {
   138→    return [];
   139→  }
   140→}
   141→
   142→/** Per-trade adherence: 100% when no behavior flags are recorded, else 0%. */
   143→function tradeAdherencePct(trade: TradeRow): number {
   144→  return safeParseArr(trade.behaviorFlagsJson).length === 0 ? 100 : 0;
   145→}
   146→
   147→function formatDate(iso: string | null): string {
   148→  if (!iso) return "—";
   149→  const d = new Date(iso);
   150→  if (Number.isNaN(d.getTime())) return "—";
   151→  return d.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "2-digit" });
   152→}

   158→export function DashboardView() {

   166→  });
   167→
   168→  const { data: tradesData, isLoading: tradesLoading } = useQuery({
   169→    queryKey: ["dashboard-recent-trades", accountId],
   170→    queryFn: () => fetchRecentTrades(accountId),
   171→  });
   172→
   173→  const recentTrades: TradeRow[] = Array.isArray(tradesData) ? tradesData : (tradesData?.items ?? []);
   174→
   175→  // Adherence breakdown derived from BOTH the dashboard's period-correct
   176→  // aggregate + byBehavior (rule violations + top violated rules) AND the
   177→  // recent-trades snapshot (recent adherence rate).
   178→  const breakdown = useMemo(() => {
   179→    const totalTrades = dash?.aggregate.totalTrades ?? 0;
   180→    const byBehavior = dash?.byBehavior ?? [];
   181→
   182→    // Total violation instances (a trade with 2 flags counts as 2).
   183→    const totalViolations = byBehavior.reduce((s, b) => s + (b.trades ?? 0), 0);
   184→
   185→    // Top 3 violated rules sorted by trade count, then P&L impact.
   186→    const topViolated = [...byBehavior]
   187→      .sort((a, b) => b.trades - a.trades || Math.abs(b.pnlCents) - Math.abs(a.pnlCents))
   188→      .slice(0, 3);
   189→
   190→    // Recent-trades adherence snapshot — fraction of the recent 10 trades
   191→    // that have zero behavior flags. Reported as a percentage.
   192→    const recentCount = recentTrades.length;
   193→    const recentCompliant = recentTrades.filter(
   194→      (t) => safeParseArr(t.behaviorFlagsJson).length === 0,
   195→    ).length;
   196→    const recentAdherenceRate =
   197→      recentCount > 0 ? Math.round((recentCompliant / recentCount) * 100) : null;
   198→
   199→    return { totalTrades, totalViolations, topViolated, recentAdherenceRate, recentCount };
   200→  }, [dash, recentTrades]);
   201→
   202→  const isLoading = dashLoading || tradesLoading;
   203→  const hasNoData = !isLoading && breakdown.totalTrades === 0 && recentTrades.length === 0;
   204→
   205→  if (isLoading) {
   206→    return <DashboardSkeleton />;
   207→  }
   208→
   209→  if (hasNoData) {
   210→    return (
   211→      <div className="p-4 md:p-6 space-y-6">
   212→        <DashboardHeader preset={preset} setPreset={setPreset} />
   213→        <EmptyDashboard onLogTrade={() => navigate("tradeNew")} />
   214→      </div>
   215→    );
   216→  }
   217→
   218→  return (
   219→    <div className="p-4 md:p-6 space-y-6">
   220→      <DashboardHeader preset={preset} setPreset={setPreset} />
   221→
   222→      {/* Section 1 — Process & Adherence Breakdown */}
   223→      <section className="space-y-3">
   224→        <SectionTitle
   225→          icon={ShieldCheck}
   226→          title="PROCESS & ADHERENCE BREAKDOWN"
   227→          subtitle="How well are you following your trading checklist?"
   228→        />
   229→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   230→          <MetricTile
   231→            label="Total Trades"
   232→            value={String(breakdown.totalTrades)}
   233→            sub={dash?.aggregate ? `${dash.aggregate.wins}W · ${dash.aggregate.losses}L` : undefined}
   234→            icon={ClipboardList}
   235→          />
   236→          <MetricTile
   237→            label="Adherence Rate"
   238→            value={
   239→              breakdown.recentAdherenceRate != null
   240→                ? `${breakdown.recentAdherenceRate}%`
   241→                : "—"
   242→            }
   243→            sub={
   244→              breakdown.recentCount > 0
   245→                ? `last ${breakdown.recentCount} trades`
   246→                : undefined
   247→            }
   248→            icon={CheckCircle2}
   249→            tone={
   250→              breakdown.recentAdherenceRate == null
   251→                ? "neutral"
   252→                : breakdown.recentAdherenceRate >= 80
   253→                  ? "profit"
   254→                  : breakdown.recentAdherenceRate < 50
   255→                    ? "loss"
   256→                    : "neutral"
   257→            }
   258→          />
   259→          <MetricTile
   260→            label="Rule Violations"
   261→            value={String(breakdown.totalViolations)}
   262→            sub={
   263→              breakdown.totalViolations > 0
   264→                ? `${dash?.byBehavior?.length ?? 0} distinct rule${(dash?.byBehavior?.length ?? 0) === 1 ? "" : "s"}`
   265→                : "no violations"
   266→            }
   267→            icon={AlertTriangle}
   268→            tone={breakdown.totalViolations > 0 ? "loss" : "profit"}
   269→          />
   270→          <MetricTile
   271→            label="Net P&L"
   272→            value={formatSignedCents(dash?.aggregate.totalPnlCents ?? 0).text}
   273→            sub={dash?.aggregate ? `${dash.aggregate.closedTrades} closed` : undefined}
   274→            icon={Activity}
   275→            tone={
   276→              (dash?.aggregate.totalPnlCents ?? 0) > 0
   277→                ? "profit"
   278→                : (dash?.aggregate.totalPnlCents ?? 0) < 0
   279→                  ? "loss"
   280→                  : "neutral"
   281→            }
   282→          />
   283→        </div>
   284→
   285→        {/* Top violated rules */}
   286→        <Card className="bg-card border border-border rounded-xl shadow-sm p-4">
   287→          <div className="flex items-center justify-between mb-3">
   288→            <h3 className="text-sm font-semibold text-foreground">Top Violated Rules</h3>
   289→            <Badge
   290→              variant="outline"
   291→              className="bg-muted/50 text-muted-foreground border-border text-[10px] uppercase tracking-wide"
   292→            >
   293→              {breakdown.topViolated.length} rule{breakdown.topViolated.length === 1 ? "" : "s"}
   294→            </Badge>
   295→          </div>
   296→          {breakdown.topViolated.length === 0 ? (
   297→            <div className="flex items-center gap-2 py-6 text-sm text-muted-foreground">

   326→                    >
   327→                      {pnl.text}

   675→/**
   676→ * PERFORMANCE OVERVIEW — 8 MetricCards + 3 charts (Equity Curve, Daily P&L,
   677→ * R Distribution). Renders only when aggregate data exists and the relevant
   678→ * chart data arrays have length > 0; otherwise the whole section is skipped
   679→ * to avoid empty chart placeholders.
   680→ */
   681→function PerformanceOverview({
   682→  aggregate,
   683→  equityCurve,
   684→  dailyPnl,
   685→  rDistribution,
   686→  startingBalanceCents,
   687→}: {
   688→  aggregate: NonNullable<DashboardResponse["aggregate"]>;
   689→  equityCurve: NonNullable<DashboardResponse["equityCurve"]>;
   690→  dailyPnl: NonNullable<DashboardResponse["dailyPnl"]>;
   691→  rDistribution: NonNullable<DashboardResponse["rDistribution"]>;
   692→  startingBalanceCents: number;
   693→}) {
   694→  const pnl = aggregate.totalPnlCents ?? 0;
   695→  const pf = aggregate.profitFactor ? Number(aggregate.profitFactor) : null;
   696→  const expectancy = aggregate.expectancyR ? Number(aggregate.expectancyR) : null;
   697→
   698→  const hasEquity = (equityCurve?.length ?? 0) > 0;
   699→  const hasDaily = (dailyPnl?.length ?? 0) > 0;
   700→  const hasRDist = (rDistribution?.length ?? 0) > 0;
   701→  if (!hasEquity && !hasDaily && !hasRDist) return null;
   702→
   703→  return (
   704→    <section className="space-y-3">
   705→      <SectionTitle
   706→        icon={TrendingUp}
   707→        title="PERFORMANCE OVERVIEW"
   708→        subtitle="Aggregate performance and key metrics for this period."
   709→      />
   710→
   711→      <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-4 gap-3">
   712→        <MetricCard
   713→          label="Net P&L"
   714→          value={formatSignedCents(pnl).text}
   715→          tone={pnl > 0 ? "profit" : pnl < 0 ? "loss" : "neutral"}
   716→          icon={Activity}
   717→        />
   718→        <MetricCard
   719→          label="Win Rate"
   720→          value={aggregate.winRate ? formatPct(aggregate.winRate) : "N/A"}
   721→          tone="neutral"
   722→          icon={Target}
   723→        />
   724→        <MetricCard
   725→          label="Avg R"
   726→          value={formatR(aggregate.avgR)}
   727→          tone={aggregate.avgR && Number(aggregate.avgR) > 0 ? "profit" : aggregate.avgR && Number(aggregate.avgR) < 0 ? "loss" : "neutral"}
   728→          icon={TrendingUp}
   729→        />
   730→        <MetricCard
   731→          label="Trade Count"
   732→          value={String(aggregate.totalTrades ?? 0)}
   733→          tone="neutral"
   734→          icon={ClipboardList}
   735→        />
   736→        <MetricCard
   737→          label="Profit Factor"
   738→          value={pf != null ? pf.toFixed(2) : "N/A"}
   739→          tone={pf != null && pf >= 1 ? "profit" : pf != null ? "loss" : "neutral"}
   740→          icon={Activity}
   741→        />
   742→        <MetricCard
   743→          label="Expectancy"
   744→          value={formatR(aggregate.expectancyR)}
   745→          tone={expectancy != null && expectancy > 0 ? "profit" : expectancy != null && expectancy < 0 ? "loss" : "neutral"}
   746→          icon={TrendingUp}
   747→        />
   748→        <MetricCard
   749→          label="Avg Win"
   750→          value={aggregate.avgWinCents != null ? formatSignedCents(aggregate.avgWinCents).text : "N/A"}
   751→          tone="profit"
   752→          icon={TrendingUp}
   753→        />
   754→        <MetricCard
   755→          label="Avg Loss"
   756→          value={aggregate.avgLossCents != null ? formatSignedCents(aggregate.avgLossCents).text : "N/A"}
   757→          tone="loss"
   758→          icon={AlertTriangle}
   759→        />
   760→      </div>
   761→
   762→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-3">
   763→        {hasEquity && (
   764→          <Card className="bg-card border border-border rounded-xl shadow-sm p-4">
   765→            <h3 className="text-sm font-semibold text-foreground mb-3">Equity Curve</h3>
   766→            <EquityCurveChart data={equityCurve} startingBalanceCents={startingBalanceCents} />
   767→          </Card>
   768→        )}
   769→        {hasDaily && (
   770→          <Card className="bg-card border border-border rounded-xl shadow-sm p-4">
   771→            <h3 className="text-sm font-semibold text-foreground mb-3">Daily P&L</h3>
   772→            <DailyPnlChart data={dailyPnl} />
   773→          </Card>
   774→        )}
   775→        {hasRDist && (
   776→          <Card className="bg-card border border-border rounded-xl shadow-sm p-4">
   777→            <h3 className="text-sm font-semibold text-foreground mb-3">R Distribution</h3>
   778→            <RDistributionChart data={rDistribution} />
   779→          </Card>
   780→        )}
   781→      </div>
   782→    </section>
   783→  );
   784→}
